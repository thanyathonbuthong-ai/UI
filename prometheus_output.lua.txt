        if (value13.Parent.Position - vector4).Magnitude < 10 then
          return true
        end
      end
    end

    return false
  end

  function runMainLoop()
    local v30 = "INIT"

    while _G.AutoFarmEnabled do
      task.wait(0.3)

      pcall(function()
        local mainMenu = playerGui:FindFirstChild("MainMenu")
        local remotes2 = replicatedStorage:FindFirstChild("Remotes")
        local gameFolder = workspace:FindFirstChild("GameFolder")
        local timer = gameFolder and gameFolder:FindFirstChild("Timer")
        local intValue = timer and timer:IsA("IntValue") and timer.Value > 100
        local findFirstChild, skip

        if mainMenu then
          local mainScreen = mainMenu:FindFirstChild("MainScreen")
          local visible = mainScreen and mainScreen:IsA("GuiObject") and mainScreen.Visible

          if intValue then
            v30 = "INGAME"
          elseif visible then
            findFirstChild = mainScreen:FindFirstChild("Play", true)

            if findFirstChild and findFirstChild:IsA("GuiButton") and findFirstChild.Visible then
              if v30 == "INGAME" or v30 == "INIT" then
                local v31 = v30
                v30 = "LOBBY"

                if v31 ~= "INIT" then
                  f2("Game Ended. Sending Webhook...")
                  local value14 = currency.Value
                  local v32 = value14 - v4
                  v5 = v5 + 1
                  local v33 = os.time() - v3
                  local v34 = os.time()
                  text = f4(v33)
                  v7 = tostring(v32)
                  f1(v33, v32, v34 - v2, v5, value14 - value, value14)
                  v4 = value14
                end
              end

              f2("Firing Play Button...")

              if firesignal then
                pcall(function() firesignal(findFirstChild.MouseButton1Click) end)
                pcall(function() firesignal(findFirstChild.Activated) end)
              end
            end
          end

          local playMenu = mainMenu:FindFirstChild("PlayMenu")

          local votingMenu = playMenu
          votingMenu = playMenu and playMenu:FindFirstChild("VotingMenu")

          if votingMenu then
            local resultsMenu = votingMenu:FindFirstChild("ResultsMenu")

            if resultsMenu and resultsMenu:IsA("GuiObject") and resultsMenu.Visible then
              f2("Skipping Results Timer...")

              if remotes2 and remotes2:FindFirstChild("VIPCommandEvent") then
                remotes2.VIPCommandEvent:FireServer("SkipTimer", true)
              end
            end

            local mapVoting = votingMenu:FindFirstChild("MapVoting")

            if mapVoting and mapVoting:IsA("GuiObject") and mapVoting.Visible then
              f2("Choosing Map (Carnival)...")

              if remotes2 and remotes2:FindFirstChild("VIPCommandEvent") then
                remotes2.VIPCommandEvent:FireServer("SetMap", "Carnival")
                task.wait(0.1)
                remotes2.VIPCommandEvent:FireServer("SkipTimer", true)
              end
            end

            local modeVoting = votingMenu:FindFirstChild("ModeVoting")

            if modeVoting and modeVoting:IsA("GuiObject") and modeVoting.Visible then
              f2("Choosing Mode (Swarm)...")

              if remotes2 and remotes2:FindFirstChild("VIPCommandEvent") then
                remotes2.VIPCommandEvent:FireServer("SetMode", "Swarm")
                task.wait(0.1)
                remotes2.VIPCommandEvent:FireServer("SkipTimer", true)
              end
            end
          end

          skip = mainMenu:FindFirstChild("Skip")

          if skip and skip:IsA("GuiObject") and skip.Visible then
            if firesignal then
              pcall(function() firesignal(skip.MouseButton1Click) end)
              pcall(function() firesignal(skip.Activated) end)
            end
          end
        end

        if intValue and not _G.RunningSpeedrun then
          _G.RunningSpeedrun = true

          task.spawn(function()
            f2("Waiting Cutscene / Load...")
            v3 = os.time()
            task.wait(5)
            f2("Playing Game (Speedrun)...")
            f3()
            task.wait(3)
            _G.RunningSpeedrun = false
          end)
        end
      end)
    end
  end

  task.spawn(runMainLoop)
  return
end
