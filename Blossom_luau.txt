-- Animator: one render-step loop driving springs. Retargeting keeps velocity,
-- so interrupted motion stays continuous, and it costs far less than TweenService.
-------------------------------------------------------------------------------

local Presets = {
	Fast = { W = 42, Z = 1 },      -- hover and colour feedback
	Knob = { W = 44, Z = 0.86 },   -- toggles and slider knobs
	Follow = { W = 34, Z = 1 },    -- things attached to the pointer
	Smooth = { W = 26, Z = 1 },    -- layout and popups
	Soft = { W = 19, Z = 1 },      -- page transitions, theme morphs
	Settle = { W = 24, Z = 0.8 },  -- small overshoot for reflow
}

local Active, Tracks, Jobs = {}, {}, {}
local EPS = { number = 0.0015, Color3 = 0.003, UDim2 = 0.4, Vector2 = 0.4 }

local function encode(v)
	local kind = typeof(v)
	if kind == "number" then return { v } end
	if kind == "Color3" then return { v.R, v.G, v.B } end
	if kind == "UDim2" then return { v.X.Scale, v.X.Offset, v.Y.Scale, v.Y.Offset } end
	if kind == "Vector2" then return { v.X, v.Y } end
	error("cannot animate " .. kind)
end

local function decode(kind, x)
	if kind == "number" then return x[1] end
	if kind == "Color3" then return Color3.new(math.clamp(x[1], 0, 1), math.clamp(x[2], 0, 1), math.clamp(x[3], 0, 1)) end
	if kind == "UDim2" then return Dim2(x[1], round(x[2]), x[3], round(x[4])) end
	return Vec2(round(x[1]), round(x[2]))
end

local function untrack(entry)
	entry.Dead = true
	local map = Tracks[entry.Target]
	if map and map[entry.Prop] == entry then
		map[entry.Prop] = nil
		if next(map) == nil then Tracks[entry.Target] = nil end
	end
end

local function animate(target, prop, goal, preset, opts)
	local map = Tracks[target]
	if not map then map = {}; Tracks[target] = map end
	local entry = map[prop]
	if not entry then
		local from = encode(target[prop])
		local vel = {}
		for i = 1, #from do vel[i] = 0 end
		entry = { Target = target, Prop = prop, Kind = typeof(goal), X = from, V = vel }
		map[prop] = entry
		table.insert(Active, entry)
	end
	entry.G, entry.Goal = encode(goal), goal
	entry.P = Presets[preset or "Smooth"]
	entry.Done = opts and opts.Done or nil
	entry.Step = opts and opts.Step or nil
end

local function snap(target, prop, value)
	local map = Tracks[target]
	local entry = map and map[prop]
	if entry then untrack(entry) end
	target[prop] = value
end

local function place(target, prop, value, animated, preset)
	if animated then animate(target, prop, value, preset) else snap(target, prop, value) end
end

local function stepAnimator(dt)
	dt = math.min(dt, 1 / 20)
	local cache = {}
	local i = 1
	while i <= #Active do
		local e = Active[i]
		if e.Dead then
			Active[i] = Active[#Active]
			Active[#Active] = nil
		else
			local c = cache[e.P]
			if not c then
				local w, z = e.P.W, e.P.Z
				local t = dt
				if z >= 1 then
					local k = math.exp(-w * t)
					c = { k * (1 + w * t), k * t, -k * w * w * t, k * (1 - w * t) }
				else
					local wd = w * math.sqrt(1 - z * z)
					local k, co, si = math.exp(-z * w * t), math.cos(wd * t), math.sin(wd * t)
					c = { k * (co + z * w * si / wd), k * si / wd, -k * w * w * si / wd, k * (co - z * w * si / wd) }
				end
				cache[e.P] = c
			end
			local x, v, g = e.X, e.V, e.G
			local eps, settled = EPS[e.Kind], true
			local velEps = eps * e.P.W
			for n = 1, #x do
				local d = x[n] - g[n]
				local nv = c[3] * d + c[4] * v[n]
				local nd = c[1] * d + c[2] * v[n]
				x[n], v[n] = g[n] + nd, nv
				local channelEps = (e.Kind == "UDim2" and n % 2 == 1) and 0.0005 or eps
				if math.abs(nd) > channelEps or math.abs(nv) > velEps then settled = false end
			end
			if settled then
				e.Target[e.Prop] = e.Goal
				if e.Step then e.Step(e.Goal) end
				local done = e.Done
				untrack(e)
				Active[i] = Active[#Active]
				Active[#Active] = nil
				if done then task.spawn(done) end
			else
				local value = decode(e.Kind, x)
				if value ~= e.Last then
					e.Last = value
					e.Target[e.Prop] = value
					if e.Step then e.Step(value) end
				end
				i += 1
			end
		end
	end
end

local FrameConnection
local sweepHover
local function ensureLoop()
	if FrameConnection then return end
	FrameConnection = RunService.RenderStepped:Connect(function(dt)
		stepAnimator(dt)
		for job in Jobs do job(dt) end
		sweepHover()
	end)
end

local Bound, ByKey, Dynamic, Cased, NoFade = {}, {}, {}, {}, {}

local function resolve(source)
	if type(source) == "function" then return source() end
	return Current[source]
end

local function forget(inst)
	local bindings = Bound[inst]
	if bindings then
		for prop, source in bindings do
			if type(source) == "string" then
				local set = ByKey[source]
				if set then set[inst] = nil end
			end
		end
	end
	Bound[inst], Dynamic[inst], Cased[inst], Tracks[inst], NoFade[inst] = nil, nil, nil, nil, nil
end

local function bind(inst, prop, source)
	local bindings = Bound[inst]
	if not bindings then
		bindings = {}
		Bound[inst] = bindings
		inst.Destroying:Once(function() forget(inst) end)
	end
	bindings[prop] = source
	if type(source) == "string" then
		local set = ByKey[source]
		if not set then set = {}; ByKey[source] = set end
		set[inst] = prop
	else
		Dynamic[inst] = true
	end
	inst[prop] = resolve(source)
end

local function refresh(inst, preset)
	local bindings = Bound[inst]
	if not bindings then return end
	for prop, source in bindings do
		local value = resolve(source)
		if preset == false or typeof(value) == "ColorSequence" then
			snap(inst, prop, value)
		else
			animate(inst, prop, value, preset or "Fast")
		end
	end
end

local function applyCase(text)
	if Casing == "Lower Case" then return string.lower(text) end
	if Casing == "Upper Case" or Casing == "Bold Case" then return string.upper(text) end
	return text
end

local function fontOf(weight)
	if Casing == "Bold Case" then weight = Enum.FontWeight.Bold end
	return Font.new(FontFamily, weight or BodyWeight, Enum.FontStyle.Normal)
end

local function setText(inst, text, prop)
	prop = prop or "Text"
	local entry = Cased[inst]
	if not entry then entry = {}; Cased[inst] = entry end
	entry[prop] = text
	inst[prop] = applyCase(text)
end

local function applyTextSettings()
	for inst, entry in Cased do
		for prop, raw in entry do
			if prop == "Text" or prop == "PlaceholderText" then inst[prop] = applyCase(raw) end
		end
		inst.FontFace = fontOf(entry.Weight)
	end
end

local TEXT_CLASSES = { TextLabel = true, TextButton = true, TextBox = true }

local function new(class, props, parent)
	local inst = Instance.new(class)
	if inst:IsA("GuiObject") then
		inst.BorderSizePixel = 0
		inst.BackgroundTransparency = 1
	end
	if class == "TextButton" then
		inst.AutoButtonColor = false
		inst.Text = ""
	end
	if TEXT_CLASSES[class] then
		if class ~= "TextButton" then Cased[inst] = {} end
		inst.FontFace = fontOf(nil)
		inst.TextSize = 13
		inst.TextXAlignment = Enum.TextXAlignment.Left
		inst.TextYAlignment = Enum.TextYAlignment.Top
	end
	local bg, fg, radius, border
	if props then
		for key, value in props do
			if key == "Bg" then bg = value
			elseif key == "Fg" then fg = value
			elseif key == "Radius" then radius = value
			elseif key == "Border" then border = value
			elseif key == "Weight" then
				Cased[inst].Weight = value
				inst.FontFace = fontOf(value)
			elseif (key == "Text" and class ~= "TextBox" or key == "PlaceholderText") and TEXT_CLASSES[class] then
				setText(inst, value, key)
			else
				inst[key] = value
			end
		end
	end
	if bg then
		if not (props and props.BackgroundTransparency) then inst.BackgroundTransparency = 0 end
		bind(inst, "BackgroundColor3", bg)
	end
	if fg then bind(inst, "TextColor3", fg) end
	if radius then
		local corner = Instance.new("UICorner")
		corner.CornerRadius = radius >= FULL and UDim.new(0.5, 0) or UDim.new(0, radius)
		corner.Parent = inst
	end
	if border then
		local stroke = Instance.new("UIStroke")
		stroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
		stroke.Thickness = 1
		bind(stroke, "Color", border)
		stroke.Parent = inst
	end
	if parent then inst.Parent = parent end
	return inst
end

-- Top-aligned text at a computed whole-pixel offset. Roblox blurs glyphs that land on half pixels,
-- which is what centred alignment produces when the free space is odd.
local function capOffset(rowH, size)
	local capHeight = round(size * 0.7)
	return math.ceil((rowH - capHeight) / 2 - round(size * 0.16))
end

local function label(parent, o)
	local size, rowH = o.TextSize or 12, o.RowH or Metrics.Row
	local x = o.X or 0
	return new("TextLabel", {
		Fg = o.Fg or "Text", Text = o.Text or "", TextSize = size, Weight = o.Weight, TextTruncate = o.Truncate,
		TextTransparency = o.TextTransparency, ZIndex = o.ZIndex,
		Position = Offset(x, (o.Y or 0) + capOffset(rowH, size)),
		Size = o.W and Offset(o.W, rowH) or Dim2(1, -x - (o.Right or 0), 0, rowH),
	}, parent)
end

-- Centres text inside a fixed width using its measured bounds, floored to a whole pixel.
local function centeredLabel(parent, width, o)
	o.Truncate = Enum.TextTruncate.AtEnd
	local text = label(parent, o)
	local y, rowH = text.Position.Y.Offset, o.RowH or Metrics.Row
	local function fit()
		-- The box keeps the full available width. Sizing a truncating label from its own
		-- bounds shrinks it further each time, because truncated text measures narrower.
		text.Size = Offset(width - 12, rowH)
		text.Position = Offset(floor((width - math.min(math.ceil(text.TextBounds.X), width - 12)) / 2), y)
	end
	text:GetPropertyChangedSignal("TextBounds"):Connect(fit)
	fit()
	return text, function(newWidth)
		width = newWidth
		fit()
	end
end

local function clamp01(n) return math.clamp(n, 0, 1) end

local function isClick(input)
	return input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch
end

-- AbsolutePosition and GetMouseLocation use different origins whenever a top bar inset exists.
-- Offsetting by the ScreenGui's own AbsolutePosition puts the pointer in AbsolutePosition space.
local ScreenRef

local function mousePos()
	local location = UserInputService:GetMouseLocation()
	if ScreenRef then return location + ScreenRef.AbsolutePosition end
	return location
end

local function toLayer(point)
	return ScreenRef and point - ScreenRef.AbsolutePosition or point
end

local function inside(frame, point, margin)
	margin = margin or 0
	local p, s = frame.AbsolutePosition, frame.AbsoluteSize
	return point.X >= p.X - margin and point.X <= p.X + s.X + margin and point.Y >= p.Y - margin and point.Y <= p.Y + s.Y + margin
end

-- Polls once per rendered frame, so dragging costs one update per frame however fast the mouse reports.
local function beginDrag(onMove, onEnd, onFrame)
	local last
	local function job(dt)
		local point = mousePos()
		if not last or point ~= last then
			last = point
			if onMove then onMove(point) end
		end
		if onFrame then onFrame(point, dt) end
	end
	Jobs[job] = true
	local ended
	ended = UserInputService.InputEnded:Connect(function(input)
		if isClick(input) then
			ended:Disconnect()
			Jobs[job] = nil
			local point = mousePos()
			if onMove and point ~= last then onMove(point) end
			if onEnd then onEnd(point) end
		end
	end)
end

-- Roblox skips MouseLeave on fast movement and fires MouseEnter for covered objects, so hover
-- is confirmed against the pointer every frame instead of trusting those events alone.
local Focus
local HoverSet = {}

local function isShown(inst)
	local node = inst
	while node and node:IsA("GuiObject") do
		if not node.Visible then return false end
		node = node.Parent
	end
	return true
end

local function covered(inst, point)
	local win = Focus
	if not win then return false end
	local dropdown = win.OpenDropdown
	if dropdown then
		if inst:IsDescendantOf(dropdown.Panel) then return false end
		if inside(dropdown.Panel, point) then return true end
	end
	local popup = win.ActivePopup
	if popup and inside(popup.Frame, point) and not inst:IsDescendantOf(popup.Frame) then return true end
	if win.Search and win.Search.Open and not inst:IsDescendantOf(win.Search.Overlay) then return true end
	return false
end

local function hoverable(inst, onEnter, onLeave)
	local entered = false
	local function leave()
		if not entered then return end
		entered = false
		HoverSet[inst] = nil
		onLeave()
	end
	inst.MouseEnter:Connect(function()
		if entered or covered(inst, mousePos()) then return end
		entered = true
		HoverSet[inst] = leave
		onEnter()
	end)
	inst.MouseLeave:Connect(leave)
	inst.Destroying:Once(function() HoverSet[inst] = nil end)
end

function sweepHover()
	if next(HoverSet) == nil then return end
	local point = mousePos()
	for inst, leave in HoverSet do
		if not inside(inst, point) or not isShown(inst) or covered(inst, point) then leave() end
	end
end

-------------------------------------------------------------------------------
-- Fader: one animated alpha drives every transparency beneath a root
-------------------------------------------------------------------------------

local Fader = {}
Fader.__index = Fader

local function newFader(root)
	return setmetatable({ Root = root, Alpha = 1, Entries = nil }, Fader)
end

local function captureFade(root)
	local entries = {}
	local function add(inst, prop)
		-- A value that's still animating would be restored stale when the fade ends.
		if NoFade[inst] or (Tracks[inst] and Tracks[inst][prop]) then return end
		local base = inst[prop]
		if base < 1 then table.insert(entries, { inst, prop, base }) end
	end
	local function visit(inst)
		if inst:IsA("GuiObject") then
			if inst ~= root and not inst.Visible then return end
			add(inst, "BackgroundTransparency")
			if TEXT_CLASSES[inst.ClassName] then add(inst, "TextTransparency") end
			if inst:IsA("ImageLabel") then add(inst, "ImageTransparency") end
		elseif inst:IsA("UIStroke") or inst.ClassName == "UIShadow" then
			add(inst, "Transparency")
		end
		for _, child in inst:GetChildren() do visit(child) end
	end
	visit(root)
	return entries
end

function Fader:_apply(alpha)
	local entries = self.Entries
	if not entries then return end
	for _, e in entries do
		e[1][e[2]] = 1 - (1 - e[3]) * alpha
	end
end

function Fader:Show(onDone, preset)
	if not self.Entries then
		self.Entries = captureFade(self.Root)
		self.Alpha = 0
		self:_apply(0)
	end
	self.Root.Visible = true
	animate(self, "Alpha", 1, preset or "Smooth", {
		Step = function(value) self:_apply(value) end,
		Done = function()
			self.Entries = nil
			if onDone then onDone() end
		end,
	})
end

function Fader:Hide(onDone, preset)
	if not self.Entries then
		self.Entries = captureFade(self.Root)
		self.Alpha = 1
	end
	animate(self, "Alpha", 0, preset or "Fast", {
		Step = function(value) self:_apply(value) end,
		Done = function()
			self.Root.Visible = false
			self:_apply(1)
			self.Entries = nil
			if onDone then onDone() end
		end,
	})
end

local IconImages = {}
local function makeIcon(parent, name, size, color, hole, props)
	props = props or {}
	name = IconImages[name] or name
	local root = new("Frame", { Name = "Icon", Size = Offset(size, size), Position = props.Position, AnchorPoint = props.AnchorPoint, ZIndex = props.ZIndex }, parent)
	local icon = { Root = root, Parts = {} }
	if type(name) == "string" and string.find(name, "rbxasset", 1, true) then
		local image = new("ImageLabel", { Size = Dim2(1, 0, 1, 0), Image = name }, root)
		bind(image, "ImageColor3", color)
		table.insert(icon.Parts, image)
	else
		local scale = size / 16
		for _, part in Icons[name] or Icons.cube do
			local kind, x, y, w, h, radius, rotation, thickness = table.unpack(part, 1, 8)
			local frame = new("Frame", {
				Position = UDim2.fromScale(x / 16, y / 16), Size = UDim2.fromScale(w / 16, h / 16),
				Rotation = rotation or 0, Radius = radius and radius > 0 and (radius >= FULL and FULL or math.max(1, round(radius * scale))) or nil,
			}, root)
			if kind == "r" then
				local stroke = new("UIStroke", { Thickness = (thickness or 1.5) * scale }, frame)
				bind(stroke, "Color", color)
				table.insert(icon.Parts, stroke)
			else
				frame.BackgroundTransparency = 0
				bind(frame, "BackgroundColor3", kind == "h" and hole or color)
				table.insert(icon.Parts, frame)
			end
		end
	end
	function icon.Update(preset)
		for _, part in icon.Parts do refresh(part, preset) end
	end
	return icon
end

-- Hover and press colours resolved through the theme, so a theme change can't strand a stale colour.
local function interactive(target, base, hover, press, handle)
	local state = { Hover = false, Press = false }
	local source = handle or target
	bind(target, "BackgroundColor3", function()
		if state.Press and press then return Current[press] end
		if state.Hover then return Current[hover or "HoveredElement"] end
		return Current[base]
	end)
	hoverable(source, function() state.Hover = true; refresh(target) end, function() state.Hover, state.Press = false, false; refresh(target) end)
	if source:IsA("GuiButton") then
		source.MouseButton1Down:Connect(function() state.Press = true; refresh(target) end)
		source.MouseButton1Up:Connect(function() state.Press = false; refresh(target) end)
	end
	return state
end

local function formatNumber(value, decimals)
	if decimals and decimals > 0 then return string.format("%." .. decimals .. "f", value) end
	return tostring(floor(value + 0.5))
end

local function keyName(key)
	if key == nil then return "-" end
	local short = {
		LeftShift = "LS", RightShift = "RS", LeftControl = "LC", RightControl = "RC", LeftAlt = "LA", RightAlt = "RA",
		MouseButton1 = "M1", MouseButton2 = "M2", MouseButton3 = "M3", Return = "Ent", Backspace = "Bk",
		One = "1", Two = "2", Three = "3", Four = "4", Five = "5", Six = "6", Seven = "7", Eight = "8", Nine = "9", Zero = "0",
	}
	return short[key.Name] or key.Name
end

local function makeChip(parent, text, anchorRight, outerHeight)
	local height = (outerHeight or Metrics.ChipH) - 2
	local chip = new("Frame", {
		Bg = "Element", Radius = Metrics.Radius, Border = "Outline", Size = Offset(16, height),
		AnchorPoint = anchorRight and Vec2(1, 0) or nil,
	}, parent)
	addGradient(chip)
	local caption = label(chip, { Fg = "InactiveText", Text = text, TextSize = 11, X = 4, RowH = height })
	return chip, caption
end

-- Sizes a chip around its text. Widths given to onFit are outer widths, borders included.
local function fitChip(chip, text, minimum, onFit)
	local height = chip.Size.Y.Offset
	local function apply(bounds)
		local width = math.max((minimum or 2) - 2, bounds + 8)
		chip.Size = Offset(width, height)
		text.Position = Offset(floor((width - bounds) / 2), text.Position.Y.Offset)
		if onFit then onFit(width + 2) end
		return width
	end
	local function fit() apply(math.ceil(text.TextBounds.X)) end
	text:GetPropertyChangedSignal("TextBounds"):Connect(fit)
	fit()
	return apply, fit
end

-- Click a slider amount to type a value; Enter or clicking away commits, anything non-numeric reverts.
local function makeEditable(chip, text, read, write, apply, fit)
	local height = chip.Size.Y.Offset
	local hit = new("TextButton", { Size = Dim2(1, 0, 1, 0), ZIndex = 2 }, chip)
	interactive(chip, "Element", nil, nil, hit)
	local editing = false
	local stroke = chip:FindFirstChildOfClass("UIStroke")
	bind(stroke, "Color", function() return editing and Current.Accent or Current.Outline end)
	local box = new("TextBox", {
		Fg = "Text", TextSize = 11, Visible = false, ZIndex = 3, ClearTextOnFocus = false,
		Position = Offset(4, capOffset(height, 11)), Size = Dim2(1, -8, 0, height),
	}, chip)
	box:GetPropertyChangedSignal("Text"):Connect(function()
		if editing then apply(math.ceil(box.TextBounds.X)) end
	end)
	hit.MouseButton1Click:Connect(function()
		editing = true
		refresh(stroke)
		box.Text = read()
		text.Visible, box.Visible = false, true
		box:CaptureFocus()
		box.CursorPosition = #box.Text + 1
		box.SelectionStart = 1
	end)
	box.FocusLost:Connect(function()
		if not editing then return end
		editing = false
		refresh(stroke)
		local number = tonumber(string.match(box.Text, "%-?%d*%.?%d+") or "")
		box.Visible, text.Visible = false, true
		if number then write(number) end
		fit()
	end)
end

function Column:Layout(animated)
	local y = 0
	for _, section in self.Sections do
		if section.Hidden then continue end
		section.Y = y
		if not section.Dragging and not section.Dropping then
			place(section.Root, "Position", Offset(0, y), animated, "Settle")
		end
		y += section.TotalH + Metrics.SectionGap
	end
	self.ContentH = math.max(0, y - Metrics.SectionGap)
	self.Scroll.CanvasSize = Offset(0, self.ContentH)
	self:ClampScroll()
end

function Column:ClampScroll()
	-- Uses the window's own height so it's right on the same frame as a resize.
	self.Target = math.clamp(self.Target, 0, math.max(0, self.ContentH - (self.Tab.Window.Height - Metrics.Pad * 2)))
end

function Column:ScrollTo(y)
	self.Target = y
	self:ClampScroll()
	animate(self.Scroll, "CanvasPosition", Vec2(0, self.Target), "Soft")
end

function Section:Layout(animated)
	local y, any = Metrics.Pad, false
	for _, item in self.Items do
		if item.Visible then
			item.Y, any = y, true
			place(item.Frame, "Position", Offset(0, y), animated)
			y += item.Height + Metrics.Gap
		end
	end
	if any then y -= Metrics.Gap end
	self.CardH = y + Metrics.Pad
	self.TotalH = self.HeaderH + self.CardH
	place(self.Card, "Size", Dim2(1, -2, 0, self.CardH - 2), animated)
	if not self.Dragging then place(self.Root, "Size", Dim2(1, 0, 0, self.TotalH), animated) end
	if self.Column then self.Column:Layout(animated) end
end

function Section:_item(height, kind, name, indexed)
	local frame = new("Frame", { Size = Dim2(1, 0, 0, height) }, self.Body)
	local item = { Frame = frame, Height = height, Kind = kind, Name = name, Section = self, Slots = {}, Visible = true, Y = 0 }
	table.insert(self.Items, item)
	if name and indexed ~= false then table.insert(self.Window.Index, item) end
	self:Layout(false)
	return item
end

local function layoutSlots(item)
	local x = 0
	for _, slot in item.Slots do
		x += slot.Gap
		slot.Frame.Position = Dim2(1, -x - 1, 0, floor((Metrics.Row - slot.Height) / 2) + 1)
		x += slot.Width
	end
	if item.Label then item.Label.Size = Dim2(1, -(x + Metrics.Gap), 0, Metrics.Row) end
end

local function addSlot(item, frame, width, height, gap)
	local slot = { Frame = frame, Width = width, Height = height, Gap = #item.Slots == 0 and 0 or gap }
	frame.AnchorPoint = Vec2(1, 0)
	table.insert(item.Slots, slot)
	layoutSlots(item)
	return slot
end

local function attachKeybind(section, item, options)
	local win = section.Window
	local object = {
		Key = options.Key, Mode = options.Mode or "Toggle", Listening = false,
		OnPress = options.OnPress, OnRelease = options.OnRelease, Name = options.Name,
	}
	local chip, text = makeChip(item.Frame, keyName(object.Key))
	local button = new("TextButton", { Size = Dim2(1, 0, 1, 0), ZIndex = 2 }, chip)
	local slot = addSlot(item, chip, 18, Metrics.ChipH, Metrics.Gap + 2)
	fitChip(chip, text, 18, function(width)
		slot.Width = width
		layoutSlots(item)
	end)
	interactive(chip, "Element", nil, nil, button)
	bind(text, "TextColor3", function()
		if object.Listening then return Current.Accent end
		return Current.InactiveText
	end)
	function object.Refresh()
		setText(text, object.Listening and "..." or keyName(object.Key))
		refresh(text)
		if object.OnKeyChanged then object.OnKeyChanged() end
	end
	function object:Set(key)
		object.Key = key
		object.Refresh()
	end
	button.MouseButton1Click:Connect(function()
		object.Listening = true
		win.Listening = object
		object.Refresh()
	end)
	button.MouseButton2Click:Connect(function() win:OpenKeybindMenu(chip, object) end)
	object.Chip, object.Label = chip, text
	table.insert(win.Binds, object)
	return object
end

local function attachColor(section, item, options)
	local win = section.Window
	local color = {
		Value = options.Default or Color3.new(1, 1, 1), Alpha = options.Alpha or 0, Animation = "None",
		Callback = options.Callback, Flag = options.Flag,
	}
	local swatch = new("TextButton", {
		Size = Offset(Metrics.Swatch - 2, Metrics.Swatch - 2), Radius = Metrics.Radius, BackgroundTransparency = 0,
		BackgroundColor3 = color.Value, Border = "Outline",
	}, item.Frame)
	addSlot(item, swatch, Metrics.Swatch, Metrics.Swatch, Metrics.Gap + 2)
	color.Swatch = swatch
	function color:Set(value, alpha, silent)
		color.Value = value
		if alpha ~= nil then color.Alpha = alpha end
		swatch.BackgroundColor3 = value
		if win.Picker.Target == color and not win.Picker.Dragging then win.Picker.Sync() end
		if not silent then fire(color.Callback, value, color.Alpha) end
	end
	swatch.MouseButton1Click:Connect(function() win:OpenPicker(swatch, color) end)
	table.insert(win.Colors, color)
	return color
end

local function addInlineControls(section, item, options, name)
	local keybind, color
	if options.Color then
		color = attachColor(section, item, { Default = options.Color, Alpha = options.ColorAlpha, Callback = options.ColorCallback })
	end
	if options.Keybind ~= nil then
		keybind = attachKeybind(section, item, { Key = options.Keybind, Mode = options.KeybindMode, Name = options.KeybindName or name })
	end
	return keybind, color
end

-- Every element joins configs automatically under Tab/Section/Name unless NoSave is set.
local Hooks = {}
local function register(section, object, options, item)
	local win = section.Window
	win.FlagOwners[item] = object
	if options.NoSave then
		if Hooks.Register then Hooks.Register(section, object, options, item) end
		return nil
	end
	local base = options.Flag or (section.Tab.Name .. "/" .. section.Name .. "/" .. tostring(options.Name or item.Kind))
	local flag, count = base, 1
	while win.Flags[flag] do
		count += 1
		flag = base .. "#" .. count
	end
	win.Flags[flag] = object
	object.Flag = flag
	if Hooks.Register then Hooks.Register(section, object, options, item) end
	return flag
end

function Section:Toggle(options)
	local win, name = self.Window, options.Name
	local item = self:_item(Metrics.Row, "Toggle", name)
	local toggle = { Value = options.Default == true, Item = item, Callback = options.Callback }
	local hit = new("TextButton", { Position = Offset(0, -3), Size = Dim2(1, 0, 1, 6) }, item.Frame)
	local hovering = false
	local text = label(item.Frame, {
		Text = name, Truncate = Enum.TextTruncate.AtEnd,
		Fg = function()
			if toggle.Value then return Current.Text end
			return hovering and Current.InactiveText:Lerp(Current.Text, 0.45) or Current.InactiveText
		end,
	})
	item.Label = text
	local boxStyle = options.Style == "Checkbox"
	local track = new("Frame", {
		Bg = function() return toggle.Value and Current.Accent or Current.Element end,
		Size = boxStyle and Offset(Metrics.ToggleH - 2, Metrics.ToggleH - 2) or Offset(Metrics.ToggleW - 2, Metrics.ToggleH - 2),
		Radius = boxStyle and Metrics.Radius or FULL, Border = "Outline",
	}, item.Frame)
	addGradient(track)
	local knob = new("Frame", {
		Bg = function() return toggle.Value and Current.Background or Current.InactiveText end,
		Size = Offset(Metrics.ToggleKnob, Metrics.ToggleKnob), Radius = FULL,
	}, track)
	addSlot(item, track, boxStyle and Metrics.ToggleH or Metrics.ToggleW, Metrics.ToggleH, 0)

	local function render(animated)
		if boxStyle then
			knob.Size, knob.Position, knob.Visible = Offset(Metrics.ToggleH - 10, Metrics.ToggleH - 10), Offset(4, 4), toggle.Value
		else
			local x = toggle.Value and Metrics.ToggleW - 2 - Metrics.ToggleKnob - 2 or 2
			place(knob, "Position", Offset(x, 2), animated, "Knob")
		end
		local preset = animated and "Fast" or false
		refresh(track, preset); refresh(knob, preset); refresh(text, preset)
	end
	local function syncHotkey()
		if toggle.Keybind then win:SetHotkeyActive(toggle.Keybind, toggle.Value) end
	end

	function toggle:Set(value, silent)
		value = value == true
		if value == toggle.Value then return end
		toggle.Value = value
		render(true)
		syncHotkey()
		if not silent then fire(toggle.Callback, value) end
	end

	hoverable(hit, function()
		hovering = true
		refresh(text)
	end, function()
		hovering = false
		refresh(text)
	end)
	hit.MouseButton1Click:Connect(function() toggle:Set(not toggle.Value) end)

	local hotkeyName = options.KeybindName or (name == "Enabled" and self.Name) or name
	options.KeybindName = hotkeyName
	toggle.Keybind, toggle.Color = addInlineControls(self, item, options, hotkeyName)
	if toggle.Keybind then
		toggle.Keybind.OnPress = function()
			if toggle.Keybind.Mode == "Hold" then toggle:Set(true) else toggle:Set(not toggle.Value) end
		end
		toggle.Keybind.OnRelease = function()
			if toggle.Keybind.Mode == "Hold" then toggle:Set(false) end
		end
		toggle.Keybind.OnKeyChanged = syncHotkey
	end
	render(false)
	syncHotkey()
	local flag = register(self, toggle, options, item)
	if flag and toggle.Keybind then win.Flags[flag .. "/Key"] = toggle.Keybind end
	if flag and toggle.Color then win.Flags[flag .. "/Color"] = toggle.Color end
	return toggle
end

local function trackWidth(win, section)
	if section and section.InnerW then return section.InnerW() end
	return win.ColumnW - Metrics.Pad * 2 - 2
end

local function makeTrack(parent)
	local track = new("Frame", {
		Bg = "Element", Position = Offset(1, Metrics.TrackY + 1), Size = Dim2(1, -2, 0, Metrics.TrackH - 2),
		Radius = FULL, Border = "Outline",
	}, parent)
	addGradient(track)
	local fill = new("Frame", { Name = "Fill", Bg = "Accent", Radius = FULL, Size = Dim2(0, 0, 1, 0) }, track)
	addGradient(fill)
	local hit = new("TextButton", { Position = Offset(-4, Metrics.Row), Size = Dim2(1, 8, 0, Metrics.SliderH - Metrics.Row), ZIndex = 4 }, parent)
	return track, fill, hit
end

local function makeKnob(track)
	return new("Frame", { Name = "Knob", Bg = "Text", Size = Offset(Metrics.Knob, Metrics.Knob), Radius = FULL, ZIndex = 3 }, track)
end

local function knobPixels(alpha, width)
	return round(Metrics.TrackInset + alpha * (width - Metrics.TrackInset * 2))
end

-- Pixel positions keep the knob centred exactly: the same overhang above and below the track.
local function placeKnob(knob, pixels, animated)
	place(knob, "Position", Offset(pixels - floor(Metrics.Knob / 2), floor((Metrics.TrackH - 2 - Metrics.Knob) / 2)), animated, "Follow")
end

local function alphaAt(track, point)
	local inset = Metrics.TrackInset
	return clamp01((point.X - track.AbsolutePosition.X - inset) / (track.AbsoluteSize.X - inset * 2))
end

function Section:Slider(options)
	local item = self:_item(Metrics.SliderH, "Slider", options.Name)
	local caption = label(item.Frame, { Text = options.Name, Truncate = Enum.TextTruncate.AtEnd })
	local minimum, maximum = options.Min or 0, options.Max or 100
	local decimals, suffix = options.Decimals or 0, options.Suffix or ""
	local slider = { Value = minimum, Item = item, Callback = options.Callback }
	local chip, chipText = makeChip(item.Frame, "", true, Metrics.ValueChipH)
	chip.Position = Dim2(1, -1, 0, 0)
	local apply, fit = fitChip(chip, chipText, 24, function(width) caption.Size = Dim2(1, -(width + Metrics.Gap), 0, Metrics.Row) end)
	local track, fill, hit = makeTrack(item.Frame)
	local knob = makeKnob(track)

	local function render(animated)
		local alpha = maximum == minimum and 0 or (slider.Value - minimum) / (maximum - minimum)
		local pixels = knobPixels(alpha, trackWidth(self.Window, self))
		placeKnob(knob, pixels, animated)
		place(fill, "Size", Dim2(0, pixels, 1, 0), animated, "Follow")
	end

	function slider:Set(value, silent, animated)
		local step = 10 ^ -decimals
		value = math.clamp(floor(value / step + 0.5) * step, minimum, maximum)
		local changed = value ~= slider.Value
		slider.Value = value
		setText(chipText, formatNumber(value, decimals) .. suffix)
		render(animated ~= false)
		if changed and not silent then fire(slider.Callback, value) end
	end

	makeEditable(chip, chipText, function() return formatNumber(slider.Value, decimals) end, function(number) slider:Set(number) end, apply, fit)

	local function fromMouse(point)
		slider:Set(minimum + (maximum - minimum) * alphaAt(track, point))
	end
	hit.InputBegan:Connect(function(input)
		if not isClick(input) then return end
		fromMouse(mousePos())
		beginDrag(fromMouse)
	end)

	table.insert(self.Window.ResizeJobs, function() render(false) end)
	slider:Set(options.Default or minimum, true, false)
	register(self, slider, options, item)
	return slider
end

function Section:RangeSlider(options)
	local item = self:_item(Metrics.SliderH, "Range Slider", options.Name)
	local caption = label(item.Frame, { Text = options.Name, Truncate = Enum.TextTruncate.AtEnd })
	local minimum, maximum = options.Min or 0, options.Max or 100
	local decimals, suffix = options.Decimals or 0, options.Suffix or ""
	local default = options.Default or { minimum, maximum }
	local range = { Value = { default[1], default[2] }, Item = item, Callback = options.Callback }
	local highChip, highText = makeChip(item.Frame, "", true, Metrics.ValueChipH)
	local lowChip, lowText = makeChip(item.Frame, "", true, Metrics.ValueChipH)
	highChip.Position = Dim2(1, -1, 0, 0)
	local highWidth, lowWidth = 24, 24
	local function layoutChips()
		lowChip.Position = Dim2(1, -(highWidth + Metrics.ChipGap + 1), 0, 0)
		caption.Size = Dim2(1, -(highWidth + lowWidth + Metrics.ChipGap + Metrics.Gap), 0, Metrics.Row)
	end
	local applyHigh, fitHigh = fitChip(highChip, highText, 24, function(width) highWidth = width; layoutChips() end)
	local applyLow, fitLow = fitChip(lowChip, lowText, 24, function(width) lowWidth = width; layoutChips() end)
	local track, fill, hit = makeTrack(item.Frame)
	local lowKnob, highKnob = makeKnob(track), makeKnob(track)

	local function alphaOf(value) return (value - minimum) / (maximum - minimum) end

	local function render(animated)
		local width = trackWidth(self.Window, self)
		local low, high = knobPixels(alphaOf(range.Value[1]), width), knobPixels(alphaOf(range.Value[2]), width)
		placeKnob(lowKnob, low, animated)
		placeKnob(highKnob, high, animated)
		place(fill, "Position", Dim2(0, low, 0, 0), animated, "Follow")
		place(fill, "Size", Dim2(0, high - low, 1, 0), animated, "Follow")
	end

	function range:Set(value, silent, animated)
		range.Value = { math.min(value[1], value[2]), math.max(value[1], value[2]) }
		setText(lowText, formatNumber(range.Value[1], decimals) .. suffix)
		setText(highText, formatNumber(range.Value[2], decimals) .. suffix)
		render(animated ~= false)
		if not silent then fire(range.Callback, range.Value) end
	end

	local function snapped(value)
		local step = 10 ^ -decimals
		return math.clamp(floor(value / step + 0.5) * step, minimum, maximum)
	end

	makeEditable(lowChip, lowText, function() return formatNumber(range.Value[1], decimals) end, function(number)
		range:Set({ math.min(snapped(number), range.Value[2]), range.Value[2] })
	end, applyLow, fitLow)
	makeEditable(highChip, highText, function() return formatNumber(range.Value[2], decimals) end, function(number)
		range:Set({ range.Value[1], math.max(snapped(number), range.Value[1]) })
	end, applyHigh, fitHigh)

	local active = 1
	local function valueAt(point)
		return snapped(minimum + (maximum - minimum) * alphaAt(track, point))
	end
	local function fromMouse(point)
		local value, nextValue = valueAt(point), { range.Value[1], range.Value[2] }
		if active == 1 then
			nextValue[1] = math.min(value, nextValue[2])
		else
			nextValue[2] = math.max(value, nextValue[1])
		end
		range:Set(nextValue)
	end
	hit.InputBegan:Connect(function(input)
		if not isClick(input) then return end
		local point = mousePos()
		local value = valueAt(point)
		active = math.abs(value - range.Value[1]) <= math.abs(value - range.Value[2]) and 1 or 2
		fromMouse(point)
		beginDrag(fromMouse)
	end)

	table.insert(self.Window.ResizeJobs, function() render(false) end)
	range:Set(range.Value, true, false)
	register(self, range, options, item)
	return range
end

-- The option list is its own floating panel above the UI, so opening it never moves the section.
local function buildDropdown(win, parent, options)
	local dropdown = {
		Options = options.Options or {}, Multi = options.Multi == true, Open = false,
		Value = options.Multi and {} or options.Default, Callback = options.Callback,
	}
	if options.Multi and options.Default then
		for _, value in options.Default do dropdown.Value[value] = true end
	end

	local input = new("TextButton", {
		Position = Offset(1, (options.Y or 0) + 1), Size = Dim2(1, -2, 0, Metrics.InputH - 2), Radius = Metrics.Radius, BackgroundTransparency = 0,
	}, parent)
	interactive(input, "Element")
	addGradient(input)
	local stroke = new("UIStroke", { ApplyStrokeMode = Enum.ApplyStrokeMode.Border, Thickness = 1 }, input)
	bind(stroke, "Color", function() return dropdown.Open and Current.Accent or Current.Outline end)
	local value = label(input, { Fg = "InactiveText", TextSize = 13, X = 7, RowH = Metrics.InputH - 2, Right = 28, Truncate = Enum.TextTruncate.AtEnd })
	makeIcon(input, "layout", 14, "InactiveText", "Element", { AnchorPoint = Vec2(1, 0), Position = Dim2(1, -5, 0, floor((Metrics.InputH - 2 - 14) / 2)) })

	local panel = new("TextButton", { Bg = "Background", BackgroundTransparency = 1, Radius = Metrics.Radius, Visible = false, Size = Offset(10, 0) }, win.Dropdowns)
	local panelStroke = new("UIStroke", { ApplyStrokeMode = Enum.ApplyStrokeMode.Border, Thickness = 1, Transparency = 1 }, panel)
	bind(panelStroke, "Color", "Outline")
	local list = new("ScrollingFrame", { Size = Dim2(1, 0, 1, 0), ScrollBarThickness = 0, CanvasSize = Offset(0, 0) }, panel)
	local rows, panelH, flipped = {}, 0, false
	local emptyText = label(panel, { Fg = "InactiveText", Text = options.Empty or "No options", TextSize = 13, X = 9, RowH = Metrics.ListRowH, Right = 8, TextTransparency = 1 })
	dropdown.Reveal = 0

	local function isSelected(option)
		if dropdown.Multi then return dropdown.Value[option] == true end
		return dropdown.Value == option
	end

	local function caption()
		if not dropdown.Multi then return dropdown.Value and tostring(dropdown.Value) or "None" end
		local names = {}
		for _, option in dropdown.Options do
			if dropdown.Value[option] then table.insert(names, tostring(option)) end
		end
		return #names > 0 and table.concat(names, ", ") or "None"
	end

	-- Selection, hover and the panel's reveal are separate numbers combined on every write.
	-- Nothing stores a transparency to restore later, so interrupted animations can't leave stale rows.
	local function applyRow(row)
		local reveal = dropdown.Reveal
		row.Button.BackgroundTransparency = 1 - math.max(row.Sel, row.Hov) * reveal
		row.Bar.BackgroundTransparency = 1 - row.Sel * reveal
		row.Text.TextTransparency = 1 - reveal
	end

	local function applyReveal()
		local reveal = dropdown.Reveal
		panel.BackgroundTransparency = 1 - reveal
		panelStroke.Transparency = 1 - reveal
		emptyText.TextTransparency = 1 - reveal
		for _, row in rows do applyRow(row) end
	end

	local function paint(animated)
		local preset = animated and "Fast" or false
		for _, row in rows do
			local selected = isSelected(row.Option)
			if animated then
				animate(row, "Sel", selected and 1 or 0, "Fast", { Step = function() applyRow(row) end })
			else
				snap(row, "Sel", selected and 1 or 0)
				applyRow(row)
			end
			place(row.Text, "Position", Offset(selected and 17 or 8, row.TextY), animated, "Smooth")
			refresh(row.Text, preset)
			refresh(row.Button, preset)
		end
		setText(value, caption())
	end

	local function rebuild()
		for _, row in rows do row.Button:Destroy() end
		rows = {}
		local count = #dropdown.Options
		for index, option in dropdown.Options do
			local row = { Option = option, Sel = 0, Hov = 0, Hovering = false }
			row.Button = new("TextButton", {
				Bg = function() return row.Hovering and Current.Inline or Current.Tab end,
				BackgroundTransparency = 1, Position = Offset(0, (index - 1) * Metrics.ListRowH), Size = Dim2(1, 0, 0, Metrics.ListRowH),
				Radius = (index == 1 or index == count) and Metrics.Radius or nil,
			}, list)
			row.Bar = new("Frame", { Bg = "Accent", BackgroundTransparency = 1, Position = Offset(7, 6), Size = Offset(2, 12) }, row.Button)
			row.Text = label(row.Button, {
				Text = tostring(option), X = 8, TextSize = 13, RowH = Metrics.ListRowH, Right = 18, Truncate = Enum.TextTruncate.AtEnd, TextTransparency = 1,
				Fg = function() return isSelected(option) and Current.Text or Current.InactiveText end,
			})
			row.TextY = row.Text.Position.Y.Offset
			local function step() applyRow(row) end
			hoverable(row.Button, function()
				row.Hovering = true
				refresh(row.Button)
				animate(row, "Hov", 1, "Fast", { Step = step })
			end, function()
				row.Hovering = false
				refresh(row.Button)
				animate(row, "Hov", 0, "Fast", { Step = step })
			end)
			row.Button.MouseButton1Click:Connect(function()
				if dropdown.Multi then
					dropdown.Value[option] = not dropdown.Value[option] or nil
					paint(true)
					fire(dropdown.Callback, dropdown.Value)
				else
					dropdown:Set(option)
					dropdown:Toggle(false)
				end
			end)
			table.insert(rows, row)
		end
		list.CanvasSize = Offset(0, count * Metrics.ListRowH)
		emptyText.Visible = count == 0
		paint(false)
	end

	local function measure()
		panelH = math.clamp(#dropdown.Options, 1, options.MaxRows or 10) * Metrics.ListRowH
	end

	-- Called each frame while open, so the panel stays attached while the page scrolls or the window moves.
	local function follow()
		local p, s = input.AbsolutePosition, input.AbsoluteSize
		local screen = win.Screen.AbsolutePosition.Y + win.Screen.AbsoluteSize.Y
		flipped = p.Y + s.Y + 6 + panelH > screen - 8 and p.Y - 6 - panelH > win.Screen.AbsolutePosition.Y + 8
		local goal = toLayer(Vec2(p.X, flipped and p.Y - 6 or p.Y + s.Y + 6))
		panel.AnchorPoint = flipped and Vec2(0, 1) or Vec2(0, 0)
		panel.Position = Offset(goal.X, goal.Y)
	end
	dropdown.Follow = follow

	function dropdown:Toggle(open)
		if open == nil then open = not dropdown.Open end
		if open == dropdown.Open then return end
		dropdown.Open = open
		refresh(stroke)
		local width = input.AbsoluteSize.X
		if open then
			if win.OpenDropdown and win.OpenDropdown ~= dropdown then win.OpenDropdown:Toggle(false) end
			win.OpenDropdown = dropdown
			measure()
			follow()
			panel.Visible = true
			panel.Size = Offset(width, panelH)
			animate(dropdown, "Reveal", 1, "Fast", { Step = applyReveal })
			local selectedIndex = not dropdown.Multi and table.find(dropdown.Options, dropdown.Value)
			local furthest = math.max(0, #dropdown.Options * Metrics.ListRowH - panelH)
			list.CanvasPosition = Vec2(0, selectedIndex and math.clamp(floor((selectedIndex - 1) * Metrics.ListRowH - (panelH - Metrics.ListRowH) / 2), 0, furthest) or 0)
			for index, row in rows do
				local y = (index - 1) * Metrics.ListRowH
				row.Button.Position = Offset(0, y + 6)
				task.delay(math.min(index, 10) * 0.012, function()
					if row.Button.Parent then place(row.Button, "Position", Offset(0, y), dropdown.Open, "Smooth") end
				end)
			end
		else
			if win.OpenDropdown == dropdown then win.OpenDropdown = nil end
			animate(dropdown, "Reveal", 0, "Fast", {
				Step = applyReveal,
				Done = function() panel.Visible = false end,
			})
		end
	end

	function dropdown:Set(newValue, silent)
		if dropdown.Multi then
			dropdown.Value = {}
			for _, option in newValue do dropdown.Value[option] = true end
		else
			dropdown.Value = newValue
		end
		paint(true)
		if not silent then fire(dropdown.Callback, dropdown.Multi and dropdown.Value or newValue) end
	end

	function dropdown:SetOptions(newOptions)
		dropdown.Options = newOptions
		rebuild()
		if dropdown.Open then
			measure()
			panel.Size = Offset(input.AbsoluteSize.X, panelH)
		end
	end

	dropdown.Input, dropdown.Panel = input, panel
	input.MouseButton1Click:Connect(function()
		-- The press that closes an open list must not reopen it on release.
		if win.DropdownClosedBy == input and os.clock() - win.DropdownClosedAt < 0.25 then return end
		dropdown:Toggle()
	end)
	rebuild()
	return dropdown
end

function Section:Dropdown(options)
	local item = self:_item(Metrics.DropdownH, "Dropdown", options.Name)
	label(item.Frame, { Text = options.Name, Truncate = Enum.TextTruncate.AtEnd })
	options.Y = Metrics.DropdownH - Metrics.InputH
	local dropdown = buildDropdown(self.Window, item.Frame, options)
	dropdown.Item = item
	register(self, dropdown, options, item)
	return dropdown
end

function Section:Listbox(options)
	local height = options.Height or 177
	local item = self:_item(height, "Listbox", options.Name, false)
	local frame = new("ScrollingFrame", {
		Bg = "Inline", Position = Offset(1, 1), Size = Dim2(1, -2, 0, height - 2), Radius = Metrics.Radius, Border = "Outline",
		ScrollBarThickness = 0, CanvasSize = Offset(0, 0),
	}, item.Frame)
	local listbox = { Options = {}, Value = nil, Item = item, Callback = options.Callback, Multi = options.Multi == true }
	if listbox.Multi then listbox.Value = {} end
	local rows = {}
	local emptyText = label(frame, { Fg = "InactiveText", Text = options.Empty or "Nothing here yet", TextSize = 13, X = 10, Y = 2, RowH = Metrics.ListRowH, Right = 8 })

	local function paint(animated)
		local preset = animated and "Fast" or false
		for _, row in rows do
			local selected = listbox.Multi and listbox.Value[row.Option] == true or listbox.Value == row.Option
			refresh(row.Text, preset)
			place(row.Text, "Position", Offset(selected and 17 or 8, row.TextY), animated, "Smooth")
			place(row.Bar, "BackgroundTransparency", selected and 0 or 1, animated, "Fast")
		end
	end

	function listbox:Set(value, silent)
		if listbox.Multi then
			local map = {}
			if type(value) == "table" then
				if #value > 0 then for _, entry in value do map[entry] = true end
				else for key, on in value do if on then map[key] = true end end end
			end
			value = map
		end
		listbox.Value = value
		paint(true)
		if not silent then fire(listbox.Callback, value) end
	end

	function listbox:SetOptions(optionsList)
		for _, row in rows do row.Button:Destroy() end
		rows, listbox.Options = {}, optionsList
		for index, option in optionsList do
			local row = { Option = option }
			row.Button = new("TextButton", {
				Bg = "HoveredElement", BackgroundTransparency = 1, Position = Offset(2, 2 + (index - 1) * Metrics.ListRowH),
				Size = Dim2(1, -4, 0, Metrics.ListRowH), Radius = Metrics.SmallRadius,
			}, frame)
			NoFade[row.Button] = true
			row.Bar = new("Frame", { Bg = "Accent", BackgroundTransparency = 1, Position = Offset(6, 6), Size = Offset(2, 12) }, row.Button)
			row.Text = label(row.Button, {
				Text = option, X = 8, TextSize = 13, RowH = Metrics.ListRowH, Right = 18, Truncate = Enum.TextTruncate.AtEnd,
				Fg = function() return (listbox.Multi and listbox.Value[option] or listbox.Value == option) and Current.Text or Current.InactiveText end,
			})
			row.TextY = row.Text.Position.Y.Offset
			hoverable(row.Button, function() animate(row.Button, "BackgroundTransparency", 0, "Fast") end, function() animate(row.Button, "BackgroundTransparency", 1, "Fast") end)
			row.Button.MouseButton1Click:Connect(function()
				if listbox.Multi then
					local nextValue = table.clone(listbox.Value)
					nextValue[option] = not nextValue[option] or nil
					listbox:Set(nextValue)
				else
					listbox:Set(option)
				end
			end)
			table.insert(rows, row)
		end
		frame.CanvasSize = Offset(0, 4 + #optionsList * Metrics.ListRowH)
		emptyText.Visible = #optionsList == 0
		paint(false)
	end

	listbox:SetOptions(options.Options or options.Values or {})
	local start = options.Default
	if start == nil then start = options.Value end
	if start ~= nil then listbox:Set(start, true) end
	options.NoSave = options.NoSave ~= false and options.Flag == nil
	register(self, listbox, options, item)
	return listbox
end

function Section:_beginDrag()
	if self.Dropping then return end
	local win, start = self.Window, mousePos()
	local grab = start - self.Root.AbsolutePosition
	local stroke = self.Card:FindFirstChildOfClass("UIStroke")
	local lifted, ghost, ghostFader = false, nil, nil

	local function slotPosition(column)
		return column.Scroll.AbsolutePosition + Vec2(0, self.Y - column.Scroll.CanvasPosition.Y)
	end

	local function targetFor(point)
		local page = self.Tab
		local left, right = page.Columns[1], page.Columns[2]
		local boundary = (left.Scroll.AbsolutePosition.X + left.Scroll.AbsoluteSize.X + right.Scroll.AbsolutePosition.X) / 2
		local column = point.X < boundary and left or right
		local current = table.find(column.Sections, self)
		local index = 1
		for position, other in column.Sections do
			if other ~= self then
				local middle = column.Scroll.AbsolutePosition.Y - column.Scroll.CanvasPosition.Y + other.Y + other.TotalH / 2
				-- A dead zone either side of each midpoint stops sections trading places repeatedly.
				local bias = (current and position < current) and -14 or 14
				if point.Y > middle + bias then index += 1 end
			end
		end
		return column, index
	end

	local function lift()
		lifted = true
		self.Dragging = true
		win:CloseFloating()
		local absolute, size = self.Root.AbsolutePosition, self.Root.AbsoluteSize
		local layer = toLayer(absolute)
		self.Root.Parent = win.DragLayer
		self.Root.Size = Offset(size.X, self.TotalH)
		self.Root.Position = Offset(layer.X, layer.Y)
		refresh(stroke)
		ghost = new("Frame", {
			Bg = "Accent", BackgroundTransparency = 0.92, Radius = Metrics.CardRadius, Border = "Accent",
			Position = Offset(layer.X, layer.Y), Size = Offset(size.X, self.TotalH), ZIndex = -1,
		}, win.DragLayer)
		ghostFader = newFader(ghost)
		ghost.Visible = false
		ghostFader:Show(nil, "Fast")
	end

	local function move(point)
		if not lifted then
			if (point - start).Magnitude < 5 then return end
			lift()
		end
		local goal = toLayer(point - grab)
		animate(self.Root, "Position", Offset(goal.X, goal.Y), "Follow")
		local column, index = targetFor(point)
		local current = table.find(column.Sections, self)
		if column ~= self.Column or current ~= index then
			local previous = self.Column
			table.remove(previous.Sections, table.find(previous.Sections, self))
			table.insert(column.Sections, math.min(index, #column.Sections + 1), self)
			self.Column = column
			if previous ~= column then previous:Layout(true) end
			column:Layout(true)
		end
	end

	local function frame(point)
		if not lifted then return end
		-- Scroll the column while the pointer rests near its top or bottom edge.
		local column = self.Column
		local scroll = column.Scroll
		local top, bottom = scroll.AbsolutePosition.Y, scroll.AbsolutePosition.Y + scroll.AbsoluteSize.Y
		if point.X >= scroll.AbsolutePosition.X and point.X <= scroll.AbsolutePosition.X + scroll.AbsoluteSize.X then
			if point.Y < top + 36 then column:ScrollTo(column.Target - 8)
			elseif point.Y > bottom - 36 then column:ScrollTo(column.Target + 8) end
		end
		local slot = toLayer(slotPosition(column))
		animate(ghost, "Position", Offset(slot.X, slot.Y), "Smooth")
	end

	beginDrag(move, function()
		if not lifted then return end
		local column = self.Column
		column:Layout(true)
		self.Dragging = false
		self.Dropping = true
		refresh(stroke)
		local goal = toLayer(slotPosition(column))
		local doomed = ghost
		ghostFader:Hide(function() doomed:Destroy() end)
		animate(self.Root, "Position", Offset(goal.X, goal.Y), "Smooth", {
			Done = function()
				self.Dropping = false
				self.Root.Parent = column.Scroll
				self.Root.Size = Dim2(1, 0, 0, self.TotalH)
				self.Root.Position = Offset(0, self.Y)
			end,
		})
	end, frame)
end

function Tab:Section(options)
	local win = self.Window
	local column = self.Columns[math.min(options.Side or 1, #self.Columns)]
	local headerH = options.Description and Metrics.HeaderDescH or Metrics.HeaderH
	local root = new("Frame", { Name = options.Name, Size = Dim2(1, 0, 0, headerH) }, column.Scroll)
	local header = new("TextButton", { Size = Dim2(1, 0, 0, headerH - 8) }, root)
	local hovering = false
	local titleColor = function() return hovering and Current.Text or Current.InactiveText end
	local icon = makeIcon(header, options.Icon or "cube", 16, titleColor, "Tab")
	local title = label(header, { Text = options.Name, X = 27, Fg = titleColor, Truncate = Enum.TextTruncate.AtEnd })
	if options.Description then
		label(header, { Fg = "InactiveText", Text = options.Description, TextSize = 11, RowH = 14, Y = 18, Truncate = Enum.TextTruncate.AtEnd })
	end
	local section
	local card = new("Frame", {
		Bg = "Inline", Active = true, Position = Offset(1, headerH + 1), Size = Dim2(1, -2, 0, 0),
		Radius = Metrics.CardRadius,
	}, root)
	local stroke = new("UIStroke", { ApplyStrokeMode = Enum.ApplyStrokeMode.Border, Thickness = 1 }, card)
	bind(stroke, "Color", function() return section and section.Dragging and Current.Accent or Current.Outline end)
	local body = new("Frame", { Position = Offset(Metrics.Pad - 1, -1), Size = Dim2(1, -(Metrics.Pad - 1) * 2, 1, 2) }, card)
	section = setmetatable({
		Window = win, Tab = self, Column = column, Name = options.Name, Root = root, Card = card, Body = body,
		Items = {}, HeaderH = headerH, TotalH = headerH, CardH = 0, Y = 0, Dragging = false, Dropping = false, Title = title,
	}, Section)
	table.insert(column.Sections, section)
	table.insert(win.Sections, section)
	hoverable(header, function() hovering = true; icon.Update(); refresh(title) end, function() hovering = false; icon.Update(); refresh(title) end)
	header.MouseButton1Down:Connect(function() section:_beginDrag() end)
	section:Layout(false)
	return section
end

local function guiParent(screen)
	local ok = false
	if typeof(gethui) == "function" then
		ok = pcall(function() screen.Parent = gethui() end)
	end
	if not ok then
		ok = pcall(function() screen.Parent = game:GetService("CoreGui") end)
	end
	if not ok or not screen.Parent then
		screen.Parent = Players.LocalPlayer:WaitForChild("PlayerGui")
	end
end

function Window:Category(name)
	local y = self.SidebarY + (self.SidebarHasTabs and (Metrics.TabToCategory + ((self.LastTabH or Metrics.TabH) - Metrics.TabH) / 2) or 0)
	self.LastTabH = Metrics.TabH
	label(self.Sidebar, { Fg = "InactiveText", Text = name, TextSize = 9, X = Metrics.Pad, RowH = 12, Y = y - 6, W = 120 })
	self.SidebarY = y
	self.SidebarAfterCategory = true
end

function Window:Tab(options)
	-- a tab with a description is taller: name on the first line, description under it
	local description = options.Description
	local hasDesc = description ~= nil and description ~= ""
	local th, nameH, descH = Metrics.TabH, 0, 0
	if hasDesc then
		nameH, descH = 16, 12
		th = math.max(Metrics.TabH + 8, 8 + nameH + descH)
		th = th + th % 2
	end
	local lastH = self.LastTabH or Metrics.TabH
	local y = self.SidebarY + (self.SidebarAfterCategory and Metrics.CategoryToTab or (Metrics.TabPitch + (lastH - Metrics.TabH) / 2)) + (th - Metrics.TabH) / 2
	self.SidebarAfterCategory, self.SidebarHasTabs = false, true
	self.SidebarY, self.LastTabH = y, th

	local tab = setmetatable({ Window = self, Name = options.Name, Y = y - th / 2, H = th }, Tab)
	local button = new("TextButton", { Position = Offset(Metrics.Pad, tab.Y), Size = Offset(Metrics.TabW, th), ZIndex = 2 }, self.Sidebar)
	tab.Button = button
	local hovering = false
	local isActive = function() return self.CurrentTab == tab end
	local color = function() return (isActive() or hovering) and Current.Text or Current.InactiveText end
	local hole = function() return isActive() and Current.Inline or Current.Background end
	tab.Icon = makeIcon(button, options.Icon or "cube", 16, color, hole, { Position = Offset(6, floor((th - 16) / 2)) })
	local textTop = hasDesc and floor((th - nameH - descH) / 2) or 0
	tab.Label = label(button, { Text = options.Name, X = 33, Y = textTop, TextSize = 13, RowH = hasDesc and nameH or th, Fg = color, Right = 6, Truncate = Enum.TextTruncate.AtEnd })
	tab.LabelY = tab.Label.Position.Y.Offset
	if hasDesc then
		tab.Desc = label(button, { Text = description, X = 33, Y = textTop + nameH - 1, TextSize = 10, RowH = descH, Fg = "InactiveText", Right = 6, Truncate = Enum.TextTruncate.AtEnd })
		tab.DescY = tab.Desc.Position.Y.Offset
	end

	tab.Page = new("Frame", { Name = options.Name, Size = Dim2(1, 0, 1, 0), Visible = false }, self.Pages)
	tab.Columns = {}
	for index = 1, 2 do
		local scroll = new("ScrollingFrame", {
			Position = Offset(Metrics.Pad + (index - 1) * (self.ColumnW + Metrics.Pad), 0),
			Size = Dim2(0, self.ColumnW, 1, 0), ScrollBarThickness = 0, CanvasSize = Offset(0, 0),
			ScrollingEnabled = not UserInputService.MouseEnabled, ClipsDescendants = true,
		}, tab.Page)
		local column = setmetatable({ Scroll = scroll, Sections = {}, Target = 0, ContentH = 0, Tab = tab, ShowTop = false, ShowBottom = false }, Column)
		local function edge(position, rotation)
			local frame = new("Frame", { Bg = "Tab", BackgroundTransparency = 1, Position = position, Size = Offset(self.ColumnW, 14), ZIndex = 4 }, tab.Page)
			new("UIGradient", { Rotation = rotation, Transparency = NumberSequence.new(1, 0) }, frame)
			return frame
		end
		local edgeX = Metrics.Pad + (index - 1) * (self.ColumnW + Metrics.Pad)
		column.FadeTop, column.FadeBottom = edge(Offset(edgeX, 0), 270), edge(Dim2(0, edgeX, 1, -14), 90)
		table.insert(self.Columns, column)
		tab.Columns[index] = column
	end

	local function paint()
		tab.Icon.Update()
		refresh(tab.Label)
		local x = hovering and not isActive() and 35 or 33
		animate(tab.Label, "Position", Offset(x, tab.LabelY), "Smooth")
		if tab.Desc then animate(tab.Desc, "Position", Offset(x, tab.DescY), "Smooth") end
	end
	tab.Paint = paint
	hoverable(button, function() hovering = true; paint() end, function() hovering = false; paint() end)
	button.MouseButton1Click:Connect(function() self:SelectTab(tab) end)
	table.insert(self.Tabs, tab)
	if not self.CurrentTab then self:SelectTab(tab, true) end
	return tab
end

function Window:SelectTab(tab, instant)
	if self.CurrentTab == tab then return end
	local previous = self.CurrentTab
	self.CurrentTab = tab
	self:CloseFloating()
	if previous then
		previous.Page.Visible = false
		previous.Paint()
	end
	tab.Page.Visible = true
	tab.Paint()
	self.Indicator.Visible = true
	place(self.Indicator, "Position", Offset(Metrics.Pad + 1, tab.Y + 1), not instant, "Soft")
	place(self.Indicator, "Size", Offset(Metrics.TabW - 2, (tab.H or Metrics.TabH) - 2), not instant, "Soft")
	if not instant then
		self.Veil.BackgroundTransparency = 0
		animate(self.Veil, "BackgroundTransparency", 1, "Soft")
		tab.Page.Position = Offset(0, 10)
		animate(tab.Page, "Position", Offset(0, 0), "Soft")
	end
end

function Window:SetHotkeyActive(object, active)
	local entry = self.HotkeyEntries[object]
	local shouldShow = active and object.Key ~= nil
	if shouldShow and not entry then
		local frame = new("Frame", { Bg = "Background", Size = Offset(40, 24), Radius = Metrics.FloatRadius, Border = "Outline", Visible = false }, self.Hotkeys)
		new("Frame", { Bg = "Accent", Position = Offset(1, 5), Size = Offset(2, 14), Radius = Metrics.SmallRadius }, frame)
		local text = label(frame, { TextSize = 12, X = Metrics.Pad, RowH = 24, Weight = Enum.FontWeight.SemiBold })
		entry = { Frame = frame, Label = text, Fader = newFader(frame) }
		self.HotkeyEntries[object] = entry
		table.insert(self.HotkeyOrder, object)
		text:GetPropertyChangedSignal("TextBounds"):Connect(function()
			frame.Size = Offset(math.ceil(text.TextBounds.X) + Metrics.Pad * 2, 24)
		end)
		setText(text, (object.Name or "Key") .. " - " .. keyName(object.Key))
		frame.Size = Offset(math.ceil(text.TextBounds.X) + Metrics.Pad * 2, 24)
		frame.Position = Offset(-14, 33 * #self.HotkeyOrder + 1)
		entry.Fader:Show()
	elseif entry and shouldShow then
		setText(entry.Label, (object.Name or "Key") .. " - " .. keyName(object.Key))
		return
	elseif entry then
		self.HotkeyEntries[object] = nil
		table.remove(self.HotkeyOrder, table.find(self.HotkeyOrder, object))
		animate(entry.Frame, "Position", Offset(-14, entry.Frame.Position.Y.Offset), "Smooth")
		entry.Fader:Hide(function() entry.Frame:Destroy() end)
	else
		return
	end
	for index, key in self.HotkeyOrder do
		animate(self.HotkeyEntries[key].Frame, "Position", Offset(1, 33 * index + 1), "Settle")
	end
end

function Window:Notify(options)
	if type(options) == "string" then options = { Text = options } end
	local frame = new("Frame", { Bg = "Background", Size = Offset(60, 28), Radius = Metrics.FloatRadius, Border = "Outline", Visible = false }, self.Notifications)
	makeIcon(frame, options.Icon or "check", 14, function() return options.IconColor or RGB(112, 205, 132) end, "Background", { Position = Offset(8, 7) })
	local text = label(frame, { Text = options.Text, TextSize = 12, X = 28, RowH = 28, W = 340, Truncate = Enum.TextTruncate.AtEnd })
	local entry = { Frame = frame, Fader = newFader(frame) }
	local function fit() frame.Size = Offset(math.min(math.ceil(text.TextBounds.X), 340) + 38, 28) end
	text:GetPropertyChangedSignal("TextBounds"):Connect(fit)
	fit()
	table.insert(self.NotifyStack, 1, entry)
	local function stack()
		local count = #self.NotifyStack
		for index, item in self.NotifyStack do
			local y = (count - index) * 36
			if not item.Placed then
				item.Placed = true
				item.Frame.Position = Offset(-36, y)
			end
			animate(item.Frame, "Position", Offset(0, y), "Settle")
		end
	end
	function entry.Dismiss()
		if entry.Gone then return end
		entry.Gone = true
		table.remove(self.NotifyStack, table.find(self.NotifyStack, entry))
		animate(frame, "Position", Offset(-36, frame.Position.Y.Offset), "Smooth")
		entry.Fader:Hide(function() frame:Destroy() end)
		stack()
	end
	stack()
	entry.Fader:Show()
	-- Rapid actions shouldn't bury the screen.
	if #self.NotifyStack > 6 then self.NotifyStack[#self.NotifyStack].Dismiss() end
	task.delay(options.Duration or 3, entry.Dismiss)
end

function Window:_flushThemeColors(dt)
	self.ThemeClock += dt
	if next(self.PendingThemeKeys) == nil or self.ThemeClock < 1 / 30 then return end
	self.ThemeClock = 0
	for key in self.PendingThemeKeys do
		local set = ByKey[key]
		if set then
			for inst, prop in set do snap(inst, prop, Current[key]) end
		end
		self.PendingThemeKeys[key] = nil
	end
	for inst in Dynamic do refresh(inst, false) end
end

function Window:ClosePopup()
	local popup = self.ActivePopup
	if not popup then return end
	self.ActivePopup = nil
	self.LastPopupAnchor, self.LastPopupTime = popup.Anchor, os.clock()
	if popup.OnClose then popup.OnClose() end
	local position = popup.Frame.Position
	animate(popup.Frame, "Position", Offset(position.X.Offset, position.Y.Offset - 4), "Fast")
	popup.Fader:Hide()
end

function Window:_openPopup(popup, anchor, absolute)
	if self.LastPopupAnchor == anchor and os.clock() - self.LastPopupTime < 0.2 then return false end
	self:ClosePopup()
	local point = toLayer(absolute)
	local viewport, size = self.Screen.AbsoluteSize, popup.Frame.Size
	local x = math.clamp(floor(point.X), 8, math.max(8, viewport.X - size.X.Offset - 8))
	local y = math.clamp(floor(point.Y), 8, math.max(8, viewport.Y - size.Y.Offset - 8))
	popup.Anchor = anchor
	popup.Frame.Position = Offset(x, y - 6)
	animate(popup.Frame, "Position", Offset(x, y), "Settle")
	popup.Fader:Show()
	self.ActivePopup = popup
	return true
end

function Window:_clamped(target, x, y)
	local screen, size, anchor = self.Screen.AbsoluteSize, target.AbsoluteSize, target.AnchorPoint
	local keep = math.min(60, size.X)
	local left = math.clamp(x - anchor.X * size.X, keep - size.X, math.max(keep - size.X, screen.X - keep))
	local top = math.clamp(y - anchor.Y * size.Y, 0, math.max(0, screen.Y - 30))
	return Offset(round(left + anchor.X * size.X), round(top + anchor.Y * size.Y))
end

function Window:_modalPosition(offsetY)
	return Offset(floor((self.Width - 420) / 2), floor((self.Height - 340) / 2) + offsetY)
end

-- Sizes stay even so a centre anchor still lands on whole pixels.
local function evenClamp(value, minimum, maximum)
	local clamped = math.clamp(floor(value), minimum, math.max(minimum, maximum))
	return clamped - clamped % 2
end

-- The tab list and search box must both fit, however many tabs were added.
function Window:_minHeight()
	local needed = math.ceil(self.SidebarY + (self.LastTabH or Metrics.TabH) / 2 + Metrics.Gap + Metrics.SearchH + Metrics.Pad)
	return evenClamp(math.max(self.MinSize.Y, needed + needed % 2), 0, math.huge)
end

function Window:_applySize(width, height)
	self.Width, self.Height = width, height
	self.ColumnW = (width - Metrics.ContentX - Metrics.Pad * 3) / 2
	self.Holder.Size = Offset(width, height)
	for _, tab in self.Tabs do
		for index, column in tab.Columns do
			local x = Metrics.Pad + (index - 1) * (self.ColumnW + Metrics.Pad)
			column.Scroll.Position = Offset(x, 0)
			column.Scroll.Size = Dim2(0, self.ColumnW, 1, 0)
			column.FadeTop.Position, column.FadeTop.Size = Offset(x, 0), Offset(self.ColumnW, 14)
			column.FadeBottom.Position, column.FadeBottom.Size = Dim2(0, x, 1, -14), Offset(self.ColumnW, 14)
			column:ClampScroll()
			if column.Scroll.CanvasPosition.Y > column.Target then
				snap(column.Scroll, "CanvasPosition", Vec2(0, column.Target))
			end
		end
	end
	for _, job in self.ResizeJobs do job() end
	if self.Search and not self.Search.Open then self.Search.Modal.Position = self:_modalPosition(0) end
end

-- Resizes from code, keeping the window's top-left corner where it is.
function Window:SetSize(width, height)
	width = evenClamp(width, self.MinSize.X, math.huge)
	height = evenClamp(height, self:_minHeight(), math.huge)
	local position = self.Holder.Position
	snap(self.Holder, "Position", Offset(position.X.Offset + (width - self.Width) / 2, position.Y.Offset + (height - self.Height) / 2))
	self.Moved[self.Holder] = true
	self:_applySize(width, height)
end

function Window:_beginResize(axisX, axisY, onEnd)
	if self.Search.Open then return end
	self:CloseFloating()
	local start, startW, startH = mousePos(), self.Width, self.Height
	local minHeight = self:_minHeight()
	local position = self.Holder.Position
	local left, top = position.X.Offset - startW / 2, position.Y.Offset - startH / 2
	beginDrag(function(point)
		local delta, screen = point - start, self.Screen.AbsoluteSize
		local width = axisX == 1 and evenClamp(startW + delta.X, self.MinSize.X, screen.X - left - 4) or startW
		local height = axisY == 1 and evenClamp(startH + delta.Y, minHeight, screen.Y - top - 4) or startH
		if width == self.Width and height == self.Height then return end
		self.Moved[self.Holder] = true
		snap(self.Holder, "Position", Offset(left + width / 2, top + height / 2))
		self:_applySize(width, height)
	end, onEnd)
end

function Window:_buildResize()
	local hovering, resizing = 0, false
	local color = function() return (hovering > 0 or resizing) and Current.InactiveText or Current.Outline end
	local grip = makeIcon(self.Root, "grip", 12, color, "Tab", { AnchorPoint = Vec2(1, 1), Position = Dim2(1, -2, 1, -2), ZIndex = 15 })
	local function handle(name, position, size, anchor, axisX, axisY)
		local button = new("TextButton", { Name = name, AnchorPoint = anchor, Position = position, Size = size, ZIndex = 16 }, self.Root)
		hoverable(button, function() hovering += 1; grip.Update() end, function() hovering -= 1; grip.Update() end)
		button.MouseButton1Down:Connect(function()
			resizing = true
			grip.Update()
			self:_beginResize(axisX, axisY, function()
				resizing = false
				grip.Update()
			end)
		end)
		table.insert(self.ResizeHandles, button)
	end
	handle("ResizeRight", Dim2(1, 3, 0, 12), Dim2(0, 6, 1, -30), Vec2(1, 0), 1, 0)
	handle("ResizeBottom", Dim2(0, 12, 1, 3), Dim2(1, -30, 0, 6), Vec2(0, 1), 0, 1)
	handle("ResizeCorner", Dim2(1, 3, 1, 3), Offset(18, 18), Vec2(1, 1), 1, 1)
end

function Window:_makeDraggable(handle, target, allowed)
	handle.InputBegan:Connect(function(input)
		if not isClick(input) then return end
		local start, origin = mousePos(), target.Position
		if allowed and not allowed(start) then return end
		beginDrag(function(point)
			local delta = point - start
			if delta.Magnitude < 1 and not self.Moved[target] then return end
			self.Moved[target] = true
			animate(target, "Position", self:_clamped(target, origin.X.Offset + delta.X, origin.Y.Offset + delta.Y), "Follow")
		end)
	end)
end

-- Input can reach the window background through anything above it, so check the pointer directly.
function Window:_overControl(point)
	if self.Search.Open then return true end
	if self.OpenDropdown and inside(self.OpenDropdown.Panel, point) then return true end
	if self.ActivePopup and inside(self.ActivePopup.Frame, point) then return true end
	if inside(self.SearchButton, point) then return true end
	for _, handle in self.ResizeHandles do
		if inside(handle, point) then return true end
	end
	for _, tab in self.Tabs do
		if inside(tab.Button, point) then return true end
	end
	if self.CurrentTab then
		for _, column in self.CurrentTab.Columns do
			if inside(column.Scroll, point) then
				for _, section in column.Sections do
					if inside(section.Root, point) then return true end
				end
			end
		end
	end
	return false
end

function Window:_popupFrame(width, height)
	local wrapper = new("Frame", { Size = Offset(width, height), Visible = false, Active = true }, self.Popups)
	-- A button, so nothing beneath the popup can take the click.
	local body = new("TextButton", { Bg = "Background", Size = Dim2(1, 0, 1, 0), Radius = Metrics.PopupRadius, Border = "Outline" }, wrapper)
	addShadow(body, "Large")
	local controls = {}
	self:_makeDraggable(body, wrapper, function(point)
		for _, control in controls do
			if inside(control, point, 6) then return false end
		end
		return true
	end)
	return wrapper, body, controls
end

function Window:_buildPicker()
	local pad, gap = Metrics.Pad, Metrics.Gap
	local width = 216
	local svW, svH, barW = width - pad * 3 - 12, 168, 12
	local alphaY = pad + svH + pad
	local labelY = alphaY + barW + pad
	local inputY = labelY + Metrics.Row + gap
	local frame, body, controls = self:_popupFrame(width, inputY + Metrics.InputH + pad)
	local picker = { Frame = frame, Fader = newFader(frame), Target = nil, H = 0, S = 0, V = 1, Alpha = 0, Dragging = false }

	local sv = new("Frame", { BackgroundTransparency = 0, Position = Offset(pad + 1, pad + 1), Size = Offset(svW - 2, svH - 2), Radius = Metrics.Radius, Border = "Outline" }, body)
	local white = new("Frame", { BackgroundTransparency = 0, BackgroundColor3 = Color3.new(1, 1, 1), Size = Dim2(1, 0, 1, 0), Radius = Metrics.Radius }, sv)
	new("UIGradient", { Transparency = NumberSequence.new(0, 1) }, white)
	local black = new("Frame", { BackgroundTransparency = 0, BackgroundColor3 = Color3.new(0, 0, 0), Size = Dim2(1, 0, 1, 0), Radius = Metrics.Radius }, sv)
	new("UIGradient", { Rotation = 90, Transparency = NumberSequence.new(1, 0) }, black)

	local function ringKnob(parent)
		local size = Metrics.PickerKnob
		local knob = new("Frame", { Name = "Knob", BackgroundTransparency = 0, AnchorPoint = Vec2(0.5, 0.5), Size = Offset(size, size), Radius = FULL, ZIndex = 3 }, parent)
		local ring = new("UIStroke", { Thickness = 2 }, knob)
		bind(ring, "Color", "Text")
		addShadow(knob, "PickerKnob")
		return knob
	end
	local svKnob = ringKnob(sv)
	picker.SVKnob = svKnob
	-- The knob's centre travels 6px inside each edge, matching the original.
	local INSET = (Metrics.PickerKnob + 4) / 2 - 1
	local function along(scale) return round(INSET - INSET * 2 * scale) end
	local function fromPointer(position, origin, length)
		return clamp01((position - origin - INSET) / (length - INSET * 2))
	end

	local hue = new("Frame", { BackgroundTransparency = 0, BackgroundColor3 = Color3.new(1, 1, 1), Position = Offset(pad * 2 + svW + 1, pad + 1), Size = Offset(barW - 2, svH - 2), Radius = Metrics.Radius, Border = "Outline" }, body)
	local keys = {}
	for index = 0, 6 do
		table.insert(keys, ColorSequenceKeypoint.new(index / 6, Color3.fromHSV((index / 6) % 1, 1, 1)))
	end
	new("UIGradient", { Rotation = 90, Color = ColorSequence.new(keys) }, hue)
	local hueKnob = ringKnob(hue)

	local alphaBar = new("Frame", { Bg = "Background", Position = Offset(pad + 1, alphaY + 1), Size = Offset(width - pad * 2 - 2, barW - 2), Radius = Metrics.Radius, Border = "Outline" }, body)
	-- A CanvasGroup is the only container whose UICorner clips its children, so the square
	-- checker cells stay inside the rounded ends.
	local alphaClip = new("CanvasGroup", { Name = "AlphaClip", BackgroundTransparency = 0, BackgroundColor3 = Color3.new(1, 1, 1), Size = Dim2(1, 0, 1, 0), Radius = Metrics.Radius }, alphaBar)
	for column = 0, 48 do
		for row = 0, 2 do
			if (column + row) % 2 == 1 then
				new("Frame", { BackgroundTransparency = 0, BackgroundColor3 = RGB(190, 190, 190), Position = Offset(column * 4, row * 4), Size = Offset(4, 4) }, alphaClip)
			end
		end
	end
	local alphaFill = new("Frame", { BackgroundTransparency = 0, Size = Dim2(1, 0, 1, 0) }, alphaClip)
	new("UIGradient", { Transparency = NumberSequence.new(0, 1) }, alphaFill)
	local alphaKnob = ringKnob(alphaBar)

	label(body, { Text = "Animations", X = pad, Y = labelY })
	local holder = new("Frame", { Position = Offset(pad, inputY), Size = Offset(width - pad * 2, Metrics.InputH) }, body)
	for _, control in { sv, hue, alphaBar, holder } do table.insert(controls, control) end

	local function paint()
		local color = Color3.fromHSV(picker.H, picker.S, picker.V)
		local pure = Color3.fromHSV(picker.H, 1, 1)
		sv.BackgroundColor3 = pure
		svKnob.BackgroundColor3 = color
		svKnob.Position = Dim2(picker.S, along(picker.S), 1 - picker.V, along(1 - picker.V))
		hueKnob.Position = Dim2(0.5, 0, picker.H, along(picker.H))
		hueKnob.BackgroundColor3 = pure
		alphaFill.BackgroundColor3 = color
		alphaKnob.Position = Dim2(picker.Alpha, along(picker.Alpha), 0.5, 0)
		alphaKnob.BackgroundColor3 = color
	end

	local function commit()
		paint()
		if picker.Target then picker.Target:Set(Color3.fromHSV(picker.H, picker.S, picker.V), picker.Alpha) end
	end

	function picker.Sync()
		local target = picker.Target
		if not target then return end
		local h, s, v = target.Value:ToHSV()
		if s > 0 and v > 0 then picker.H = h end
		picker.S, picker.V, picker.Alpha = s, v, target.Alpha or 0
		paint()
		if picker.Animation then picker.Animation:Set(target.Animation or "None", true) end
	end

	-- Each hit area overhangs its control so knobs resting on an edge can still be grabbed.
	local function dragArea(area, knob, update)
		local hit = new("TextButton", { Position = Offset(-6, -6), Size = Dim2(1, 12, 1, 12), ZIndex = 5 }, area)
		hit.InputBegan:Connect(function(input)
			if not isClick(input) then return end
			picker.Dragging = true
			update(mousePos())
			beginDrag(update, function() picker.Dragging = false end)
		end)
	end
	dragArea(sv, svKnob, function(point)
		picker.S = fromPointer(point.X, sv.AbsolutePosition.X, sv.AbsoluteSize.X)
		picker.V = 1 - fromPointer(point.Y, sv.AbsolutePosition.Y, sv.AbsoluteSize.Y)
		commit()
	end)
	dragArea(hue, hueKnob, function(point)
		picker.H = math.min(fromPointer(point.Y, hue.AbsolutePosition.Y, hue.AbsoluteSize.Y), 0.9999)
		commit()
	end)
	dragArea(alphaBar, alphaKnob, function(point)
		picker.Alpha = fromPointer(point.X, alphaBar.AbsolutePosition.X, alphaBar.AbsoluteSize.X)
		commit()
	end)

	self.Picker = picker
	picker.Animation = buildDropdown(self, holder, {
		Options = { "None", "Rainbow", "Pulse" }, Default = "None", MaxRows = 3,
		Callback = function(value)
			if picker.Target then picker.Target.Animation = value end
		end,
	})
	picker.OnClose = function() picker.Animation:Toggle(false) end
end

function Window:_buildSearch()
	local width, height, pad = 420, 340, Metrics.Pad
	local overlay = new("TextButton", {
		BackgroundTransparency = 0.45, BackgroundColor3 = Color3.new(0, 0, 0), Size = Dim2(1, 0, 1, 0), Visible = false,
		ZIndex = 20, Radius = Metrics.WindowRadius,
	}, self.Root)
	local modal = new("TextButton", {
		Bg = "Background", Position = self:_modalPosition(0), Size = Offset(width, height), Radius = Metrics.WindowRadius, Border = "Outline",
	}, overlay)
	local title = label(modal, { Text = "Search", TextSize = 14, Weight = Enum.FontWeight.SemiBold, X = pad * 2, Y = pad + 2, RowH = 20 })
	local field = new("Frame", { Bg = "Inline", Position = Offset(pad * 2 + 1, 43), Size = Offset(width - pad * 4 - 2, 34), Radius = Metrics.PillRadius, Border = "Outline" }, modal)
	makeIcon(field, "search", 14, "InactiveText", "Inline", { Position = Offset(10, 10) })
	local box = new("TextBox", {
		Fg = "Text", PlaceholderText = "Type to search...", Position = Offset(34, capOffset(34, 13)), Size = Dim2(1, -45, 0, 20), ClearTextOnFocus = false,
	}, field)
	bind(box, "PlaceholderColor3", "InactiveText")
	local results = new("ScrollingFrame", {
		Position = Offset(pad * 2, 88), Size = Offset(width - pad * 4 + 6, height - 88 - pad), CanvasSize = Offset(0, 0),
		ScrollBarThickness = 2, ScrollBarImageTransparency = 0,
	}, modal)
	bind(results, "ScrollBarImageColor3", "Accent")

	local noResults = centeredLabel(modal, width, { Fg = "InactiveText", Text = "No results", Y = 170 })
	noResults.Visible = false
	local search = { Overlay = overlay, Modal = modal, Fader = newFader(overlay), Open = false, Rows = {}, Dirty = false, Box = box, Order = {} }

	local function goTo(entry)
		self:CloseSearch()
		local section = entry.Section or entry
		self:SelectTab(section.Tab)
		section.Column:ScrollTo(section.Y + (entry.Section and section.HeaderH + entry.Y or 0) - 40)
		if entry.Frame then
			local flash = new("Frame", {
				Bg = "Accent", BackgroundTransparency = 0.7, Position = Offset(-6, -3), Size = Dim2(1, 12, 1, 6), Radius = Metrics.Radius, ZIndex = 0,
			}, entry.Frame)
			task.delay(0.2, function()
				animate(flash, "BackgroundTransparency", 1, "Soft", { Done = function() flash:Destroy() end })
			end)
		end
	end

	local function retire(row)
		row.Retired = true
		row.Frame:Destroy()
	end

	local function makeRow(entry, name, sub, y)
		local frame = new("TextButton", {
			Bg = "HoveredElement", BackgroundTransparency = 1, Position = Offset(10, y), Size = Offset(width - pad * 4, 44), Radius = Metrics.Radius,
		}, results)
		NoFade[frame] = true
		local row = { Frame = frame, Y = y, Retired = false }
		row.Bar = new("Frame", { Bg = "Accent", BackgroundTransparency = 1, Position = Offset(4, 8), Size = Offset(3, 28), Radius = Metrics.SmallRadius }, frame)
		row.Name = label(frame, { Text = name, TextSize = 13, X = 14, Y = 5, RowH = 18, TextTransparency = 1, Right = 10, Truncate = Enum.TextTruncate.AtEnd })
		row.Sub = label(frame, { Fg = "InactiveText", Text = sub, TextSize = 11, X = 14, Y = 22, RowH = 16, TextTransparency = 1, Right = 10, Truncate = Enum.TextTruncate.AtEnd })
		hoverable(frame, function() animate(frame, "BackgroundTransparency", 0.4, "Fast") end, function() animate(frame, "BackgroundTransparency", 1, "Fast") end)
		frame.MouseButton1Click:Connect(function() goTo(entry) end)
		return row
	end

	-- Rows are matched to results and reused, so a keystroke only touches what actually changed.
	local function sync()
		local text = box.Text
		setText(title, text ~= "" and text or "Search")
		setText(self.SearchLabel, text ~= "" and text or "Search...")
		local wanted, order = {}, {}
		if text ~= "" then
			local needle = string.lower(text)
			local function add(entry, name, sub)
				if #order < 30 and name and string.find(string.lower(name), needle, 1, true) then
					wanted[entry] = true
					table.insert(order, { entry, name, sub })
				end
			end
			for _, section in self.Sections do
				add(section, section.Name, section.Tab.Name .. " / " .. section.Name .. "  \u{00B7}  Section")
			end
			for _, item in self.Index do
				add(item, item.Name, item.Section.Tab.Name .. " / " .. item.Section.Name .. "  \u{00B7}  " .. item.Kind)
			end
		end
		for entry, row in search.Rows do
			if not wanted[entry] then
				search.Rows[entry] = nil
				retire(row)
			end
		end
		local fresh = 0
		for index, data in order do
			local entry, y = data[1], (index - 1) * 48
			local row = search.Rows[entry]
			if row then
				row.Y = y
				animate(row.Frame, "Position", Offset(0, y), "Smooth")
			else
				row = makeRow(entry, data[2], data[3], y)
				search.Rows[entry] = row
				fresh += 1
				task.delay(math.min(fresh, 8) * 0.025, function()
					if row.Retired then return end
					animate(row.Frame, "Position", Offset(0, row.Y), "Smooth")
					animate(row.Bar, "BackgroundTransparency", 0, "Smooth")
					animate(row.Name, "TextTransparency", 0, "Smooth")
					animate(row.Sub, "TextTransparency", 0, "Smooth")
				end)
			end
		end
		results.CanvasSize = Offset(0, #order * 48)
		search.Order = order
		noResults.Visible = text ~= "" and #order == 0
	end

	box:GetPropertyChangedSignal("Text"):Connect(function() search.Dirty = true end)
	box.FocusLost:Connect(function(enterPressed)
		if UserInputService:IsKeyDown(Enum.KeyCode.Escape) then
			self:CloseSearch()
		elseif enterPressed then
			if search.Dirty then
				search.Dirty = false
				sync()
			end
			if search.Order[1] then goTo(search.Order[1][1]) end
		end
	end)
	overlay.MouseButton1Click:Connect(function() self:CloseSearch() end)
	search.Sync = sync
	self.Search = search
end

function Window:OpenSearch()
	local search = self.Search
	if search.Open then return end
	search.Open = true
	self:CloseFloating()
	search.Modal.Position = self:_modalPosition(10)
	animate(search.Modal, "Position", self:_modalPosition(0), "Smooth")
	search.Fader:Show()
	search.Box:CaptureFocus()
end

function Window:CloseSearch()
	local search = self.Search
	if not search.Open then return end
	search.Open = false
	search.Box:ReleaseFocus()
	animate(search.Modal, "Position", self:_modalPosition(7), "Fast")
	search.Fader:Hide()
end

function Window:SetVisible(visible)
	if self.Visible == visible then return end
	self.Visible = visible
	self:CloseFloating()
	self.ModalButton.Visible = visible
	if not self.Scale.Parent then
		if not visible then self.Scale.Scale = 1 end
		self.Scale.Parent = self.Holder
	end
	if visible then
		animate(self.Scale, "Scale", 1, "Smooth", {
			-- A UIScale is removed once idle so text is never resampled at rest.
			Done = function()
				if self.Visible then self.Scale.Parent = nil end
			end,
		})
		self.Fader:Show()
	else
		animate(self.Scale, "Scale", 0.97, "Fast")
		self.Fader:Hide()
	end
end

	self:SetVisible(not self.Visible)
end

local ConfigMemory = {}
local function configPath(name) return "Blossom/" .. name .. ".json" end

function Window:_serialize()
	local data = {}
	for flag, object in self.Flags do
		local value = object.Value
		if object.Pack then
			value = object:Pack()
		elseif object.Swatch then
			value = { Color = object.Value:ToHex(), Alpha = object.Alpha }
		elseif object.Chip then
			value = { Key = object.Key and object.Key.Name or false, Mode = object.Mode }
		elseif object.Multi then
			local list = {}
			for option, enabled in object.Value do
				if enabled then table.insert(list, option) end
			end
			value = list
		end
		if value ~= nil then data[flag] = value end
	end
	return HttpService:JSONEncode(data)
end

function Window:_deserialize(json)
	local data = HttpService:JSONDecode(json)
	for flag, value in data do
		local object = self.Flags[flag]
		if object then
			if object.Unpack then
				object:Unpack(value)
			elseif object.Swatch then
				object:Set(Color3.fromHex(value.Color), value.Alpha)
			elseif object.Chip then
				object.Mode = value.Mode or object.Mode
				local key = nil
				if value.Key then
					key = string.find(value.Key, "MouseButton", 1, true) and Enum.UserInputType[value.Key] or Enum.KeyCode[value.Key]
				end
				object:Set(key)
			else
				object:Set(value)
			end
		end
	end
end

function Window:ListConfigs()
	local names = {}
	if typeof(listfiles) == "function" and typeof(isfolder) == "function" and isfolder("Blossom") then
		for _, path in listfiles("Blossom") do
			local name = string.match(path, "([^/\\]+)%.json$")
			if name then table.insert(names, name) end
		end
	else
		for name in ConfigMemory do table.insert(names, name) end
	end
	table.sort(names)
	return names
end

function Window:SaveConfig(name)
	local json = self:_serialize()
	if typeof(writefile) == "function" then
		if typeof(isfolder) == "function" and not isfolder("Blossom") then makefolder("Blossom") end
		writefile(configPath(name), json)
	else
		ConfigMemory[name] = json
	end
end

function Window:LoadConfig(name)
	local json
	if typeof(readfile) == "function" and typeof(isfile) == "function" and isfile(configPath(name)) then
		json = readfile(configPath(name))
	else
		json = ConfigMemory[name]
	end
	if json then self:_deserialize(json); return true end
	return false
end

function Window:DeleteConfig(name)
	if typeof(delfile) == "function" and typeof(isfile) == "function" and isfile(configPath(name)) then
		delfile(configPath(name))
	end
	ConfigMemory[name] = nil
end

local AUTOLOAD_PATH = "Blossom/autoload.txt"
local AutoloadMemory

local function readAutoload()
	if typeof(readfile) == "function" and typeof(isfile) == "function" then
		return isfile(AUTOLOAD_PATH) and readfile(AUTOLOAD_PATH) or nil
	end
	return AutoloadMemory
end

local function writeAutoload(name)
	if typeof(writefile) ~= "function" then
		AutoloadMemory = name
	elseif name then
		if typeof(isfolder) == "function" and not isfolder("Blossom") then makefolder("Blossom") end
		writefile(AUTOLOAD_PATH, name)
	elseif typeof(delfile) == "function" and isfile(AUTOLOAD_PATH) then
		delfile(AUTOLOAD_PATH)
	end
end

function Window:_buildWatermark(options)
	local player = Players.LocalPlayer
	local frame = new("Frame", { Bg = "Background", Size = Offset(298, 28), Radius = Metrics.FloatRadius, Border = "Outline", Active = true }, self.Screen)
	makeIcon(frame, "logo", 16, "Accent", "Background", { Position = Offset(8, 6) })
	local function text(value, size, weight)
		return label(frame, { Text = value, TextSize = size, Weight = weight, RowH = 28, W = 10 })
	end
	local function sep()
		return new("Frame", { Bg = "Outline", Position = Offset(0, 7), Size = Offset(1, 14) }, frame)
	end
	local first, firstSep = text(options.Script or "blossom.lua", 12, Enum.FontWeight.Medium), sep()
	local second, secondSep = text(options.Build or "release", 12, Enum.FontWeight.SemiBold), sep()
	self.WatermarkFirst, self.WatermarkSecond = first, second
	local userIcon = makeIcon(frame, "user", 13, "Text", "Background")
	local third = text(("%s (@%s)"):format(player.DisplayName, player.Name), 11, Enum.FontWeight.SemiBold)
	local function layout()
		local x = 31
		for _, entry in { { first, firstSep }, { second, secondSep } } do
			local width = math.ceil(entry[1].TextBounds.X)
			entry[1].Size = Offset(width + 1, 28)
			entry[1].Position = Offset(x, entry[1].Position.Y.Offset)
			x += width + 8
			entry[2].Position = Offset(x, 7)
			x += 9
		end
		userIcon.Root.Position = Offset(x, 7)
		x += 20
		local width = math.ceil(third.TextBounds.X)
		third.Size = Offset(width + 1, 28)
		third.Position = Offset(x, third.Position.Y.Offset)
		frame.Size = Offset(x + width + Metrics.Pad, 28)
		self:_placeOverlays()
	end
	for _, entry in { first, second, third } do
		entry:GetPropertyChangedSignal("TextBounds"):Connect(layout)
	end
	self.Watermark = frame
	self.WatermarkLayout = layout
	self:_makeDraggable(frame, frame)
end

function Window:_buildHotkeys()
	local holder = new("Frame", { Size = Offset(200, 300) }, self.Screen)
	local head = new("Frame", { Bg = "Background", Position = Offset(1, 1), Size = Offset(88, 25), Radius = Metrics.FloatRadius, Border = "Outline", Active = true }, holder)
	makeIcon(head, "key", 14, "Accent", "Background", { Position = Offset(9, 5) })
	local text = label(head, { Text = "Hotkeys", TextSize = 12, Weight = Enum.FontWeight.SemiBold, X = 31, RowH = 25 })
	local function fit() head.Size = Offset(math.ceil(text.TextBounds.X) + 41, 25) end
	text:GetPropertyChangedSignal("TextBounds"):Connect(fit)
	fit()
	self.Hotkeys = holder
	self:_makeDraggable(head, holder)
end

-- Everything anchored to the screen is placed at whole pixels, the main defence against blurry text.
function Window:_placeOverlays()
	local size = self.Screen.AbsoluteSize
	if size.X <= 0 then return end
	if not self.Moved[self.Holder] then self.Holder.Position = Offset(floor(size.X / 2), floor(size.Y / 2)) end
	if self.Watermark and not self.Moved[self.Watermark] then
		self.Watermark.Position = Offset(floor((size.X - self.Watermark.Size.X.Offset) / 2), size.Y - 44)
	end
	if self.Hotkeys and not self.Moved[self.Hotkeys] then self.Hotkeys.Position = Offset(14, floor((size.Y - 300) / 2)) end
end

function Window:Destroy()
	for _, connection in self.Connections do connection:Disconnect() end
	table.clear(Jobs)
	table.clear(HoverSet)
	for _, entry in Active do entry.Dead = true end
	table.clear(Active)
	table.clear(Tracks)
	if FrameConnection then
		FrameConnection:Disconnect()
		FrameConnection = nil
	end
	if Focus == self then Focus = nil end
	if ScreenRef == self.Screen then ScreenRef = nil end
	self.Screen:Destroy()
end

local Blossom = { Themes = Themes, ThemeOrder = ThemeOrder, Metrics = Metrics, Fonts = Fonts }

local LastWindow

function Blossom.new(options)
	options = options or {}
	local globalEnv = typeof(getgenv) == "function" and getgenv() or nil
	local previous = LastWindow or (globalEnv and globalEnv.__BlossomWindow)
	if previous then pcall(previous.Destroy, previous) end
	if options.Theme and Themes[options.Theme] then
		for key, value in Themes[options.Theme] do Current[key] = value end
	end
	if options.Font then FontFamily = Fonts[options.Font] or options.Font end
	local minSize = options.MinSize or Vec2(Metrics.MinWindowW, Metrics.MinWindowH)
	minSize = Vec2(minSize.X - minSize.X % 2, minSize.Y - minSize.Y % 2)
	local startSize = options.Size or Vec2(Metrics.WindowW, Metrics.WindowH)
	local startW, startH = evenClamp(startSize.X, minSize.X, math.huge), evenClamp(startSize.Y, minSize.Y, math.huge)
	local win = setmetatable({
		Width = startW, Height = startH, MinSize = minSize,
		ColumnW = (startW - Metrics.ContentX - Metrics.Pad * 3) / 2,
		ResizeJobs = {}, ResizeHandles = {},
		Title = options.Title or "Blossom", Tabs = {}, Sections = {}, Columns = {}, Index = {}, Binds = {}, Colors = {}, Flags = {},

		ThemeClock = 0, SidebarY = Metrics.FirstCategory, SidebarHasTabs = false, SidebarAfterCategory = false,
		Visible = true, ThemeName = "Black", MenuKey = options.MenuKey or Enum.KeyCode.RightShift,
		LastPopupTime = 0, DropdownClosedAt = 0,
	}, Window)
	ensureLoop()
	LastWindow, Focus = win, win
	if globalEnv then globalEnv.__BlossomWindow = win end

	local screen = new("ScreenGui", { Name = "Blossom", ResetOnSpawn = false, IgnoreGuiInset = true, ZIndexBehavior = Enum.ZIndexBehavior.Sibling, DisplayOrder = 100 })
	guiParent(screen)
	win.Screen = screen

		Name = "Window", AnchorPoint = Vec2(0.5, 0.5), Size = Offset(win.Width, win.Height), ZIndex = 10,

	win.ModalButton = new("TextButton", { Modal = true, Size = Offset(1, 1) }, holder)

	local sidebar = new("Frame", { Size = Dim2(0, Metrics.SidebarW, 1, 0) }, root)
	win.Sidebar = sidebar
	makeIcon(sidebar, "logo", 20, "Accent", "Background", { Position = Offset(Metrics.Pad * 2, 15) })
	win.TitleLabel = label(sidebar, { Text = win.Title, TextSize = 16, Weight = Enum.FontWeight.Bold, X = 51, Y = 11, RowH = 28, W = 85 })
	win.Indicator = new("Frame", { Bg = "Inline", Size = Offset(Metrics.TabW - 2, Metrics.TabH - 2), Radius = Metrics.PillRadius, Border = "Outline", Visible = false }, sidebar)
	local searchButton = new("TextButton", {

	new("Frame", { Name = "CornerPatch", Bg = "Tab", Size = Dim2(0, 12, 1, 0) }, content)
	win.Pages = new("Frame", {
		Name = "Pages", Position = Offset(0, Metrics.Pad), Size = Dim2(1, 0, 1, -Metrics.Pad * 2), ClipsDescendants = true,
	}, content)
	win.Veil = new("Frame", { Bg = "Tab", BackgroundTransparency = 1, Size = Dim2(1, 0, 1, 0), ZIndex = 5 }, win.Pages)

	win.DragLayer = new("Frame", { Name = "Drag", Size = Dim2(1, 0, 1, 0), ZIndex = 30 }, screen)
	win.Popups = new("Frame", { Name = "Popups", Size = Dim2(1, 0, 1, 0), ZIndex = 40 }, screen)
	win.Dropdowns = new("Frame", { Name = "Dropdowns", Size = Dim2(1, 0, 1, 0), ZIndex = 45 }, screen)
	win.Notifications = new("Frame", { Name = "Notifications", Position = Offset(16, 16), Size = Offset(400, 600), ZIndex = 50 }, screen)

	win:_buildWatermark(options)
	win:_buildHotkeys()
	win:_buildKeybindMenu()
	win:_buildPicker()
	win:_buildSearch()
	if options.Resizable ~= false then win:_buildResize() end
	win:_placeOverlays()
	win.WatermarkLayout()
	table.insert(win.Connections, screen:GetPropertyChangedSignal("AbsoluteSize"):Connect(function() win:_placeOverlays() end))

	win:_makeDraggable(root, holder, function(point) return not win:_overControl(point) end)

	table.insert(win.Connections, UserInputService.InputBegan:Connect(function(input, processed)
		if isClick(input) then
			local point = mousePos()
			local dropdown = win.OpenDropdown
			local overDropdown = dropdown and inside(dropdown.Panel, point)
			if dropdown and not overDropdown then
				if inside(dropdown.Input, point) then
					win.DropdownClosedBy, win.DropdownClosedAt = dropdown.Input, os.clock()
				end
				dropdown:Toggle(false)
			end
			if win.ActivePopup and not overDropdown and not inside(win.ActivePopup.Frame, point, 6) then
				win:ClosePopup()
			end
		end
		local listening = win.Listening
		if listening then
			local isKey = input.UserInputType == Enum.UserInputType.Keyboard
			local isMouse = input.UserInputType == Enum.UserInputType.MouseButton2 or input.UserInputType == Enum.UserInputType.MouseButton3
			if isKey or isMouse then
				win.Listening = nil
				listening.Listening = false
				if isKey and input.KeyCode == Enum.KeyCode.Escape then
					listening.Key = nil
				else
					listening.Key = isKey and input.KeyCode or input.UserInputType
				end
				listening.Refresh()
			elseif isClick(input) then
				win.Listening = nil
				listening.Listening = false
				listening.Refresh()
			end
			return
		end
		if processed then return end
		if input.KeyCode == win.MenuKey then win:Toggle() end
		for _, object in win.Binds do
			if object.Key and (object.Key == input.KeyCode or object.Key == input.UserInputType) and object.OnPress then
				object.OnPress()
			end
		end
	end))

	table.insert(win.Connections, UserInputService.InputEnded:Connect(function(input)
		for _, object in win.Binds do
			if object.Key and (object.Key == input.KeyCode or object.Key == input.UserInputType) and object.OnRelease then
				object.OnRelease()
			end
		end
	end))

	table.insert(win.Connections, UserInputService.InputChanged:Connect(function(input)
		if input.UserInputType ~= Enum.UserInputType.MouseWheel or not win.Visible or win.Search.Open or not win.CurrentTab then return end
		local point = mousePos()
		local dropdown = win.OpenDropdown
		if dropdown then
			if inside(dropdown.Panel, point) then return end
			dropdown:Toggle(false)
		end
		for _, column in win.CurrentTab.Columns do
			if inside(column.Scroll, point) then
				column:ScrollTo(column.Target - input.Position.Z * 60)
			end
		end
	end))

	local clock, colorClock = 0, 0
	local function perFrame(dt)
		clock += dt
		colorClock += dt
		win:_flushThemeColors(dt)
		if win.OpenDropdown then win.OpenDropdown.Follow() end
		if win.CurrentTab then
			for _, column in win.CurrentTab.Columns do
				local y = column.Scroll.CanvasPosition.Y
				local showTop, showBottom = y > 1, y < column.ContentH - column.Scroll.AbsoluteSize.Y - 1
				if showTop ~= column.ShowTop then
					column.ShowTop = showTop
					animate(column.FadeTop, "BackgroundTransparency", showTop and 0 or 1, "Fast")
				end
				if showBottom ~= column.ShowBottom then
					column.ShowBottom = showBottom
					animate(column.FadeBottom, "BackgroundTransparency", showBottom and 0 or 1, "Fast")
				end
			end
		end
		if win.Search.Dirty then
			win.Search.Dirty = false
			win.Search.Sync()
		end
		if colorClock >= 1 / 30 then
			colorClock = 0
			for _, color in win.Colors do
				if color.Animation == "Rainbow" then
					local _, s, v = color.Value:ToHSV()
					color:Set(Color3.fromHSV((clock * 0.25) % 1, math.max(s, 0.5), math.max(v, 0.6)))
				elseif color.Animation == "Pulse" then
					local h, s = color.Value:ToHSV()
					color:Set(Color3.fromHSV(h, s, 0.65 + math.sin(clock * 4) * 0.35))
				end
			end
		end
	end
	Jobs[perFrame] = true
	table.insert(win.Jobs, perFrame)

	holder.Visible, win.Visible = false, false
	win.Watermark.Visible, win.Hotkeys.Visible = false, false
	task.defer(function()
		if win.Screen.Parent == nil then return end
		win:SetVisible(true)
		newFader(win.Watermark):Show()
		newFader(win.Hotkeys):Show()
	end)

	return win
end

-- tooltips ------------------------------------------------------------------

local function hideTip(win)
	if win.Tip then win.Tip.Frame.Visible = false end
	win.TipShown = false
end

local function showTip(win, text)
	local tip = win.Tip
	if not tip then
		local frame = new("Frame", { Bg = "Background", Size = Offset(40, 22), Radius = Metrics.Radius, Border = "Outline", Visible = false }, win.Popups)
		local caption = label(frame, { TextSize = 11, X = 7, RowH = 22, W = 260 })
		tip = { Frame = frame, Text = caption }
		caption:GetPropertyChangedSignal("TextBounds"):Connect(function()
			local width = math.min(math.ceil(caption.TextBounds.X), 260)
			caption.Size = Offset(width + 1, 22)
			frame.Size = Offset(width + 15, 22)
		end)
		win.Tip = tip
	end
	setText(tip.Text, text)
	local screen = win.Screen.AbsoluteSize
	local at = toLayer(mousePos()) + Vec2(14, 20)
	tip.Frame.Position = Offset(math.max(0, math.min(at.X, screen.X - 280)), math.max(0, math.min(at.Y, screen.Y - 30)))
	tip.Frame.Visible = true
	win.TipShown = true
end

local function tipText(item)
	local object = item.Object
	if not object then return nil end
	if object.Locked and object.LockReason and object.LockReason ~= "" then return object.LockReason end
	return object.Description
end

function Window:_tipStep(dt)
	local list = self.TipItems
	if not list or #list == 0 or not self.Visible then
		if self.TipShown then hideTip(self) end
		return
	end
	local point = mousePos()
	local found
	for _, item in ipairs(list) do
		local frame = item.Frame
		if frame.Parent and item.Visible and inside(frame, point) and isShown(frame) and not covered(frame, point) then
			local column = item.Section and item.Section.Column
			if not column or inside(column.Scroll, point) then
				if tipText(item) then found = item break end
			end
		end
	end
	if found ~= self.TipItem then
		self.TipItem, self.TipTime = found, 0
		if self.TipShown then hideTip(self) end
	elseif found and not self.TipShown then
		self.TipTime = (self.TipTime or 0) + dt
		if self.TipTime >= 0.45 then showTip(self, tipText(found)) end
	end
end

-- shared element methods ----------------------------------------------------

local function findCaption(item, name)
	if item.Label and item.Label:IsA("TextLabel") then return item.Label end
	for _, child in ipairs(item.Frame:GetChildren()) do
		if child:IsA("TextLabel") then
			local entry = Cased[child]
			if entry and entry.Text == name then return child end
		end
	end
	return nil
end

local function applyLock(object, locked, reason)
	local item = object.Item
	object.Locked, object.LockReason = locked, reason
	if locked then
		if not item.LockFrame then
			item.LockFrame = new("TextButton", {
				Bg = "Inline", BackgroundTransparency = 0.4, Position = Offset(-4, -3), Size = Dim2(1, 8, 1, 6), ZIndex = 20, Radius = Metrics.Radius,
			}, item.Frame)
		end
		item.LockFrame.Visible = true
	elseif item.LockFrame then
		item.LockFrame.Visible = false
	end
end

local function decorate(section, object, item, options)
	local win = section.Window
	object.Item, object.Section, object.Kind = item, section, item.Kind
	item.Object = object
	object.Title = nameOf(options)
	object.Description = descOf(options)
	object.Initial = object.Chip and object.Key or copyValue(object.Value)
	object.Listeners = {}
	object.UserCallback = object.Callback
	object.Callback = function(...)
		if item.Kind == "Toggle" and options.Sound ~= false then
			playSound((...) and "On" or "Off", options.Sound)
		end
		if object.UserCallback then object.UserCallback(...) end
		for _, listener in ipairs(object.Listeners) do task.spawn(listener, ...) end
	end

	local function add(name, fn)
		if object[name] == nil then object[name] = fn end
	end
	add("Get", function(self)
		if self.Chip then return self.Key end
		return self.Value
	end)
	add("OnChanged", function(self, fn)
		table.insert(self.Listeners, fn)
		return { Disconnect = function()
			local index = table.find(self.Listeners, fn)
			if index then table.remove(self.Listeners, index) end
		end }
	end)
	add("SetCallback", function(self, fn) self.UserCallback = fn return self end)
	add("SetTitle", function(self, text)
		local caption = findCaption(item, self.Title)
		if caption then setText(caption, text) end
		self.Title, item.Name = text, text
		return self
	end)
	add("SetDescription", function(self, text) self.Description = text return self end)
	add("SetLocked", function(self, locked, reason)
		applyLock(self, locked == true, reason)
		return self
	end)
	add("Lock", function(self, reason) return self:SetLocked(true, reason) end)
	add("Unlock", function(self) return self:SetLocked(false) end)
	add("SetVisible", function(self, visible)
		visible = visible ~= false
		self.Hidden = not visible
		item.Visible = visible
		item.Frame.Visible = visible
		section:Layout(true)
		return self
	end)
	add("Reset", function(self)
		if self.Set and self.Initial ~= nil then self:Set(copyValue(self.Initial)) end
		return self
	end)
	add("Save", function(self) return self:Get() end)
	add("Load", function(self, value) if self.Set and value ~= nil then self:Set(value) end return self end)
	add("Matches", function(self, query)
		return string.find(string.lower(tostring(self.Title or "")), string.lower(query), 1, true) ~= nil
	end)
	add("DependsOn", function(self, source, predicate, mode)
		local function check()
			local value = source:Get()
			local ok
			if type(predicate) == "function" then
				ok = predicate(value)
			elseif predicate == nil then
				ok = value ~= nil and value ~= false and value ~= 0 and value ~= ""
			else
				ok = value == predicate
			end
			ok = ok and true or false
			if mode == "Lock" or mode == "lock" then
				self:SetLocked(not ok, "Requires " .. tostring(source.Title or "another option"))
			else
				self:SetVisible(ok)
			end
		end
		source:OnChanged(check)
		check()
		return self
	end)
	add("Highlight", function(self, pulses)
		task.spawn(function()
			if section.Tab and section.Tab.Page and win.CurrentTab ~= section.Tab then win:SelectTab(section.Tab) end
			if section.Column then section.Column:ScrollTo(section.Y + section.HeaderH + item.Y - 40) end
			local glow = new("Frame", { Bg = "Accent", BackgroundTransparency = 1, Position = Offset(-6, -3), Size = Dim2(1, 12, 1, 6), Radius = Metrics.Radius, ZIndex = 0 }, item.Frame)
			for _ = 1, pulses or 2 do
				animate(glow, "BackgroundTransparency", 0.7, "Fast")
				task.wait(0.25)
				animate(glow, "BackgroundTransparency", 1, "Soft")
				task.wait(0.25)
			end
			glow:Destroy()
		end)
		return self
	end)
	add("Destroy", function(self)
		local index = table.find(section.Items, item)
		if index then table.remove(section.Items, index) end
		index = table.find(win.Index, item)
		if index then table.remove(win.Index, index) end
		if win.TipItems then
			index = table.find(win.TipItems, item)
			if index then table.remove(win.TipItems, index) end
		end
		for _, extra in ipairs({ self.Keybind, self.Color, self }) do
			index = table.find(win.Binds, extra)
			if index then table.remove(win.Binds, index) end
			index = table.find(win.Colors, extra)
			if index then table.remove(win.Colors, index) end
		end
		if self.Flag then
			win.Flags[self.Flag] = nil
			Blossom.Flags[self.Flag] = nil
		end
		win.FlagOwners[item] = nil
		if self.OnDestroy then self:OnDestroy() end
		item.Frame:Destroy()
		section:Layout(true)
	end)

	win.TipItems = win.TipItems or {}
	table.insert(win.TipItems, item)
	if object.Flag then Blossom.Flags[object.Flag] = object end
	if options.Locked then object:SetLocked(true, options.LockedReason or options.LockReason) end
	if options.Visible == false then object:SetVisible(false) end
end

	register(section, object, {
		NoSave = true, Name = nameOf(o), Description = descOf(o), Locked = o.Locked,
		LockedReason = o.LockedReason or o.LockReason, Visible = o.Visible,

-- a section inside a section: groups, columns and panels all use it
local function subSection(parent, body, fit, innerW, name)
	local sub = setmetatable({
		Window = parent.Window, Tab = parent.Tab, Name = name, Body = body, Items = {}, HeaderH = 0, TotalH = 0, CardH = 0,
		Dragging = false, Dropping = false, Parent = parent, InnerW = innerW, ContentH = 0,
	}, Section)
	sub._item = function(self, height, kind, itemName) return Section._item(self, height, kind, itemName, false) end
	sub.Layout = function(self, animated)
		local y, any = 0, false
		for _, item in ipairs(self.Items) do
			if item.Visible then
				item.Y, any = y, true
				place(item.Frame, "Position", Offset(0, y), animated)
				y = y + item.Height + Metrics.Gap
			end
		end
		if any then y = y - Metrics.Gap end
		self.ContentH = y
		fit(y, animated)
	end
	return sub
end

-- Label / Heading / Divider / Spacer ----------------------------------------

function Section:Label(o)
	local name = nameOf(o) or ""
	local size = o.Size or 12
	local item = self:_item(o.Height or Metrics.Row, "Label", name, false)
	local hasValue = o.Value ~= nil
	local text = label(item.Frame, { Text = name, TextSize = size, Weight = o.Weight, Truncate = AtEnd, Fg = o.TextColor or "Text" })
	item.Label = text
	local object = { Item = item, Instance = text, Value = name }
	local valueColor = o.ValueColor
	local valueLabel
	if hasValue then
		valueLabel = label(item.Frame, { Text = tostring(o.Value), TextSize = size, Truncate = AtEnd, Fg = function() return colorOf(valueColor, "InactiveText") end })
		valueLabel.TextXAlignment = Enum.TextXAlignment.Right
		valueLabel.Position = Dim2(0.55, 0, 0, valueLabel.Position.Y.Offset)
		valueLabel.Size = Dim2(0.45, 0, 0, Metrics.Row)
		text.Size = Dim2(0.55, -Metrics.Gap, 0, Metrics.Row)
	end
	function object:Set(value)
		self.Value = tostring(value)
		setText(text, self.Value)
	end
	function object:SetText(value) self:Set(value) end
	function object:SetValue(value)
		if valueLabel then setText(valueLabel, tostring(value)) end
	end
	function object:SetValueColor(color)
		valueColor = color
		if valueLabel then refresh(valueLabel) end
	end
	object.Keybind, object.Color = addInlineControls(self, item, o, name)
	return plain(self, object, item, o)
end

function Section:Heading(o)
	o.Size, o.Weight, o.Height = o.Size or 15, Bold, o.Height or 22
	return self:Label(o)
end

function Section:Divider(o)
	o = o or {}
	local text = o.Text or o.Title or o.Name
	local item = self:_item(text and Metrics.Row or Metrics.DividerH, "Divider", nil, false)
	if text then
		new("Frame", { Bg = "Outline", Position = Offset(0, 7), Size = Offset(10, 1) }, item.Frame)
		local caption = label(item.Frame, { Fg = "InactiveText", Text = text, TextSize = 10, X = 16, W = 10 })
		local right = new("Frame", { Bg = "Outline", Position = Offset(30, 7), Size = Dim2(1, -30, 0, 1) }, item.Frame)
		caption:GetPropertyChangedSignal("TextBounds"):Connect(function()
			local w = math.ceil(caption.TextBounds.X)
			caption.Size = Offset(w + 1, Metrics.Row)
			right.Position = Offset(16 + w + 6, 7)
			right.Size = Dim2(1, -(16 + w + 6), 0, 1)
		end)
	else
		new("Frame", { Bg = "Element", Position = Offset(1, 1), Size = Dim2(1, -2, 1, -2), Radius = FULL, Border = "Outline" }, item.Frame)
	end
	return plain(self, { Item = item }, item, o)
end

function Section:Spacer(o)
	o = o or {}
	local item = self:_item(o.Height or o.Size or 8, "Spacer", nil, false)
	local object = { Item = item }
	function object:SetHeight(h) resizeItem(item, h) end
	return plain(self, object, item, o)
end

-- Button / ButtonRow --------------------------------------------------------

local function buttonPalette(style)
	style = string.lower(tostring(style or "primary"))
	local tone = Tones[style]
	local P = {
		Bg = function() return Current.Element end, Hover = function() return Current.HoveredElement end,
		Press = function() return Current.Inline end, Fg = function() return Current.Text end,
		Stroke = function() return Current.Outline end, Gradient = true,
	}
	if tone then
		P.Bg = function() return Current.Element:Lerp(tone, 0.2) end
		P.Hover = function() return Current.Element:Lerp(tone, 0.34) end
		P.Press = function() return Current.Element:Lerp(tone, 0.12) end
		P.Fg, P.Stroke = function() return tone end, function() return tone:Lerp(Current.Element, 0.45) end
	elseif style == "soft" then
		P.Bg = function() return Current.Element:Lerp(Current.Accent, 0.2) end
		P.Hover = function() return Current.Element:Lerp(Current.Accent, 0.34) end
		P.Press = function() return Current.Element:Lerp(Current.Accent, 0.12) end
		P.Fg, P.Stroke = function() return Current.Accent end, function() return Current.Element:Lerp(Current.Accent, 0.35) end
	elseif style == "outline" then
		P.Bg, P.Hover = function() return Current.Inline end, function() return Current.Element end
		P.Fg, P.Stroke, P.Gradient = function() return Current.Accent end, function() return Current.Accent end, false
	elseif style == "neon" then
		P.Bg = function() return Current.Inline:Lerp(Current.Accent, 0.12) end
		P.Hover = function() return Current.Inline:Lerp(Current.Accent, 0.28) end
		P.Fg, P.Stroke = function() return Current.Accent end, function() return Current.Accent end
	elseif style == "tertiary" then
		P.Bg, P.Hover, P.Press = function() return Current.Inline end, function() return Current.Element end, function() return Current.Tab end
		P.Fg, P.Stroke, P.Gradient = function() return Current.InactiveText end, function() return Current.Inline end, false
	elseif style == "link" then
		P.Bg, P.Hover, P.Press = function() return Current.Inline end, function() return Current.Inline end, function() return Current.Inline end
		P.Fg, P.Stroke, P.Gradient, P.Clear = function() return Current.Accent end, function() return Current.Inline end, false, true
	elseif style == "gradient" or style == "solid" or style == "accent" or style == "light" then
		P.Bg = function() return Current.Accent end
		P.Hover = function() return Current.Accent:Lerp(Color3.new(0, 0, 0), 0.08) end
		P.Press = function() return Current.Accent:Lerp(Color3.new(0, 0, 0), 0.2) end
		P.Fg, P.Stroke, P.Gradient = function() return Current.Background end, function() return Current.Accent end, false
	end
	return P
end

-- builds one button body into parent; the caller owns the item
local function buildButton(section, parent, position, size, o, widthFn)
	local win = section.Window
	local palette = buttonPalette(o.Style or o.Variant)
	local state = { Hover = false, Press = false, Err = false }
	local height = size.Y.Offset
	local button = new("TextButton", { Position = position, Size = size, Radius = Metrics.Radius, BackgroundTransparency = palette.Clear and 1 or 0 }, parent)
	local stroke = new("UIStroke", { ApplyStrokeMode = Enum.ApplyStrokeMode.Border, Thickness = 1 }, button)
	bind(button, "BackgroundColor3", function()
		if state.Press then return palette.Press() end
		if state.Hover then return palette.Hover() end
		return palette.Bg()
	end)
	bind(stroke, "Color", function()
		if state.Err then return Tones.destructive end
		return palette.Stroke()
	end)
	if palette.Gradient then addGradient(button) end
	local object = { Instance = button, Callback = o.Callback, Locked = false, Loading = false, Name = nameOf(o) }

	local fill
	local holdTime = tonumber(o.Hold)
	if holdTime and holdTime > 0 then
		fill = new("Frame", { Bg = "Accent", BackgroundTransparency = 0.55, Size = Dim2(0, 0, 1, 0), Radius = Metrics.Radius }, button)
	end
	local text, setWidth = centeredLabel(button, widthFn(), {
		Text = nameOf(o) or "Button", TextSize = 13, RowH = height,
		Fg = function() return state.Err and Tones.destructive or palette.Fg() end,
	})
	table.insert(win.ResizeJobs, function() setWidth(widthFn()) end)

	-- spinner: an arc made from a stroke with a transparency gradient, turned every frame
	local spinner = new("Frame", { Size = Offset(12, 12), AnchorPoint = Vec2(0.5, 0.5), Position = Dim2(0.5, 0, 0.5, 0), Radius = FULL, Visible = false }, button)
	local spinStroke = new("UIStroke", { Thickness = 2, ApplyStrokeMode = Enum.ApplyStrokeMode.Border }, spinner)
	bind(spinStroke, "Color", function() return palette.Fg() end)
	local arc = new("UIGradient", {
		Transparency = NumberSequence.new({ NumberSequenceKeypoint.new(0, 0), NumberSequenceKeypoint.new(0.6, 0), NumberSequenceKeypoint.new(0.61, 1), NumberSequenceKeypoint.new(1, 1) }),
	}, spinStroke)
	local function spin(dt) arc.Rotation = (arc.Rotation + dt * 540) % 360 end

	local badge, badgeText
	local function paint()
		refresh(button, "Fast"); refresh(text, "Fast"); refresh(stroke, "Fast")
	end
	function object:SetLoading(on)
		on = on == true
		if on == self.Loading then return self end
		self.Loading = on
		spinner.Visible = on
		text.TextTransparency = on and 1 or (self.Locked and 0.55 or 0)
		if on then Jobs[spin] = true else Jobs[spin] = nil end
		return self
	end
	function object:IsLoading() return self.Loading end
	function object:SetLocked(locked)
		self.Locked = locked == true
		text.TextTransparency = (self.Loading and 1) or (self.Locked and 0.55 or 0)
		return self
	end
	function object:SetBadge(value)
		if value == nil or value == 0 or value == "" or value == false then
			if badge then badge.Visible = false end
			return self
		end
		if not badge then
			badge = new("Frame", { Bg = function() return Tones.destructive end, Size = Offset(14, 12), Radius = FULL, ZIndex = 4, AnchorPoint = Vec2(1, 0), Position = Dim2(1, -3, 0, 2) }, button)
			badgeText = label(badge, { Text = "", TextSize = 9, Fg = function() return Color3.new(1, 1, 1) end, RowH = 12, W = 14, Weight = Bold })
			badgeText.TextXAlignment = Enum.TextXAlignment.Center
			badgeText:GetPropertyChangedSignal("TextBounds"):Connect(function()
				local w = math.max(14, math.ceil(badgeText.TextBounds.X) + 6)
				badge.Size = Offset(w, 12)
				badgeText.Size = Offset(w, 12)
			end)
		end
		badge.Visible = true
		setText(badgeText, tostring(value))
		return self
	end
	function object:SetTitle(value)
		setText(text, value)
		return self
	end
	function object:SetText(value) return self:SetTitle(value) end
	if o.Badge ~= nil then object:SetBadge(o.Badge) end

	local function flashError()
		state.Err = true
		refresh(button); refresh(text); refresh(stroke)
		task.delay(0.6, function()
			state.Err = false
			if button.Parent then refresh(button); refresh(text); refresh(stroke) end
		end)
	end
	local function run(...)
		if object.Locked or object.Loading or not object.Callback then return end
		playSound("Click", o.Sound)
		local finished = false
		local thread = coroutine.create(function(...)
			local ok, err = pcall(object.Callback, ...)
			finished = true
			object:SetLoading(false)
			if not ok then
				warn("[Blossom] button callback failed: " .. tostring(err))
				flashError()
			end
		end)
		task.spawn(thread, ...)
		-- a callback that is still running after its first yield puts the button into its loading state
		if not finished then object:SetLoading(true) end
	end
	function object:Press() run() end
	function object:Run() run() end

	hoverable(button, function() state.Hover = true; paint() end, function()
		state.Hover, state.Press = false, false
		paint()
	end)
	button.MouseButton1Down:Connect(function() state.Press = true; refresh(button) end)
	button.MouseButton1Up:Connect(function() state.Press = false; refresh(button) end)

	if fill then
		local held, elapsed = false, 0
		local function hold(dt)
			elapsed = elapsed + dt
			fill.Size = Dim2(math.min(elapsed / holdTime, 1), 0, 1, 0)
			if elapsed >= holdTime then
				held = false
				Jobs[hold] = nil
				fill.Size = Dim2(0, 0, 1, 0)
				run()
			end
		end
		local function cancel()
			if not held then return end
			held, elapsed = false, 0
			Jobs[hold] = nil
			animate(fill, "Size", Dim2(0, 0, 1, 0), "Fast")
		end
		button.MouseButton1Down:Connect(function()
			if object.Locked then return end
			held, elapsed = true, 0
			snap(fill, "Size", Dim2(0, 0, 1, 0))
			Jobs[hold] = true
		end)
		button.MouseButton1Up:Connect(cancel)
		button.MouseLeave:Connect(cancel)
	else
		button.MouseButton1Click:Connect(function() run() end)
	end

	if o.Keybind ~= nil then
		local bindEntry = { Key = o.Keybind, Mode = "Toggle", Name = nameOf(o), OnPress = function() run() end }
		table.insert(win.Binds, bindEntry)
		object.BindEntry = bindEntry
		function object:SetKey(key) bindEntry.Key = key end
	end
	if o.Loading then object:SetLoading(true) end
	if o.Locked then object:SetLocked(true) end
	return object
end

function Section:Button(o)
	local name = nameOf(o) or "Button"
	local item = self:_item(Metrics.ButtonH, "Button", name)
	local object = buildButton(self, item.Frame, Offset(1, 1), Dim2(1, -2, 0, Metrics.ButtonH - 2), o, function() return width(self) - 2 end)
	object.Item = item
	return plain(self, object, item, o)
end

function Section:ButtonRow(o)
	local list = o.Buttons or o.Items or {}
	local count = math.max(#list, 1)
	local item = self:_item(Metrics.ButtonH, "Buttons", nameOf(o), false)
	local gap = Metrics.Gap
	local object = { Item = item, Buttons = {} }
	for index, def in ipairs(list) do
		if type(def) == "string" then def = { Name = def } end
		def.Style = def.Style or o.Style
		local button = buildButton(self, item.Frame,
			Dim2((index - 1) / count, index == 1 and 1 or math.floor(gap * (index - 1) / count) + 1, 0, 1),
			Dim2(1 / count, -2 - math.floor(gap * (count - 1) / count), 0, Metrics.ButtonH - 2),
			def, function() return (width(self) - gap * (count - 1)) / count - 2 end)
		table.insert(object.Buttons, button)
	end
	function object:Press(index) if self.Buttons[index] then self.Buttons[index]:Press() end end
	function object:SetButtonTitle(index, value) if self.Buttons[index] then self.Buttons[index]:SetTitle(value) end end
	function object:SetButtonLoading(index, on) if self.Buttons[index] then self.Buttons[index]:SetLoading(on) end end
	function object:SetButtonBadge(index, value) if self.Buttons[index] then self.Buttons[index]:SetBadge(value) end end
	function object:SetButtonLocked(index, on) if self.Buttons[index] then self.Buttons[index]:SetLocked(on) end end
	return plain(self, object, item, o)
end

-- Textbox / Input -----------------------------------------------------------

function Section:Textbox(o)
	local multiline = o.Multiline == true
	local height = multiline and (o.Height or 72) or Metrics.InputH
	local name = nameOf(o)
	local item = self:_item(height, "Textbox", name or o.Placeholder)
	local frame = new("Frame", { Bg = "Element", Position = Offset(1, 1), Size = Dim2(1, -2, 0, height - 2), Radius = Metrics.Radius, Border = "Outline" }, item.Frame)
	addGradient(frame)
	local x = 7
	if o.Icon or o.InputIcon then
		makeIcon(frame, o.Icon or o.InputIcon, 14, "InactiveText", "Element", { Position = Offset(7, floor((Metrics.InputH - 16) / 2)) })
		x = 28
	end
	local start = o.Default
	if start == nil then start = o.Value end
	local box = new("TextBox", {
		Fg = "Text", PlaceholderText = o.Placeholder or name or "", Text = start ~= nil and tostring(start) or "", TextSize = 13,
		Position = Offset(x, multiline and 6 or capOffset(height - 2, 13)), Size = Dim2(1, -(x + 7), 0, multiline and height - 14 or 18),
		ClearTextOnFocus = o.ClearOnFocus == true or o.ClearTextOnFocus == true, MultiLine = multiline, TextWrapped = multiline,
	}, frame)
	bind(box, "PlaceholderColor3", "InactiveText")
	local object = { Item = item, Instance = box, Callback = o.Callback }
	local function coerce(value)
		if value == nil then value = "" end
		if o.Numeric then
			local number = tonumber(value)
			if number == nil then number = tonumber(object.Value) or o.Min or 0 end
			if o.Min then number = math.max(number, o.Min) end
			if o.Max then number = math.min(number, o.Max) end
			return number
		end
		return tostring(value)
	end
	object.Value = coerce(start)
	function object:Set(value, silent)
		value = coerce(value)
		box.Text = tostring(value)
		object.Value = value
		if not silent then fire(object.Callback, value) end
	end
	function object:Submit() object:Set(box.Text) end
	function object:Focus() box:CaptureFocus() end
	local stroke = frame:FindFirstChildOfClass("UIStroke")
	local focused = false
	bind(stroke, "Color", function() return focused and Current.Accent or Current.Outline end)
	box.Focused:Connect(function() focused = true; refresh(stroke) end)
	box.FocusLost:Connect(function()
		focused = false
		refresh(stroke)
		object:Set(box.Text)
	end)
	if o.Numeric then box.Text = tostring(object.Value) end
	register(self, object, o, item)
	return object
end

-- Paragraph / Callout / Banner / Hero / Marquee ----------------------------

local function wrapped(parent, text, y, size, fg)
	return new("TextLabel", {
		Fg = fg or "InactiveText", Text = text or "", TextSize = size or 12, TextWrapped = true,
		Position = Offset(0, y), Size = Dim2(1, 0, 0, 14),
	}, parent)
end

function Section:Paragraph(o)
	local title = o.Title or o.Name
	local item = self:_item(Metrics.Row, "Paragraph", title, false)
	local y = 0
	if title then
		label(item.Frame, { Text = title, Weight = Semi })
		y = 20
	end
	local body = wrapped(item.Frame, o.Content or o.Text or "", y, 12)
	local function fit()
		local h = math.max(14, math.ceil(body.TextBounds.Y))
		body.Size = Dim2(1, 0, 0, h)
		resizeItem(item, y + h + 2)
	end
	body:GetPropertyChangedSignal("TextBounds"):Connect(fit)
	fit()
	local object = { Item = item, Value = o.Content or o.Text or "" }
	function object:Set(text)
		self.Value = tostring(text)
		setText(body, self.Value)
	end
	function object:SetContent(text) self:Set(text) end
	return plain(self, object, item, o)
end

local CalloutTypes = {
	info = { "info", function() return Current.Accent end }, success = { "check", function() return Tones.success end },
	warning = { "warning", function() return Tones.warning end }, error = { "warning", function() return Tones.destructive end },
}

function Section:Callout(o)
	local title = o.Title or o.Name
	local item = self:_item(34, "Callout", title, false)
	local spec = CalloutTypes[string.lower(tostring(o.Type or "info"))] or CalloutTypes.info
	local tone = spec[2]
	local frame = new("Frame", { Bg = function() return Current.Inline:Lerp(tone(), 0.12) end, Size = Dim2(1, 0, 1, 0), Radius = Metrics.CardRadius }, item.Frame)
	local stroke = new("UIStroke", { ApplyStrokeMode = Enum.ApplyStrokeMode.Border, Thickness = 1 }, frame)
	bind(stroke, "Color", function() return Current.Outline:Lerp(tone(), 0.4) end)
	local icon = makeIcon(frame, spec[1], 14, tone, "Inline", { Position = Offset(9, 9) })
	local y = 8
	if title then
		label(frame, { Text = title, Weight = Semi, X = 30, Y = y - 4, Right = 8, Truncate = AtEnd })
		y = y + 16
	end
	local body = new("TextLabel", { Fg = "InactiveText", Text = o.Content or o.Text or o.Description or "", TextSize = 12, TextWrapped = true, Position = Offset(30, y - 3), Size = Dim2(1, -38, 0, 14) }, frame)
	local function fit()
		local h = math.max(14, math.ceil(body.TextBounds.Y))
		body.Size = Dim2(1, -38, 0, h)
		resizeItem(item, math.max(34, y + h + 8))
	end
	body:GetPropertyChangedSignal("TextBounds"):Connect(fit)
	fit()
	local object = { Item = item, Value = o.Content or o.Text or "" }
	function object:Set(text) self.Value = tostring(text); setText(body, self.Value) end
	function object:SetText(text) self:Set(text) end
	function object:SetType(kind)
		local next = CalloutTypes[string.lower(tostring(kind))]
		if not next then return end
		spec, tone = next, next[2]
		refresh(frame); refresh(stroke)
		icon.Update()
	end
	return plain(self, object, item, o)
end

function Section:Marquee(o)
	local item = self:_item(22, "Marquee", nameOf(o), false)
	local frame = new("Frame", { Bg = "Element", Size = Dim2(1, 0, 1, 0), Radius = Metrics.Radius, Border = "Outline", ClipsDescendants = true }, item.Frame)
	local text = label(frame, { Text = o.Text or nameOf(o) or "", TextSize = 12, RowH = 20, W = 10, Fg = "InactiveText" })
	local object = { Item = item, Value = o.Text or nameOf(o) or "" }
	local offset = 0
	local function step(dt)
		local w = math.ceil(text.TextBounds.X)
		local view = frame.AbsoluteSize.X
		text.Size = Offset(w + 1, 20)
		if w <= view - 16 then
			text.Position = Offset(8, text.Position.Y.Offset)
			return
		end
		offset = offset + dt * 28
		if offset > w + 24 then offset = 0 end
		text.Position = Offset(floor(8 - offset), text.Position.Y.Offset)
	end
	Jobs[step] = true
	object.OnDestroy = function() Jobs[step] = nil end
	function object:Set(value) self.Value = tostring(value); offset = 0; setText(text, self.Value) end
	function object:SetText(value) self:Set(value) end
	return plain(self, object, item, o)
end

local function inlineRow(section, kind, name, controlW, rowH)
	rowH = rowH or Metrics.InputH
	local item = section:_item(rowH, kind, name)
	local caption = label(item.Frame, { Text = name or "", RowH = rowH, Right = controlW + Metrics.Gap, Truncate = AtEnd })
	item.Label = caption
	local host = new("Frame", { AnchorPoint = Vec2(1, 0), Position = Dim2(1, -1, 0, 1), Size = Offset(controlW, rowH - 2) }, item.Frame)

end

local function smallButton(parent, position, size, icon, rotation)
	local button = new("TextButton", { Position = position, Size = size, Radius = Metrics.Radius, Border = "Outline", BackgroundTransparency = 0 }, parent)
	interactive(button, "Element", "HoveredElement", "Inline")
	addGradient(button)
	if icon then
		local mark = makeIcon(button, icon, 10, "Text", "Element", { Position = Offset(floor((size.X.Offset - 12) / 2), floor((size.Y.Offset - 12) / 2)) })
		if rotation then mark.Root.Rotation = rotation end
	end
	return button
end

local function flowLayout(frames, widths, maxW, rowH, gap, top, stretch)
	local rows, row, used = {}, {}, 0
	for index = 1, #frames do
		local w = math.min(widths[index], maxW)
		if #row > 0 and used + gap + w > maxW then
			table.insert(rows, row)
			row, used = {}, 0
		end
		if #row > 0 then used = used + gap end
		table.insert(row, index)
		used = used + w
	end
	if #row > 0 then table.insert(rows, row) end
	for r, indices in ipairs(rows) do
		local total = 0
		for _, index in ipairs(indices) do total = total + math.min(widths[index], maxW) end
		local free = maxW - gap * (#indices - 1) - total
		local x = 0
		for _, index in ipairs(indices) do
			local w = math.min(widths[index], maxW)
			if stretch then w = w + floor(free / #indices) end
			frames[index].Position = Offset(x, top + (r - 1) * (rowH + gap))
			frames[index].Size = Offset(w, rowH)
			x = x + w + gap
		end
	end
	return top + #rows * rowH + math.max(0, #rows - 1) * gap
end

local function estimate(text) return #tostring(text) * 6.6 + 18 end

local function listOf(value)
	local out = {}
	if type(value) ~= "table" then return out end
	if #value > 0 then
		for _, entry in ipairs(value) do out[entry] = true end
	else
		for key, on in pairs(value) do if on then out[key] = true end end
	end
	return out
end

local function setList(map, options)
	local out = {}
	for _, option in ipairs(options) do
		if map[option] then table.insert(out, option) end
	end
	return out
end

local function titleRow(item, name)
	if not name then return 0 end
	local caption = label(item.Frame, { Text = name, Truncate = AtEnd })
	item.Label = caption
	return Metrics.Row + 4
end

-- Segmented / Chips / MultiSelect --------------------------------------------

function Section:Segmented(o)
	local multi = o.Multi == true
	local name = nameOf(o)
	local item = self:_item(Metrics.Row, "Segmented", name)
	local top = titleRow(item, name)
	local object = { Options = o.Options or o.Values or {}, Multi = multi, Value = multi and {} or nil, Callback = o.Callback }
	local chips = {}
	local function chosen(option)
		if multi then return object.Value[option] == true end
		return object.Value == option
	end
	local function layout()
		if not item.Frame.Parent then return end
		local frames, widths = {}, {}
		for _, chip in ipairs(chips) do table.insert(frames, chip.Button); table.insert(widths, chip.Width) end
		local bottom = flowLayout(frames, widths, width(self), 22, 4, top, true)
		resizeItem(item, math.max(bottom, Metrics.Row))
	end
	local function build()
		for _, chip in ipairs(chips) do chip.Button:Destroy() end
		chips = {}
		for _, option in ipairs(object.Options) do
			local button = new("TextButton", {
				Bg = function() return chosen(option) and Current.Accent or Current.Element end,
				Radius = Metrics.Radius, BackgroundTransparency = 0,
			}, item.Frame)
			local stroke = new("UIStroke", { ApplyStrokeMode = Enum.ApplyStrokeMode.Border, Thickness = 1 }, button)
			bind(stroke, "Color", function() return chosen(option) and Current.Accent or Current.Outline end)
			local text = centeredLabel(button, 40, { Text = tostring(option), TextSize = 12, RowH = 22, Fg = function() return chosen(option) and Current.Background or Current.InactiveText end })
			button:GetPropertyChangedSignal("AbsoluteSize"):Connect(function()
				text.Size = Offset(math.max(10, button.AbsoluteSize.X - 8), 22)
				text.Position = Offset(4, text.Position.Y.Offset)
			end)
			button.MouseButton1Click:Connect(function()
				if multi then
					local nextValue = copyValue(object.Value)
					nextValue[option] = not nextValue[option] or nil
					object:Set(nextValue)
				else
					object:Set(option)
				end
			end)
			table.insert(chips, { Button = button, Stroke = stroke, Text = text, Width = estimate(option), Option = option })
		end
		layout()
	end
	local function paint()
		for _, chip in ipairs(chips) do
			refresh(chip.Button); refresh(chip.Stroke); refresh(chip.Text)
		end
	end
	function object:Set(value, silent)
		if multi then
			self.Value = listOf(value)
		else
			self.Value = value
		end
		paint()
		if not silent then fire(self.Callback, multi and self.Value or value) end
	end
	function object:SetOptions(list) self.Options = list; build() end
	function object:IsSelected(option) return chosen(option) end
	function object:Toggle(option)
		if multi then
			local nextValue = copyValue(self.Value)
			nextValue[option] = not nextValue[option] or nil
			self:Set(nextValue)
		else
			self:Set(option)
		end
	end
	function object:SelectAll() if multi then self:Set(listOf(self.Options)) end end
	function object:Clear() self:Set(multi and {} or nil) end
	function object:GetValues() return multi and setList(self.Value, self.Options) or { self.Value } end
	table.insert(self.Window.ResizeJobs, layout)
	build()

	end

		place(fill, "Size", Dim2(0, pixels, 1, 0), animated, "Follow")

		if changed and not silent then fire(self.Callback, self.Value) end

		if not isClick(input) then return end

		beginDrag(fromMouse)
	end)

function Section:Vector3(o)

	local item = self:_item(Metrics.Row, "Vector", name)
	local top = titleRow(item, name)
	resizeItem(item, top + 24)
	local object = { Callback = o.Callback, Value = {} }
	local boxes = {}
	for index = 1, dims do
		local cell = new("Frame", { Bg = "Element", Position = Dim2((index - 1) / dims, 0, 0, top), Size = Dim2(1 / dims, -3, 0, 24), Radius = Metrics.Radius, Border = "Outline" }, item.Frame)
		label(cell, { Fg = "InactiveText", Text = axes[index], TextSize = 10, X = 6, RowH = 22, W = 10 })
		boxes[index] = new("TextBox", { Fg = "Text", TextSize = 12, Position = Offset(17, capOffset(22, 12)), Size = Dim2(1, -20, 0, 16), ClearTextOnFocus = false, Text = "0" }, cell)
		boxes[index].FocusLost:Connect(function()
			local value = copyValue(object.Value)
			value[index] = tonumber(boxes[index].Text) or value[index] or 0
			object:Set(value)
		end)
	end
	function object:GetVector()
		local v = self.Value
		if dims == 2 then return Vector2.new(v[1], v[2]) end
		if dims == 3 then return Vector3.new(v[1], v[2], v[3]) end
		return v
	end
	function object:Set(value, silent)
		local list = {}
		if typeof(value) == "Vector3" then list = { value.X, value.Y, value.Z }
		elseif typeof(value) == "Vector2" then list = { value.X, value.Y }
		elseif type(value) == "table" then list = { value[1] or value.X, value[2] or value.Y, value[3] or value.Z, value[4] or value.W } end
		for index = 1, dims do
			list[index] = tonumber(list[index]) or 0
			boxes[index].Text = tostring(floor(list[index] * 1000 + 0.5) / 1000)
		end
		self.Value = { table.unpack(list, 1, dims) }
		if not silent then fire(self.Callback, self.Value, self:GetVector()) end
	end
	object:Set(o.Default or o.Value or {}, true)
	register(self, object, o, item)
	return object
end

function Section:Pad(o)
	local name = nameOf(o)
	local height = o.Height or 96
	local item = self:_item(20 + height, "Pad", name)
	local caption = label(item.Frame, { Text = name or "", Truncate = AtEnd })
	item.Label = caption

	chip.Position = Dim2(1, -1, 0, 0)
	fitChip(chip, chipText, 24, function(w) caption.Size = Dim2(1, -(w + Metrics.Gap), 0, Metrics.Row) end)

	addGradient(area)
	new("Frame", { Bg = "Outline", Position = Dim2(0.5, 0, 0, 0), Size = Dim2(0, 1, 1, 0) }, area)
	new("Frame", { Bg = "Outline", Position = Dim2(0, 0, 0.5, 0), Size = Dim2(1, 0, 0, 1) }, area)
	local marker = new("Frame", { Bg = "Accent", AnchorPoint = Vec2(0.5, 0.5), Size = Offset(10, 10), Radius = FULL, ZIndex = 3 }, area)
	local hit = new("TextButton", { Size = Dim2(1, 0, 1, 0), ZIndex = 4 }, area)
	local minX, maxX, minY, maxY = o.MinX or o.Min or 0, o.MaxX or o.Max or 100, o.MinY or o.Min or 0, o.MaxY or o.Max or 100
	local step = o.Step or 1
	local decimals = step % 1 ~= 0 and 2 or 0
	local object = { Callback = o.Callback, Value = { minX, minY } }
	function object:Set(value, silent)
		local x, y
		if typeof(value) == "Vector2" then x, y = value.X, value.Y
		elseif type(value) == "table" then x, y = value[1] or value.X, value[2] or value.Y end
		x = math.clamp(floor((tonumber(x) or minX) / step + 0.5) * step, minX, maxX)
		y = math.clamp(floor((tonumber(y) or minY) / step + 0.5) * step, minY, maxY)
		self.Value = { x, y }
		local ax = maxX == minX and 0 or (x - minX) / (maxX - minX)
		local ay = maxY == minY and 0 or (y - minY) / (maxY - minY)
		animate(marker, "Position", Dim2(ax, 0, 1 - ay, 0), "Follow")
		setText(chipText, formatNumber(x, decimals) .. ", " .. formatNumber(y, decimals))
		if not silent then fire(self.Callback, self.Value, Vector2.new(x, y)) end
	end
	local function fromMouse(point)
		local p, s = area.AbsolutePosition, area.AbsoluteSize
		local ax, ay = clamp01((point.X - p.X) / s.X), clamp01((point.Y - p.Y) / s.Y)
		object:Set({ minX + (maxX - minX) * ax, minY + (maxY - minY) * (1 - ay) })
	end
	hit.InputBegan:Connect(function(input)
		if not isClick(input) then return end
		fromMouse(mousePos())
		beginDrag(fromMouse)
	end)
	object:Set(o.Default or o.Value or { minX, minY }, true)
	register(self, object, o, item)
	return object
end

function Section:CooldownButton(o)
	local cooldown = o.Cooldown or o.Duration or 5
	local title = nameOf(o) or "Button"
	local user = o.Callback
	local untilTime = 0
	local button
	local function remaining() return math.max(0, untilTime - os.clock()) end
	local function step()
		local left = remaining()
		if left > 0 then
			button:SetTitle(title .. " (" .. string.format("%.1f", left) .. "s)")
		else
			button:SetTitle(title)
			button:SetLocked(false)
			Jobs[step] = nil
		end
	end
	o.Callback = function(...)
		if remaining() > 0 then return end
		untilTime = os.clock() + cooldown
		button:SetLocked(true)
		Jobs[step] = true
		if user then user(...) end
	end
	button = self:Button(o)
	function button:IsReady() return remaining() <= 0 end
	function button:Remaining() return remaining() end
	function button:Reset() untilTime = 0; step() end
	return button
end

		for _, option in ipairs(object.Options) do

					clear()
				end)
				table.insert(rows, row)
			end
		end

		if not silent then fire(self.Callback, self.Value) end
	end

		focused = false

	end)

	if start == nil then start = o.Value end
	if start ~= nil then object:Set(start, true) end
	register(self, object, o, item)
	return object

-- DateTime / ColorMixer ------------------------------------------------------------

local DateFields = {
	Date = { { "day", "D", 1, 31 }, { "month", "M", 1, 12 }, { "year", "Y", 1970, 2200 } },
	Time = { { "hour", "H", 0, 23 }, { "min", "M", 0, 59 }, { "sec", "S", 0, 59 } },
}

function Section:DateTime(o)
	local mode = o.Mode or "DateTime"
	local groups = {}
	if mode == "Date" or mode == "DateTime" then table.insert(groups, DateFields.Date) end
	if mode == "Time" or mode == "DateTime" then table.insert(groups, DateFields.Time) end
	local name = nameOf(o)
	local item = self:_item(Metrics.Row, "Date Time", name)
	local top = titleRow(item, name)
	resizeItem(item, top + #groups * 26 - 2)
	local object = { Callback = o.Callback, Mode = mode }
	local boxes = {}
	function object:GetTable() return os.date("*t", self.Value) end
	function object:Text()
		local t = self:GetTable()
		local date = string.format("%04d-%02d-%02d", t.year, t.month, t.day)
		local clock = string.format("%02d:%02d:%02d", t.hour, t.min, t.sec)
		return mode == "Date" and date or mode == "Time" and clock or date .. " " .. clock
	end
	function object:Set(value, silent)
		if type(value) == "table" then
			local t = self:GetTable()
			for key, entry in pairs(value) do t[key] = entry end
			value = os.time(t)
		end
		value = tonumber(value) or os.time()
		self.Value = value
		local t = os.date("*t", value)
		for key, box in pairs(boxes) do box.Text = string.format(key == "year" and "%04d" or "%02d", t[key]) end
		if not silent then fire(self.Callback, value, t) end
	end
	function object:Nudge(field, delta)
		local t = self:GetTable()
		t[field] = t[field] + delta
		self:Set(os.time(t))
	end
	for row, fields in ipairs(groups) do
		for index, field in ipairs(fields) do
			local key, letter, low, high = field[1], field[2], field[3], field[4]
			local cell = new("Frame", { Bg = "Element", Position = Dim2((index - 1) / 3, 0, 0, top + (row - 1) * 26), Size = Dim2(1 / 3, -3, 0, 24), Radius = Metrics.Radius, Border = "Outline" }, item.Frame)
			label(cell, { Fg = "InactiveText", Text = letter, TextSize = 10, X = 6, RowH = 22, W = 10 })
			local box = new("TextBox", { Fg = "Text", TextSize = 12, Position = Offset(16, capOffset(22, 12)), Size = Dim2(1, -18, 0, 16), ClearTextOnFocus = false }, cell)
			boxes[key] = box
			box.FocusLost:Connect(function()
				local number = tonumber(box.Text)
				if number then
					local t = object:GetTable()
					t[key] = math.clamp(floor(number), low, high)
					object:Set(os.time(t))
				else
					object:Set(object.Value, true)
				end
			end)
		end
	end
	object:Set(o.Default or o.Value or os.time(), true)
	register(self, object, o, item)
	return object
end

	sub = subSection(self, body, fit, function() return trackWidth(self.Window, self) end, self.Name .. "/" .. name)
	arrow.Root.Rotation = open and 0 or -90
	function object:SetOpened(value)
		self.Opened = value == true
		animate(arrow.Root, "Rotation", self.Opened and 0 or -90, "Fast")
		fit()
	end
	function object:ToggleOpened() self:SetOpened(not self.Opened) end
	header.MouseButton1Click:Connect(function() object:ToggleOpened() end)
	-- the group is a container: every element method works on it
	for key, value in pairs(Section) do
		if type(value) == "function" and object[key] == nil and key:sub(1, 1) ~= "_" and not NoCopy[key] then
			object[key] = function(_, ...) return value(sub, ...) end
		end
	end
	object.Content = sub
	plain(self, object, item, o)
	fit()
	return object
end

function Section:Loader(o)
	local item = self:_item(Metrics.Row + 4, "Loader", nameOf(o), false)

	local spinner = new("Frame", { Size = Offset(12, 12), Position = Offset(1, 4), Radius = FULL }, item.Frame)
	local stroke = new("UIStroke", { Thickness = 2, ApplyStrokeMode = Enum.ApplyStrokeMode.Border }, spinner)
	local mark = makeIcon(item.Frame, "check", 12, "Accent", "Inline", { Position = Offset(1, 4) })
	local arc = new("UIGradient", { Transparency = NumberSequence.new({ NumberSequenceKeypoint.new(0, 0), NumberSequenceKeypoint.new(0.6, 0), NumberSequenceKeypoint.new(0.61, 1), NumberSequenceKeypoint.new(1, 1) }) }, stroke)
	local text = label(item.Frame, { Text = nameOf(o) or "", X = 22, RowH = Metrics.Row + 4, Truncate = AtEnd, Fg = "InactiveText" })

	local function paint()
		local loading = object.State == "Loading"
		spinner.Visible = object.State == "Loading"
		mark.Root.Visible = not loading
		Jobs[spin] = loading and true or nil
		refresh(stroke)
		for _, part in ipairs(mark.Parts) do part.BackgroundColor3 = object.State == "Failed" and Tones.destructive or Tones.success end
	end

		self.State = state
		if message then setText(text, tostring(message)) end
		paint()
	end
	function object:Loading(message) self:SetStatus("Loading", message) end
	function object:Done(message) self:SetStatus("Done", message) end
	function object:Fail(message) self:SetStatus("Failed", message) end
	function object:SetText(message) setText(text, tostring(message)) end

function Section:Tree(o)
	local item = self:_item(Metrics.Row, "Tree", nameOf(o), true)
	local object = { Callback = o.Callback, Nodes = {}, Expanded = {}, Value = nil }
	local rows = {}
	local function build()
		for _, row in ipairs(rows) do row:Destroy() end
		rows = {}
		local y = 0
		local function walk(nodes, depth, path)
			for _, node in ipairs(nodes) do
				local def = type(node) == "table" and node or { Name = tostring(node) }
				local key = path .. "/" .. tostring(def.Name or def.Title)
				local children = def.Children or def.Nodes
				local row = new("TextButton", { Bg = "HoveredElement", BackgroundTransparency = 1, Position = Offset(0, y), Size = Dim2(1, 0, 0, 20), Radius = Metrics.SmallRadius }, item.Frame)
				if children then
					local arrow = makeIcon(row, "chevron", 10, "InactiveText", "Inline", { Position = Offset(depth * 12 + 2, 5) })
					arrow.Root.Rotation = object.Expanded[key] and 0 or -90
				end
				label(row, { Text = tostring(def.Name or def.Title), X = depth * 12 + 18, RowH = 20, Right = 4, Truncate = AtEnd, Fg = function() return object.Value == key and Current.Accent or Current.InactiveText end })
				row.MouseButton1Click:Connect(function()
					if children then
						object.Expanded[key] = not object.Expanded[key] or nil
						build()
					else
						object:Set(key)
					end
				end)
				table.insert(rows, row)
				y = y + 20
				if children and object.Expanded[key] then walk(children, depth + 1, key) end
			end
		end
		walk(object.Nodes, 0, "")
		resizeItem(item, math.max(Metrics.Row, y))
	end
	function object:Set(value, silent)
		self.Value = value
		for _, row in ipairs(rows) do refresh(row:FindFirstChildOfClass("TextLabel")) end
		if not silent then fire(self.Callback, value) end
	end
	function object:SetNodes(nodes) self.Nodes = nodes or {}; build() end
	function object:ExpandAll()
		local function mark(nodes, path)
			for _, node in ipairs(nodes) do
				local children = type(node) == "table" and (node.Children or node.Nodes)
				if children then
					local key = path .. "/" .. tostring(node.Name or node.Title)
					self.Expanded[key] = true
					mark(children, key)
				end
			end
		end
		mark(self.Nodes, "")
		build()
	end
	function object:CollapseAll() self.Expanded = {}; build() end
	function object:Expand(key) self.Expanded[key] = true; build() end
	function object:Collapse(key) self.Expanded[key] = nil; build() end
	object:SetNodes(o.Nodes or {})
	if o.Expanded then object:ExpandAll() end
	if o.Default ~= nil then object.Value = o.Default end
	o.NoSave = o.NoSave ~= false
	register(self, object, o, item)
	return object
end

function Section:Graph(o)
	local name = nameOf(o)

	local caption = label(item.Frame, { Text = name or "", Truncate = AtEnd })

	local object = { Item = item, Values = {}, Limit = o.Limit or 40 }
	local bars = {}
	local function build()
		for _, bar in ipairs(bars) do bar:Destroy() end
		bars = {}
		local values = object.Values
		local high = 0.0001
		for _, value in ipairs(values) do high = math.max(high, value) end
		high = o.Max or high

		for index, value in ipairs(values) do
			local fraction = clamp01(value / high)
			local bar = new("Frame", { Bg = "Accent", BackgroundTransparency = 0.25, AnchorPoint = Vec2(0, 1), Position = Dim2((index - 1) / count, 1, 1, -1), Size = Dim2(1 / count, -2, fraction, -2) }, area)

		end
	end
	function object:SetData(list) self.Values = copyValue(list or {}); build() end
	function object:Set(list) self:SetData(list) end
	function object:Push(value)
		table.insert(self.Values, tonumber(value) or 0)
		while #self.Values > self.Limit do table.remove(self.Values, 1) end
		build()
	end
	function object:Clear() self:SetData({}) end
	function object:Get() return self.Values end
	function object:Text(value) setText(caption, tostring(value)) end
	object:SetData(o.Points or o.Values or {})

function Section:Donut(o)
	local size = o.Size or 70
	local item = self:_item(size + 12, "Donut", nameOf(o), false)
	local object = { Item = item, Items = o.Items or {} }
	local host = new("Frame", { Position = Offset(4, 6), Size = Offset(size, size) }, item.Frame)
	local legend = new("Frame", { Position = Offset(size + 16, 6), Size = Dim2(1, -(size + 16), 0, size) }, item.Frame)
	local parts = {}
	local function build()
		for _, part in ipairs(parts) do part:Destroy() end
		parts = {}
		local total = 0
		for _, entry in ipairs(object.Items) do total = total + (entry.Value or 0) end
		total = math.max(total, 0.0001)
		local dots, radius, dot = 36, size / 2 - 4, 6
		local index = 0
		for entryIndex, entry in ipairs(object.Items) do
			local share = floor((entry.Value or 0) / total * dots + 0.5)
			local color = entry.Color and colorOf(entry.Color) or Current.Accent:Lerp(Current.Text, (entryIndex - 1) * 0.18)
			for _ = 1, share do
				index = index + 1
				if index > dots then break end
				local angle = math.rad(360 * (index - 1) / dots)
				table.insert(parts, new("Frame", { BackgroundColor3 = color, BackgroundTransparency = 0, Radius = FULL, Position = Offset(round(size / 2 + math.sin(angle) * radius - dot / 2), round(size / 2 - math.cos(angle) * radius - dot / 2)), Size = Offset(dot, dot) }, host))
			end
			local line = label(legend, { Text = tostring(entry.Name or entry.Title or "") .. "  " .. floor((entry.Value or 0) / total * 100 + 0.5) .. "%", TextSize = 11, X = 12, Y = (entryIndex - 1) * 16, RowH = 14, Fg = "InactiveText", Truncate = AtEnd })
			table.insert(parts, line)
			table.insert(parts, new("Frame", { BackgroundColor3 = color, BackgroundTransparency = 0, Radius = FULL, Position = Offset(0, (entryIndex - 1) * 16 + 4), Size = Offset(7, 7) }, legend))
		end
	end
	function object:Set(list) self.Items = copyValue(list or {}); build() end
	function object:SetValue(key, value)
		for _, entry in ipairs(self.Items) do
			if entry.Name == key or entry.Title == key then entry.Value = value end
		end
		build()
	end
	build()
	return plain(self, object, item, o)
end

function Section:Heatmap(o)
	local columns, rowsCount = o.Columns or 7, o.Rows or 5
	local item = self:_item(Metrics.Row, "Heatmap", nameOf(o), false)
	local object = { Item = item, Values = o.Values or {} }
	local cells = {}
	local function layout()
		if not item.Frame.Parent then return end
		local gap = 3
		local w = (width(self) - gap * (columns - 1)) / columns
		local h = floor(math.min(w, 18))
		for index, cell in ipairs(cells) do
			local column, row = (index - 1) % columns, floor((index - 1) / columns)
			cell.Position = Offset(floor(column * (w + gap)), floor(row * (h + gap)))
			cell.Size = Offset(math.max(2, floor(w)), floor(h))
		end
		resizeItem(item, floor(rowsCount * (h + gap) - gap))
	end

		for index, cell in ipairs(cells) do
			cell.BackgroundColor3 = Current.Element:Lerp(Current.Accent, clamp01(object.Values[index] or 0))
		end
	end
	for _ = 1, columns * rowsCount do
		table.insert(cells, new("Frame", { Radius = 2, BackgroundTransparency = 0 }, item.Frame))
	end
	function object:Set(values) self.Values = values or {}; paint() end
	function object:SetCell(index, value) self.Values[index] = value; paint() end
	table.insert(self.Window.ResizeJobs, layout)
	layout()
	paint()
	return plain(self, object, item, o)

function Section:Timer(o)
	local mode = o.Mode or "Stopwatch"
	local item = self:_item(Metrics.ButtonH, "Timer", nameOf(o))
	local duration = o.Duration or 60
	local object = { Item = item, Mode = mode, Running = false, Elapsed = 0, Callback = o.Callback }
	local display = new("Frame", { Bg = "Element", Position = Offset(1, 1), Size = Dim2(1, -76, 0, Metrics.ButtonH - 2), Radius = Metrics.Radius, Border = "Outline" }, item.Frame)
	addGradient(display)
	local text = label(display, { Text = "00:00.0", TextSize = 13, X = 8, RowH = Metrics.ButtonH - 2, Right = 6, Weight = Semi })
	local start = smallButton(item.Frame, Dim2(1, -72, 0, 1), Offset(34, Metrics.ButtonH - 2), "dot")
	local reset = smallButton(item.Frame, Dim2(1, -36, 0, 1), Offset(34, Metrics.ButtonH - 2), "reset")
	local startIcon = start:FindFirstChild("Icon")
	local function shown()
		local t = mode == "Countdown" and math.max(0, duration - object.Elapsed) or object.Elapsed
		return string.format("%02d:%04.1f", floor(t / 60), t % 60)
	end
	local last = os.clock()
	local function step()
		if not object.Running then return end
		local now = os.clock()
		object.Elapsed = object.Elapsed + (now - last)
		last = now
		setText(text, shown())
		if mode == "Countdown" and object.Elapsed >= duration then
			object:Pause()
			object.Elapsed = duration
			setText(text, shown())
			fire(object.Callback)
		end
	end
	function object:Start()
		if self.Running then return end
		self.Running, last = true, os.clock()
		Jobs[step] = true
		if startIcon then startIcon:Destroy(); startIcon = nil; makeIcon(start, "pause", 10, "Text", "Element", { Position = Offset(11, 8) }) end
	end
	function object:Pause()
		self.Running = false
		Jobs[step] = nil
		for _, child in ipairs(start:GetChildren()) do if child.Name == "Icon" then child:Destroy() end end
		makeIcon(start, "dot", 10, "Text", "Element", { Position = Offset(11, 8) })
	end
	function object:Toggle() if self.Running then self:Pause() else self:Start() end end
	function object:Reset() self:Pause(); self.Elapsed = 0; setText(text, shown()) end
	function object:Set(seconds) self.Elapsed = tonumber(seconds) or 0; setText(text, shown()) end
	function object:GetTime() return mode == "Countdown" and math.max(0, duration - self.Elapsed) or self.Elapsed end
	function object:IsRunning() return self.Running end
	object.OnDestroy = function() Jobs[step] = nil end
	start.MouseButton1Click:Connect(function() object:Toggle() end)
	reset.MouseButton1Click:Connect(function() object:Reset() end)
	setText(text, shown())
	if o.AutoStart then object:Start() end
	return plain(self, object, item, o)
end

function Section:Stopwatch(o) o.Mode = "Stopwatch"; return self:Timer(o) end
function Section:Countdown(o) o.Mode = "Countdown"; return self:Timer(o) end

function Section:Clock(o)
	local item = self:_item(26, "Clock", nameOf(o), false)
	local text = label(item.Frame, { Text = "", TextSize = 18, Weight = Bold, RowH = 26 })
	local object = { Item = item }
	local lastSecond = -1
	local function step()
		local now = os.time()
		if now == lastSecond then return end
		lastSecond = now
		setText(text, os.date("%H:%M:%S", now))
	end
	Jobs[step] = true
	object.OnDestroy = function() Jobs[step] = nil end
	function object:Text() return text.Text end
	step()
	return plain(self, object, item, o)
end

function Section:LiveStats(o)
	local entries = o.Items or {}
	local object = self:Stats({ Items = entries })
	local fetchers = {}
	for index, entry in ipairs(entries) do fetchers[index] = entry.Get or entry.Callback end
	local elapsed = 0
	local function step(dt)
		elapsed = elapsed + dt
		if elapsed < (o.Interval or 0.5) then return end
		elapsed = 0
		for index, entry in ipairs(entries) do
			local fn = fetchers[index]
			if fn then
				local ok, value = pcall(fn)
				if ok then object:SetStat(entry.Name or entry.Title or entry[1], value) end
			end
		end
	end
	Jobs[step] = true
	object.OnDestroy = function() Jobs[step] = nil end
	return object
end

					Headers = { ["Content-Type"] = "application/json", Origin = "https://discord.com" },

function Section:Viewport(o)
	local height = o.Height or 120
	local item = self:_item(height, "Viewport", nameOf(o), false)
	local frame = new("ViewportFrame", { BackgroundTransparency = 1, Size = Dim2(1, 0, 1, 0) }, item.Frame)
	local holder = new("Frame", { Bg = "Element", Size = Dim2(1, 0, 1, 0), Radius = Metrics.CardRadius, Border = "Outline", ClipsDescendants = true, ZIndex = 0 }, item.Frame)
	local camera = Instance.new("Camera")

	frame.CurrentCamera = camera
	local object = { Item = item, Interactive = o.Interactive ~= false, Yaw = 0.6, Pitch = 0.3, Distance = 8, Center = Vector3.zero }
	local function place_()
		local rotation = CFrame.Angles(0, object.Yaw, 0) * CFrame.Angles(object.Pitch, 0, 0)
		camera.CFrame = CFrame.new(object.Center) * rotation * CFrame.new(0, 0, object.Distance)
	end
	function object:Focus()
		local target = self.Target
		if not target then return end
		local cf, size
		if target:IsA("Model") then cf, size = target:GetBoundingBox() elseif target:IsA("BasePart") then cf, size = target.CFrame, target.Size end
		if not cf then return end
		self.Center = cf.Position
		self.Distance = math.max(size.Magnitude * 1.1, 2)
		place_()
	end
	function object:SetObject(instance)
		if self.Target then self.Target:Destroy() end
		self.Target = nil
		if instance then
			local clone = instance:Clone()
			clone.Parent = frame
			self.Target = clone
			self:Focus()
		end
	end
	function object:SetCamera(cameraInstance) if cameraInstance then camera.CFrame = cameraInstance.CFrame; camera.FieldOfView = cameraInstance.FieldOfView end end
	function object:SetInteractive(value) self.Interactive = value == true end
	function object:SetHeight(h) resizeItem(item, h) end

	hit.InputBegan:Connect(function(input)
		if not isClick(input) or not object.Interactive then return end
		local last = mousePos()
		beginDrag(function(point)
			local delta = point - last
			last = point
			object.Yaw = object.Yaw - delta.X * 0.01
			object.Pitch = math.clamp(object.Pitch - delta.Y * 0.01, -1.4, 1.4)
			place_()
		end)
	end)
	place_()
	object:SetObject(o.Object)
	if o.Camera then object:SetCamera(o.Camera) end
	return plain(self, object, item, o)
end

-- Dropdown: Source / SpecialType / Add / Remove / GetValues ----------------------------------

local TeamsService = game:GetService("Teams")
local baseDropdown = Section.Dropdown

local function optionList(o)
	local special = o.SpecialType
	if special == "Player" then
		local list = {}
		for _, player in ipairs(Players:GetPlayers()) do
			if not (o.ExcludeLocalPlayer and player == Players.LocalPlayer) then table.insert(list, player.Name) end
		end
		return list
	elseif special == "Team" then
		local list = {}
		for _, team in ipairs(TeamsService:GetTeams()) do table.insert(list, team.Name) end
		return list
	elseif special == "Tool" then
		local list = {}
		local player = Players.LocalPlayer
		local function scan(container)
			if not container then return end
			for _, child in ipairs(container:GetChildren()) do
				if child:IsA("Tool") and not table.find(list, child.Name) then table.insert(list, child.Name) end
			end
		end
		scan(player and player:FindFirstChildOfClass("Backpack"))
		scan(player and player.Character)
		return list
	elseif typeof(o.Source) == "Instance" then
		local list = {}
		for _, child in ipairs(o.Deep and o.Source:GetDescendants() or o.Source:GetChildren()) do
			if (not o.Class or child:IsA(o.Class)) and (not o.Filter or o.Filter(child)) and not table.find(list, child.Name) then
				table.insert(list, child.Name)
			end
		end
		return list
	end
	return nil
end

function Section:Dropdown(o)
	local generated = optionList(o)
	if generated then o.Options = generated end
	local dropdown = baseDropdown(self, o)
	local connections = {}
	local function reload()
		local list = optionList(o)
		if list then dropdown:SetOptions(list) end
	end
	if generated then
		local function listen(signal) table.insert(connections, signal:Connect(function() task.defer(reload) end)) end
		if o.SpecialType == "Player" then
			listen(Players.PlayerAdded); listen(Players.PlayerRemoving)
		elseif o.SpecialType == "Team" then
			listen(TeamsService.ChildAdded); listen(TeamsService.ChildRemoved)
		elseif o.SpecialType == "Tool" and Players.LocalPlayer then
			local backpack = Players.LocalPlayer:FindFirstChildOfClass("Backpack")
			if backpack then listen(backpack.ChildAdded); listen(backpack.ChildRemoved) end
		elseif typeof(o.Source) == "Instance" then
			listen(o.Deep and o.Source.DescendantAdded or o.Source.ChildAdded)
			listen(o.Deep and o.Source.DescendantRemoving or o.Source.ChildRemoved)
		end
	end
	function dropdown:Refresh() reload() end
	function dropdown:Add(option)
		if not table.find(self.Options, option) then
			local list = table.clone(self.Options)
			table.insert(list, option)
			self:SetOptions(list)
		end
	end
	function dropdown:Remove(option)
		local index = table.find(self.Options, option)
		if index then
			local list = table.clone(self.Options)
			table.remove(list, index)
			self:SetOptions(list)
		end
	end
	function dropdown:IsSelected(option)
		if self.Multi then return self.Value[option] == true end
		return self.Value == option
	end
	function dropdown:GetValues()
		if not self.Multi then return { self.Value } end
		return setList(self.Value, self.Options)
	end
	function dropdown:GetPlayers()
		local list = {}
		for _, name in ipairs(self:GetValues()) do
			local player = Players:FindFirstChild(tostring(name))
			if player then table.insert(list, player) end
		end
		return list
	end
	dropdown.OnDestroy = function()
		for _, connection in ipairs(connections) do connection:Disconnect() end
	end
	return dropdown
end

	if self.Column then self.Column:Layout(true) end

-- context menu: items are { Name, Icon, Callback, Locked }, anchor is a GuiObject or a point
function Window:Menu(items, anchor, options)
	local rowH, pad = Metrics.ListRowH, 4
	local menuW = options and options.Width or 150
	local frame, body, controls = self:_popupFrame(menuW, pad * 2 + #items * rowH)
	for index, entry in ipairs(items) do
		local row = new("TextButton", {
			Bg = "HoveredElement", BackgroundTransparency = 1, Position = Offset(pad, pad + (index - 1) * rowH),
			Size = Offset(menuW - pad * 2, rowH), Radius = Metrics.SmallRadius,
		}, body)
		NoFade[row] = true
		hoverable(row, function() animate(row, "BackgroundTransparency", 0, "Fast") end, function() animate(row, "BackgroundTransparency", 1, "Fast") end)
		if entry.Icon then makeIcon(row, entry.Icon, 12, "InactiveText", "Background", { Position = Offset(7, 6) }) end
		label(row, { Text = tostring(entry.Name or entry.Title or ""), X = entry.Icon and 26 or 8, RowH = rowH, Right = 6, Truncate = AtEnd, Fg = entry.Locked and "InactiveText" or "Text" })
		table.insert(controls, row)
		row.MouseButton1Click:Connect(function()
			if entry.Locked then return end
			self:ClosePopup()
			fire(entry.Callback, entry)
		end)
	end
	local popup = { Frame = frame, Fader = newFader(frame) }
	popup.OnClose = function() task.delay(0.5, function() frame:Destroy() end) end
	local at
	if typeof(anchor) == "Instance" then
		local p, s = anchor.AbsolutePosition, anchor.AbsoluteSize
		at = Vec2(p.X, p.Y + s.Y + Metrics.Gap)
	else
		at = anchor or mousePos()
	end
	if not self:_openPopup(popup, typeof(anchor) == "Instance" and anchor or frame, at) then frame:Destroy() end
	return popup
end

		end

	end
	body:GetPropertyChangedSignal("TextBounds"):Connect(fit)
	fit()
	backdrop.MouseButton1Click:Connect(function() if o.Dismissable ~= false then object:Close() end end)

-- floating button that opens and closes the menu
function Window:ScreenButton(o)
	o = o or {}
	local win = self
	local size = o.Size or 40
	local button = new("TextButton", { Bg = "Background", Position = o.Position or Offset(16, 160), Size = Offset(size, size), Radius = FULL, Border = "Outline", BackgroundTransparency = 0, ZIndex = 8 }, self.Screen)

	local object = { Button = button }
	button.InputBegan:Connect(function(input)
		if not isClick(input) then return end
		local start, origin = mousePos(), button.Position
		beginDrag(function(point)
			local delta = point - start
			if delta.Magnitude >= 4 then
				animate(button, "Position", win:_clamped(button, origin.X.Offset + delta.X, origin.Y.Offset + delta.Y), "Follow")
			end
		end, function(point)
			if (point - start).Magnitude < 4 then
				if o.Callback then fire(o.Callback) else win:Toggle() end
			end
		end)
	end)
	function object:SetVisible(value) button.Visible = value ~= false end
	function object:Destroy() button:Destroy() end
	return object
end

local baseNotify = Window.Notify
function Window:Notify(o)
	if type(o) == "string" then o = { Text = o } end
	o = o or {}
	local kind = string.lower(tostring(o.Type or "success"))
	if not o.Text then
		o.Text = (o.Title and o.Content) and (tostring(o.Title) .. ": " .. tostring(o.Content)) or tostring(o.Title or o.Content or "")
	end
	if not o.Icon then o.Icon = (kind == "warning" or kind == "error") and "warning" or (kind == "info" and "info" or "check") end
	if not o.IconColor then
		o.IconColor = kind == "warning" and Tones.warning or kind == "error" and Tones.destructive or kind == "info" and Current.Accent or nil
	end
	playSound(kind == "error" and "Alert" or "Notify", o.Sound)
	return baseNotify(self, o)
end

local baseSetVisible = Window.SetVisible
function Window:SetVisible(visible)
	local changed = self.Visible ~= visible
	local result = baseSetVisible(self, visible)
	if changed then playSound(visible and "Open" or "Close") end
	return result
end

-- sounds --------------------------------------------------------------------------------------------------

local SoundMap = {}
playSound = function(name, override)
	local spec = override
	if spec == nil then spec = SoundMap[name] end
	if not spec then return end
	local id, volume, speed = spec, 0.5, 1
	if type(spec) == "table" then id, volume, speed = spec.Id, spec.Volume or 0.5, spec.Speed or 1 end
	if not id then return end
	if type(id) == "number" then id = "rbxassetid://" .. id end
	local sound = Instance.new("Sound")
	sound.SoundId, sound.Volume, sound.PlaybackSpeed = tostring(id), volume, speed
	sound.Parent = game:GetService("SoundService")
	sound.Ended:Once(function() sound:Destroy() end)
	sound:Play()
	task.delay(12, function() if sound.Parent then sound:Destroy() end end)
end

function Blossom.SetSound(name, sound)
	if sound == nil then sound = false end
	SoundMap[name] = sound
end
function Blossom.GetSound(name) return SoundMap[name] end
function Blossom.PlaySound(name, sound) playSound(name, sound) end

-- accepts (def), (flag, def) or (text)
local function normalize(a, b)
	if type(a) == "string" then
		if type(b) == "table" then
			b.Flag = b.Flag or a
			b.Name = b.Name or b.Title
			return b
		end
		return { Name = a, Title = a, Text = a }
	end
	a = a or {}
	if a.Name == nil then a.Name = a.Title end
	return a
end

local UnloadListeners = {}

		warn("[Blossom] create a window before notifying")

function Blossom.IsCapturing()
	local win = currentWindow()
	return win ~= nil and win.Listening ~= nil
end
function Blossom.Toggle()
	for _, win in ipairs(Blossom.Windows) do win:Toggle() end
end
function Blossom.OnUnload(fn)
	table.insert(UnloadListeners, fn)
	return { Disconnect = function()
		local index = table.find(UnloadListeners, fn)
		if index then table.remove(UnloadListeners, index) end
	end }
end
function Blossom.Unload()
	for _, win in ipairs(table.clone(Blossom.Windows)) do pcall(win.Destroy, win) end
end
Blossom.Destroy = Blossom.Unload

local baseDestroy = Window.Destroy
function Window:Destroy()
	if self.Destroyed then return end
	self.Destroyed = true
	for flag in pairs(self.Flags) do Blossom.Flags[flag] = nil end
	local index = table.find(Blossom.Windows, self)
	if index then table.remove(Blossom.Windows, index) end
	baseDestroy(self)
	for _, fn in ipairs(table.clone(UnloadListeners)) do task.spawn(fn) end
end

-- Blossom.new / Blossom:CreateWindow(def): extra def keys Subtitle, Sounds, ToggleKey, Scale
local baseNew = Blossom.new
function Blossom.new(options)
	options = options or {}
	options.Title = options.Title or options.Name
	options.Build = options.Build or options.Subtitle
	options.MenuKey = options.MenuKey or options.ToggleKey
	local win = baseNew(options)
	table.insert(Blossom.Windows, win)
	if options.Sounds then
		for name, sound in pairs(options.Sounds) do SoundMap[name] = sound end
	end
	if options.Scale then win:SetScale(options.Scale) end
	local function tipJob(dt) win:_tipStep(dt) end
	Jobs[tipJob] = true
	table.insert(win.Jobs, tipJob)
	return win
end
function Blossom.CreateWindow(a, b)
	if a == Blossom then return Blossom.new(b) end
	return Blossom.new(a)
end
