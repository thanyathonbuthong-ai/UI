local function atan2(y, x)
	if math.atan2 then return math.atan2(y, x) end
		if x > 0 then return math.atan(y / x) end
	if x < 0 then
		if y >= 0 then return math.atan(y / x) + math.pi end
		return math.atan(y / x) - math.pi
	end
	if y > 0 then return math.pi / 2 end
	if y < 0 then return -math.pi / 2 end
	return 0
end

local function keyEquals(a, b)
	if a == b then return true end
	if a and b and a.EnumType and b.EnumType and a.Name and b.Name then
		return a.EnumType == b.EnumType and a.Name == b.Name
	end
	return false
end

local function deepCopy(t)
	local c = {}
	for k, v in pairs(t) do
		if type(v) == "table" then c[k] = deepCopy(v) else c[k] = v end
	end
	return c
end
local function deepMerge(dst, src)
	for k, v in pairs(src) do
		if dst[k] ~= nil and type(dst[k]) == "table" and type(v) == "table" then
			deepMerge(dst[k], v)
		else
			dst[k] = v
		end
	end
end

local function findVanillaGunScript()

	end

local function refreshRecord(rec)
	local plr = rec.player
	local tok = (rec.refreshToken or 0) + 1
	rec.refreshToken = tok
	rec.char = plr.Character
	rec.hum = rec.char and rec.char:FindFirstChildOfClass("Humanoid")
	rec.alive = rec.hum and rec.hum.Health > 0 or false
	rec.history = {}
	rec.lastHealth = rec.hum and rec.hum.Health or 100
	destroyRecordESP(rec)

	if rec.char and not rec.hum then
		local c = rec.char
		task.spawn(function()
			for _ = 1, 40 do
				task.wait(0.25)
				if plr.Character ~= c or rec.refreshToken ~= tok then return end
				local hum = c:FindFirstChildOfClass("Humanoid")
				if hum then
					if plr.Character == c then task.defer(refreshRecord, rec) end
					return
				end
			end
		end)
	end
	if rec.hum then
		connect(rec.hum:GetPropertyChangedSignal("Health"), function()
			local h = rec.hum and rec.hum.Health
			if not h then return end
			local old = rec.lastHealth
			rec.lastHealth = h
			if h <= 0 and old > 0 then
				local pend = pendingHits[rec]
				if pend and os.clock() - pend.t < 1.0 then
					logEvent("killed " .. plr.Name, COL.purple)
				else
					logEvent(plr.Name .. " died", COL.text_dim)
				end
				pendingHits[rec] = nil
			elseif h < old then
				local pend = pendingHits[rec]
				if pend and os.clock() - pend.t < 0.6 then
					local dmg = math.floor(old - h + 0.5)
					logEvent("hit " .. plr.Name .. " for " .. dmg .. " (" .. pend.hitbox .. ")", COL.green)
					doHitFeedback()
					pendingHits[rec] = nil
				end
			end
		end)
	end
end

local function playerIsValidTarget(rec)
		local plr = rec.player
	if plr == LocalPlayer then return false end
	if not (rec.char and rec.hum and rec.hum.Health > 0) then return false end
	if rec.char:FindFirstChildOfClass("ForceField") then return false end

		if plr.Team == LocalPlayer.Team then return false end
		local Guards = Teams:FindFirstChild("Guards")
		local Inmates = Teams:FindFirstChild("Inmates")
		if LocalPlayer.Team == Guards and plr.Team == Inmates
			and not rec.char:GetAttribute("Hostile") then
			return false
		end
	end
	if type(CFG.rage.team_targets) == "table" then
		local tn = plr.Team and plr.Team.Name
		if tn and CFG.rage.team_targets[tn] == false then
			return false
		end
	end
	return true
end

local HISTORY_SECONDS = 0.5
local function sampleHistory()
	local now = os.clock()
	for _, rec in pairs(records) do
		if rec.char and rec.hum and rec.hum.Health > 0 then
			local snap = { t = now, pos = {} }
			for _, hb in ipairs(hitboxParts(rec)) do
				snap.pos[hb.label] = hb.part.Position
			end
			table.insert(rec.history, snap)
			while #rec.history > 0 and (now - rec.history[1].t) > HISTORY_SECONDS do
				table.remove(rec.history, 1)
			end
						local rootPart = rec.char:FindFirstChild("HumanoidRootPart")
			if rootPart then
				local p = rootPart.Position
				if rec.lastPos then
					local dt = now - rec.lastPosT
					if dt > 0.016 then
						local v = (p - rec.lastPos) / dt
						rec.velocity = v
						rec.jitterSpeed = v.Magnitude
					end
				end
				rec.lastPos = p
				rec.lastPosT = now
			end
		end
	end
end

local function backtrackPosition(rec, label, currentPos)
	local ms = CFG.rage.backtrack_ms or 0
	if not (CFG.rage.backtrack and ms > 0) then return currentPos, false end
	local target = os.clock() - ms / 1000
	local best = nil
	for _, snap in ipairs(rec.history) do
		if snap.t <= target then best = snap end
	end
	if best and best.pos[label] then
		return best.pos[label], true
	end
	return currentPos, false
end

local function resolvePosition(rec, label, pos)
	local mode = CFG.rage.resolver
	if mode == "off" then return pos end
		if mode == "average" then
		local sum, n = nil, 0
		local cutoff = os.clock() - math.min(0.25, (CFG.rage.backtrack_ms or 0) / 1000 + 0.05)
		for i = #rec.history, 1, -1 do
			local snap = rec.history[i]
			if snap.t >= cutoff and snap.pos[label] then
				sum = sum and sum + snap.pos[label] or snap.pos[label]
				n = n + 1
				if n >= 12 then break end
			end
		end
		if sum and n > 2 then
			return sum / n
		end
		return pos
	end
		if mode == "extrapolate" and rec.velocity then
		return pos + rec.velocity * 0.12
	end
	return pos
end

local function chooseHitbox(rec, camPos, lookVec)
	local boxes = hitboxParts(rec)
	if #boxes == 0 then return nil, nil end
	local wanted = CFG.rage.hitbox
	local choice, choicePos, choiceLabel
	if wanted == "nearest" then
				local bestD = math.huge
		for _, hb in ipairs(boxes) do
			local p = hb.part.Position
			local to = p - camPos
			local along = to:Dot(lookVec)
			local off = (to - lookVec * along).Magnitude
			if off < bestD then bestD = off; choice = hb; choicePos = p; choiceLabel = hb.label end
		end
	else
		local wantName = (wanted == "head" and "head")
			or (wanted == "torso" and "chest")
			or "body"
		for _, hb in ipairs(boxes) do
			if hb.label == wantName then choice = hb; choicePos = hb.part.Position; choiceLabel = hb.label; break end
		end
		if not choice then
			choice = boxes[1]
			choicePos = choice.part.Position
			choiceLabel = choice.label
		end
	end
	if not choice then return nil, nil end

		local pos = resolvePosition(rec, choiceLabel, choicePos)
	pos = backtrackPosition(rec, choiceLabel, pos)

		local mp = (CFG.rage.multipoint or 0) / 100
	if mp > 0 then
		local sz = choice.part.Size
		pos = pos + Vector3.new(
			(sz.X * 0.5 * mp) * (math.random() * 2 - 1),
			(sz.Y * 0.5 * mp) * (math.random() * 2 - 1),
			(sz.Z * 0.5 * mp) * (math.random() * 2 - 1)
		)
	end
	return choice.part, pos, choiceLabel
end

local function pointVisibleFromCamera(aimPos, rec)
	local cam = getCam()
	if not cam then return true end
	local origin = cam.CFrame.Position
	local dir = aimPos - origin
	local dist = dir.Magnitude
	if dist < 1 then return true end
	local params = RaycastParams.new()
	params.FilterType = Enum.RaycastFilterType.Exclude
	params.FilterDescendantsInstances = { char or LocalPlayer.Character }
	local hit = WorkspaceService:Raycast(origin, dir, params)
	if hit and hit.Instance and rec.char and hit.Instance:IsDescendantOf(rec.char) then
		return true 	end
	if hit then return false end
	return true
end

local myLastHealth, forcedDefend, forcedDefendUntil = 100, nil, 0

local function pickDefendTarget()
	local bestRec, bestDist = nil, math.huge
	local anyRec, anyDist = nil, math.huge
	for _, rec in pairs(records) do
		local plr = rec.player
		if plr ~= LocalPlayer and rec.char and rec.hum and rec.hum.Health > 0
			and not rec.char:FindFirstChildOfClass("ForceField") then
			local r = rec.char:FindFirstChild("HumanoidRootPart") or rec.char:FindFirstChild("Head")
			if r and root then
				local d = (r.Position - root.Position).Magnitude
				local armed = false
				for _, t in ipairs(rec.char:GetChildren()) do
					if t:IsA("Tool") and (t:GetAttribute("ToolType") == "Gun" or meleeNames[t.Name]) then
						armed = true
						break
					end
				end
				if armed and d < bestDist then
					bestRec, bestDist = rec, d
				end
				if d < anyDist then
					anyRec, anyDist = rec, d
				end
			end
		end
	end
	if bestRec then return bestRec end
	if anyRec and anyDist <= 40 then return anyRec end
	return nil
end

local function onMyHealthChanged(h)
	local old = myLastHealth
	myLastHealth = h
	if h >= old then return end
	if h <= 0 then return end
	if not (CFG.rage.enabled and CFG.misc.autodefend) then return end
	local attacker = nil
	pcall(function()
		local c = human and human:FindFirstChild("creator")
		if c and c.Value and c.Value:IsA("Player") then
			attacker = records[c.Value]
		end
	end)
	if not attacker then attacker = pickDefendTarget() end
	if attacker then
		forcedDefend = attacker
		forcedDefendUntil = os.clock() + (CFG.misc.autodefend_time or 4)
		logEvent("autodefend → " .. attacker.player.Name, COL.red)
	end
end

local function scanTarget()
	if not (CFG.rage.enabled and rageActive())
		and not (CFG.rage.enabled and CFG.misc.autodefend and forcedDefend) then
		return nil
	end
	local cam = getCam()
	if not cam then return nil end
	local camPos = cam.CFrame.Position
	local lookVec = cam.CFrame.LookVector
	if forcedDefend then
		if os.clock() >= forcedDefendUntil then
			forcedDefend = nil
		elseif CFG.misc.autodefend and forcedDefend.char and forcedDefend.hum
			and forcedDefend.hum.Health > 0
			and not forcedDefend.char:FindFirstChildOfClass("ForceField") then
			local fd = forcedDefend
			local part, pos, label = chooseHitbox(fd, camPos, lookVec)
			if part and pos then
				if currentTarget ~= fd then
					currentTarget = fd
					logEvent("autodefend → " .. fd.player.Name, COL.red)
				end
				return fd, part, pos, label
			end
		else
			forcedDefend = nil
		end
	end
	local fovRad = math.rad(CFG.rage.fov or 180) * 0.5

	local best, bestPart, bestPos, bestLabel, bestScore
	local bestMode = CFG.rage.selection
	local bestHealth = math.huge
	local bestDist = math.huge

	for _, rec in pairs(records) do
		if playerIsValidTarget(rec) then
			local part, pos, label = chooseHitbox(rec, camPos, lookVec)
			if part and pos then
				local to = pos - camPos
				local dist = to.Magnitude
				local ang = math.acos(math.clamp(to.Unit:Dot(lookVec), -1, 1))
				if ang <= fovRad and dist < 4096 then
					if CFG.rage.walls == "visible" and not pointVisibleFromCamera(pos, rec) then
											else
						local better = false
						if bestMode == "distance" then
							better = dist < bestDist
						elseif bestMode == "health" then
							local hp = rec.hum and rec.hum.Health or 100
							better = hp < bestHealth
							if hp < bestHealth then bestHealth = hp end
						else 							better = ang < (bestScore or math.huge)
						end
						if better or not best then
							best, bestPart, bestPos, bestLabel = rec, part, pos, label
							bestDist = dist
							if bestMode == "crosshair" then bestScore = ang end
						end
					end
				end
			end
		end
	end
	if best then
		if currentTarget ~= best then
			currentTarget = best
			logEvent("target acquired → " .. best.player.Name, COL.accent)
		end
		return best, bestPart, bestPos, bestLabel
	end
	if currentTarget then currentTarget = nil end
	return nil
end

reloadGun = function()
	local tool = equippedTool
	if not (tool and gunStats) or reloading then return end
	local session = tool:GetAttribute("Local_ReloadSession") or 0
	if session ~= 0 then return end
	if gunStats.Behavior == "Sniper" and (tool:GetAttribute("StoredAmmo") or 0) <= 0 then return end

	reloading = true
	playToolSound(tool, "ReloadSound")
	local sess = os.clock()
	tool:SetAttribute("Local_ReloadSession", sess)
	task.spawn(function()
		if FuncReload then pcall(function() FuncReload:InvokeServer() end) end
		task.wait(CFG.misc.instant_reload and 0.05 or (gunStats.ReloadTime or 2.5))
		if equippedTool == tool and tool:GetAttribute("Local_ReloadSession") == sess then
			if gunStats.Behavior == "Sniper" then
				local stored = tool:GetAttribute("StoredAmmo") or 0
				local cur = tool:GetAttribute("Local_CurrentAmmo") or 0
				if cur < (gunStats.MaxAmmo or 1) then
					tool:SetAttribute("Local_CurrentAmmo", 1)
					tool:SetAttribute("StoredAmmo", math.max(0, stored - 1))
				end
			else
				tool:SetAttribute("Local_CurrentAmmo", gunStats.MaxAmmo or 30)
			end
			updateAmmoLabel()
		end
		if equippedTool == tool then
			tool:SetAttribute("Local_ReloadSession", 0)
		end
		reloading = false
		logEvent("reloaded", COL.text_dim)
	end)
end

local function spreadDirection(origin, target, spreadRadius, behavior)
	local r = math.random()
	if behavior == "Sniper" or behavior == "Shotgun" then
		r = math.sqrt(r)
	end
	local base = CFrame.new(origin, target)
	local roll = 360 - 720 * math.random()
	return (base * CFrame.Angles(0, 0, math.rad(roll)) * CFrame.Angles(0, r * spreadRadius, 0)).LookVector
end

local function computeCameraAimPoint(inputObj)
	local cam = getCam()
	if not cam then return nil, nil end
	local vp = cam.ViewportSize
	local x, y
	if UserInputService.MouseBehavior == Enum.MouseBehavior.LockCenter then
		x, y = vp.X / 2, vp.Y / 2
	elseif inputObj and inputObj.UserInputType == Enum.UserInputType.Touch then
		local inset = GuiService:GetGuiInset()
		x = inputObj.Position.X + inset.X
		y = inputObj.Position.Y + inset.Y - mobileCursorOffset
	else
		local m = UserInputService:GetMouseLocation()
		x, y = m.X, m.Y
	end
	if not x then return nil, nil end
	local ray = cam:ViewportPointToRay(x, y)
	local result = WorkspaceService:Raycast(ray.Origin, ray.Direction * 1500, bulletParams)
	if result then
		return result.Position, result.Instance
	end
	return ray.Origin + ray.Direction * 1500, nil
end

local function touchAllowedToFire(input)
	if HUD.mobileGunFrame and HUD.mobileGunFrame.Visible then
		local ig = HUD.ignoreTouch
		if ig then
			local p, ap, as = input.Position, ig.AbsolutePosition, ig.AbsoluteSize
			if p.X >= ap.X and p.X <= ap.X + as.X and p.Y >= ap.Y and p.Y <= ap.Y + as.Y then
				return false
			end
		end
	end
	if isInsideDynThumb then
		local ok, inside = pcall(isInsideDynThumb, input.Position.X, input.Position.Y)
		if ok and inside then return false end
	end
	return true
end

local function fireOnce(inputObj)
	local tool = equippedTool
	if not (tool and gunStats) then return false end
	local cam = getCam()
	local head = char and (char:FindFirstChild("Head") or char:FindFirstChild("HumanoidRootPart"))
	local origin = head and head.Position or (root and root.Position)
	if not (cam and origin) then return false end

	local behavior = gunStats.Behavior or "Rifle"
	local range = gunStats.Range or 1000
	local pellets = gunStats.ProjectileCount or 1

		local muzzle = nil
	do
		local m = tool:FindFirstChild("Muzzle")
		if m and m:IsA("BasePart") then muzzle = m.Position end
		if not muzzle then
			local h = tool:FindFirstChild("Handle")
			if h and h:IsA("BasePart") then muzzle = h.Position end
		end
		if not muzzle then muzzle = origin end
	end
	local function tracer(pos)
		if not GunTracers then return end
		pcall(function()
			if behavior == "Sniper" then GunTracers.createSniper(muzzle, pos)
			elseif behavior == "Taser" then GunTracers.createTaser(muzzle, pos)
			else GunTracers.createBullet(muzzle, pos) end
		end)
	end
	local function flare()
		pcall(function()
			local h = tool:FindFirstChild("Handle")
			local f = h and h:FindFirstChild("Flare")
			if f then
				f.Enabled = true
				task.delay(0.05, function()
					if f and f.Parent then f.Enabled = false end
				end)
			end
		end)
	end

	local shots = {}
	local targetRec, targetPart, aimPos, targetLabel
	if CFG.rage.enabled and rageActive() then
		targetRec, targetPart, aimPos, targetLabel = scanTarget()
	end

	if targetPart and aimPos then
				if not CFG.rage.silent then
			cameraAimPoint = aimPos
		end
		local willHit = CFG.rage.nevermiss or ((math.random() * 100) <= (CFG.rage.hitchance or 100))
		local shotOrigin = origin
		if willHit and CFG.rage.wallbang then
			local toT = aimPos - origin
			if toT.Magnitude > 2 then
				pcall(function()
					local res = WorkspaceService:Raycast(origin, toT, bulletParams)
					if res and res.Instance and targetRec.char
						and not res.Instance:IsDescendantOf(targetRec.char) then
						shotOrigin = aimPos - toT.Unit * 1
					end
				end)
			end
		end
		local mp = (CFG.rage.multipoint or 0) / 100
		local sz = targetPart.Size
		for i = 1, pellets do
			local pos, inst
			if willHit then
				pos = aimPos
				if mp > 0 then
					pos = pos + Vector3.new(
						sz.X * 0.5 * mp * (math.random() * 2 - 1),
						sz.Y * 0.5 * mp * (math.random() * 2 - 1),
						sz.Z * 0.5 * mp * (math.random() * 2 - 1)
					)
				end
				inst = targetPart
			else
								local off = Vector3.new(
					(math.random() * 2 - 1),
					(math.random() * 2 - 1) * 0.5,
					(math.random() * 2 - 1)
				) * (2 + math.random() * 5)
				local dir = (aimPos + off - origin)
				if dir.Magnitude < 0.1 then dir = cam.CFrame.LookVector end
				local res = WorkspaceService:Raycast(origin, dir.Unit * range, bulletParams)
				pos = res and res.Position or (origin + dir.Unit * range)
				inst = res and res.Instance or nil
				if inst and targetRec.char and inst:IsDescendantOf(targetRec.char) then
					willHit = true 				end
			end
			table.insert(shots, { shotOrigin, pos, inst })
			tracer(pos)
		end
		if willHit then
			pendingHits[targetRec] = { t = os.clock(), hitbox = targetLabel or "body" }
		else
			logEvent("missed " .. targetRec.player.Name .. " (hit chance)", COL.red)
		end
	else
				local aimPoint = computeCameraAimPoint(inputObj)
		if not aimPoint then return false end
		local spreadRadius = gunStats.SpreadRadius or 1
		if behavior == "Sniper" then
			local hum = char and char:FindFirstChildOfClass("Humanoid")
			if hum and hum:GetState() == Enum.HumanoidStateType.Freefall then
				spreadRadius = spreadRadius * 100
			elseif not scoped then
				spreadRadius = spreadRadius * 22
			end
		end
		for i = 1, pellets do
			local dir = spreadDirection(origin, aimPoint, spreadRadius, behavior)
			local res = WorkspaceService:Raycast(origin, dir * range, bulletParams)
			local pos = res and res.Position or (origin + dir * range)
			local inst = res and res.Instance or nil
			table.insert(shots, { origin, pos, inst })
			tracer(pos)
		end
	end

	if #shots == 0 then return false end

		playToolSound(tool, "ShootSound")
	if behavior == "Shotgun" then
		task.delay(0.2, function() playToolSound(tool, "SecondarySound") end)
	end
	flare()

		if ShootEvent then
		ShootEvent:FireServer(shots)
	end

		lastShotAt = os.clock()
	local ammo = (tool:GetAttribute("Local_CurrentAmmo") or 1) - 1
	tool:SetAttribute("Local_CurrentAmmo", ammo)
	updateAmmoLabel()
	if ammo <= 0 and CFG.rage.autoreload then
		task.delay(0.1, reloadGun)
	end
	return true
end

startFiring = function(inputObj, auto)
	if firingLoop then return end
	if not (equippedTool and gunStats) then return end
	if menuOpen then return end
	if equippedTool:GetAttribute("ToolType") ~= "Gun" then return end
	firingLoop = true
	task.spawn(function()
		local tool = equippedTool
		while not dead and equippedTool == tool and tool.Parent == char do
			local hum = char and char:FindFirstChildOfClass("Humanoid")
			if not (hum and hum.Health > 0) then break end
			if reloading then break end
			local ammo = tool:GetAttribute("Local_CurrentAmmo") or 0
			if ammo <= 0 then
				if CFG.rage.autoreload then task.spawn(reloadGun) end
				break
			end
						local holding
			if auto then
				holding = scanTarget() ~= nil and (CFG.rage.autofire or CFG.misc.autodefend)
			else
				holding = (next(heldInputs) ~= nil)
			end
			if not holding then break end
						local delayS = math.max(0.01, (gunStats.FireRate or 0.15) * ((CFG.rage.firetime or 100) / 100))
			local waitS = delayS - (os.clock() - lastShotAt)
			if waitS > 0 then task.wait(waitS) end
			if equippedTool ~= tool or reloading then break end
						if gunStats.Behavior == "Taser" and (gunStats.ChargeTime or 0) > 0 then
				task.wait(gunStats.ChargeTime)
				if equippedTool ~= tool or reloading then break end
			end
			if menuOpen then break end
			fireOnce(inputObj)
						if not (auto or (gunStats.AutoFire ~= false)) then break end
		end
		cameraAimPoint = nil
		firingLoop = false
	end)
end

	gunStats = {
		ToolType       = attrs.ToolType or "Gun",
		Behavior       = attrs.Behavior or "Rifle",
		FireRate       = attrs.FireRate or 0.15,
		MaxAmmo        = attrs.MaxAmmo or 30,
		SpreadRadius   = attrs.SpreadRadius or 1,
		Range          = attrs.Range or 1000,
		ProjectileCount= attrs.ProjectileCount or 1,
		AutoFire       = attrs.AutoFire, 		ReloadTime     = attrs.ReloadTime or 2.5,
		ChargeTime     = attrs.ChargeTime or 0,
	}
	equippedTool = tool
	reloading = false
	scoped = false
	cameraAimPoint = nil
	tool:SetAttribute("Local_CurrentAmmo", attrs.CurrentAmmo or attrs.MaxAmmo or 30)
	tool:SetAttribute("Local_ReloadSession", 0)
		if not vanillaDisabled then
		tool:SetAttribute("Local_IsShooting", true)
	end
	applyGunHUD(true)
	updateAmmoLabel()
		if gunStats.Behavior == "Sniper" then
		pcall(function() UserInputService.MouseIcon = "rbxassetid://98794608762931" end)
	end
	table.insert(toolConns, tool:GetAttributeChangedSignal("StoredAmmo"):Connect(updateAmmoLabel))
	table.insert(toolConns, tool:GetAttributeChangedSignal("CurrentAmmo"):Connect(function()
		local v = tool:GetAttribute("CurrentAmmo")
		if v ~= nil then
			tool:SetAttribute("Local_CurrentAmmo", v)
			updateAmmoLabel()
		end
	end))
	logEvent("equipped " .. tool.Name, COL.text_dim)
end

local function watchCharacter(c)
	connect(c.ChildAdded, function(child)
		if child:IsA("Tool") then
			local t = child:GetAttribute("ToolType")
			if t == nil then
				if meleeNames[child.Name] then onMeleeEquipped(child) end
				task.spawn(function()
					for _ = 1, 20 do
						task.wait(0.25)
						if child.Parent ~= c then return end
						if child:GetAttribute("ToolType") == "Gun" then
							onGunEquipped(child)
							return
						end
					end
				end)
			elseif t == "Gun" then
				onGunEquipped(child)
			elseif meleeNames[child.Name] then
				onMeleeEquipped(child)
			end
		end
	end)
	connect(c.ChildRemoved, function(child)
		if child:IsA("Tool") then onToolRemoved(child) end
	end)
		for _, child in ipairs(c:GetChildren()) do
		if child:IsA("Tool") then
			local t = child:GetAttribute("ToolType")
			if t == "Gun" then onGunEquipped(child)
			elseif meleeNames[child.Name] then onMeleeEquipped(child) end
		end
	end
end

local function knifebotTick()
	if not (CFG.misc.knifebot and equippedTool and meleeNames[equippedTool.Name]) then return end
	if menuOpen then return end
	if os.clock() - lastKnifeAt < 0.65 then return end
	local myRoot = root
	if not myRoot then return end
	local best, bestDist = nil, CFG.misc.knife_range or 7
	for _, rec in pairs(records) do
		if playerIsValidTarget(rec) and rec.char then
			local r = rec.char:FindFirstChild("HumanoidRootPart")
			if r then
				local d = (r.Position - myRoot.Position).Magnitude
				if d <= bestDist then best, bestDist = rec, d end
			end
		end
	end
	if best then
		lastKnifeAt = os.clock()
		local tool = equippedTool
				pcall(function()
			local anim = tool:FindFirstChild("Handle") and tool.Handle:FindFirstChild("Animation")
			local hum = char and char:FindFirstChildOfClass("Humanoid")
			if anim and hum then
				hum:LoadAnimation(anim):Play()
			end
		end)
		local victim = best.player
		task.delay(0.15, function()
			if MeleeEvent and equippedTool == tool then
				pcall(function()
					MeleeEvent:FireServer(victim, tool.Name == "Hammer" and math.random(1, 2) or nil, 1)
				end)
				logEvent("knifed " .. victim.Name, COL.green)
			end
		end)
	end
end

local aaAutoRotateOff = false

local function nearestEnemyRoot()
	local myRoot = root
	if not myRoot then return nil end
	local best, bestD = nil, math.huge
	for _, rec in pairs(records) do
		if playerIsValidTarget(rec) and rec.char then
			local r = rec.char:FindFirstChild("HumanoidRootPart")
			if r then
				local d = (r.Position - myRoot.Position).Magnitude
				if d < bestD then best, bestD = r, d end
			end
		end
	end
	return best
end

local function antiAimTick(dt)
	if dead then return end
	local h, r = human, root
	if not (CFG.aa.enabled and h and r and h.Health > 0) then
		if aaAutoRotateOff and h then
			pcall(function() h.AutoRotate = true end)
			aaAutoRotateOff = false
		end
		return
	end
	if h.SeatPart then return end 	if not aaAutoRotateOff then
		h.AutoRotate = false
		aaAutoRotateOff = true
	end

	local cam = getCam()
	if not cam then return end
	local look = cam.CFrame.LookVector
	local camYaw = atan2(-look.X, -look.Z)
	local yaw = camYaw + math.rad(CFG.aa.yaw or 0)

		if CFG.aa.at_targets then
		local enemy = nearestEnemyRoot()
		if enemy and r then
			local d = (r.Position - enemy.Position)
			if d.Magnitude > 0.05 then
				yaw = atan2(-d.X, -d.Z) + math.rad(CFG.aa.yaw or 0)
			end
		end
	end

		if CFG.aa.spin then
		aaSpinAngle = aaSpinAngle + math.rad(CFG.aa.spin_speed or 420) * dt
		yaw = yaw + aaSpinAngle
	end

		if CFG.aa.jitter ~= "off" then
		aaJitterTimer = aaJitterTimer + dt
		local period = 1 / math.max(1, CFG.aa.jitter_rate or 8)
		if aaJitterTimer >= period then
			aaJitterTimer = 0
			local rng = math.rad(CFG.aa.jitter_range or 60)
			if CFG.aa.jitter == "offset" then
				aaJitterSign = -aaJitterSign
				aaJitterValue = rng * aaJitterSign
			elseif CFG.aa.jitter == "center" then
				aaJitterValue = rng * (math.random() * 2 - 1)
			else 				aaJitterValue = math.pi * (math.random() * 2 - 1)
			end
		end
		yaw = yaw + aaJitterValue
	end

		if flipHeld then yaw = yaw + math.pi end

		local pitch = 0
	if CFG.aa.pitch == "down" then
		pitch = math.rad(-89)
	elseif CFG.aa.pitch == "up" then
		pitch = math.rad(89)
	end

		if CFG.aa.fakelag then
		if aaLagTicks <= 0 then
			aaLagTicks = (CFG.aa.fakelag_mode == "random")
				and math.random(2, math.max(2, CFG.aa.fakelag_ticks or 8))
				or (CFG.aa.fakelag_ticks or 8)
			aaLastYaw = yaw
		else
			aaLagTicks = aaLagTicks - 1
			yaw = aaLastYaw or yaw
		end
	end

		local pos = r.Position
	if (CFG.aa.pos_jitter or 0) > 0 then
		pos = pos + Vector3.new(math.random() * 2 - 1, 0, math.random() * 2 - 1) * CFG.aa.pos_jitter
	end
	r.CFrame = CFrame.new(pos) * CFrame.Angles(pitch, yaw, 0)
end

local function movementTick()
	local h, r = human, root
	if not (h and r and h.Health > 0) then return end
	if menuOpen then return end

		if CFG.misc.bhop ~= "off" then
		local want = false
		if CFG.misc.bhop == "always" then
			want = true
		elseif CFG.misc.bhop == "space" then
			want = UserInputService:IsKeyDown(Enum.KeyCode.Space)
				or UserInputService:IsKeyDown(Enum.KeyCode.ButtonA)
		end
		if want then
			local st = h:GetState()
			if st == Enum.HumanoidStateType.Running or st == Enum.HumanoidStateType.RunningNoPhysics then
				h.Jump = true
			end
		end
	end

		if CFG.misc.autostrafe then
		local st = h:GetState()
		if st == Enum.HumanoidStateType.Freefall then
			local v = r.AssemblyLinearVelocity
			local hv = Vector3.new(v.X, 0, v.Z)
			local spd = hv.Magnitude
			local md = h.MoveDirection
			if spd > 4 and md.Magnitude > 0.1 then
				local nv = Vector3.new(md.X * spd, v.Y, md.Z * spd)
				if not pcall(function() r.Velocity = nv end) then
					pcall(function() r.AssemblyLinearVelocity = nv end)
				end
			end
		end
	end
end

local wingSys = {}
wingSys.state = nil

function wingSys.segC0(side, i, flap, lift)
	return CFrame.new(side * (0.45 + i * 0.28), 1.05 - i * 0.22 + lift, 0.62)
		* CFrame.Angles(0.18 + flap * 0.6, side * -0.5, side * (0.55 - i * 0.06) + flap * side)
end

function wingSys.destroy()
	if wingSys.state then
		pcall(function() wingSys.state.model:Destroy() end)
		wingSys.state = nil
	end
end

function wingSys.build(c)
	wingSys.destroy()
	local hrp = c and c:FindFirstChild("HumanoidRootPart")
	if not hrp then return end
	local model = Instance.new("Model")
	model.Name = "NeverLoseWings"
	local segs = {}
	for side = -1, 1, 2 do
		for i = 1, 4 do
			local seg = Instance.new("Part")
			seg.Name = "Wing"
			seg.Material = Enum.Material.ForceField
			seg.Color = COL.accent
			seg.Transparency = 0.45
			seg.Size = Vector3.new(0.12, 0.9 - i * 0.14, 0.42 + i * 0.34)
			seg.CanCollide = false
			seg.CanQuery = false
			seg.CanTouch = false
			seg.CastShadow = false
			seg.Massless = true
			seg.CFrame = hrp.CFrame
			local weld = Instance.new("Weld")
			weld.Name = "WingWeld"
			weld.Part0 = hrp
			weld.Part1 = seg
			weld.C0 = wingSys.segC0(side, i, 0, 0)
			weld.C1 = CFrame.new()
			weld.Parent = seg
			local sp = Instance.new("Sparkles")
			sp.SparkleColor = COL.accent
			sp.Enabled = true
			sp.Parent = seg
			seg.Parent = model
			segs[#segs + 1] = { weld = weld, side = side, idx = i }
		end
	end
	model.Parent = c
	wingSys.state = { model = model, segs = segs, t = 0 }
end

function wingSys.update(dt)
	local st = wingSys.state
	if not st then return end
	st.t = st.t + (dt or 0.016)
	for _, s in ipairs(st.segs) do
		local ph = st.t * 2.4 + (s.idx - 1) * 0.55
		local flap = math.sin(ph) * 0.16
		local lift = math.sin(ph - 0.7) * 0.06
		s.weld.C0 = wingSys.segC0(s.side, s.idx, flap, lift)
	end
end

		end

	if dead then return end

		lastHudCheck = os.clock()
		if not HUD.hud or HUD.hud.Parent == nil or not HUD.actionArea then
			task.spawn(function()

local function renderESP()
	if not espRoot then return end
	local cam = getCam()
	if not cam then return end
	local camPos = cam.CFrame.Position
	local style = CFG.vis.box

	for _, rec in pairs(records) do
		local plr = rec.player
		local show = CFG.vis.esp
			and rec.char and rec.hum and rec.hum.Health > 0
			and (CFG.vis.teammates or plr.Team ~= LocalPlayer.Team)
			and plr ~= LocalPlayer
		if show and rec.esp == nil then createESP(rec) end
		if rec.esp then
			local obj = rec.esp
			if not show then
				obj.frame.Visible = false
			else
				local hrp = rec.char:FindFirstChild("HumanoidRootPart")
				if hrp then
					local dist = (hrp.Position - camPos).Magnitude
					if dist > (CFG.vis.maxdist or 1500) or dist < 2 then
						obj.frame.Visible = false
					else
						local topW = hrp.Position + Vector3.new(0, 3.15, 0)
						local botW = hrp.Position - Vector3.new(0, 3.15, 0)
						local tv, tOn = cam:WorldToViewportPoint(topW)
						local bv, bOn = cam:WorldToViewportPoint(botW)
						if (tOn or bOn) and tv.Z > 0 and bv.Z > 0 then
							local h = tv.Y - bv.Y
							local w = h * 0.45
							if h >= 5 and w >= 3 then
								local cx = (tv.X + bv.X) / 2
								local x, y = cx - w / 2, tv.Y
								obj.frame.Position = UDim2.new(0, math.floor(x + 0.5), 0, math.floor(y + 0.5))
								obj.frame.Size = UDim2.new(0, math.floor(w + 0.5), 0, math.floor(h + 0.5))
								obj.frame.Visible = true
																local boxCol = COL.white
								if CFG.vis.teamcolors and plr.Team ~= nil then
									boxCol = plr.TeamColor.Color
								end
								local isTarget = (rec == currentTarget) and CFG.vis.target_indicator
								if style ~= "off" then
									setCornerBox(obj, 0, 0, math.floor(w + 0.5), math.floor(h + 0.5), isTarget and COL.accent or boxCol, style)
									for i = 1, 8 do obj.lines[i].Visible = true end
								else
									for i = 1, 8 do obj.lines[i].Visible = false end
								end
								obj.stroke.Enabled = (isTarget == true)
																if CFG.vis.name then
									obj.name.Text = plr.DisplayName .. " [@" .. plr.Name .. "]"
									obj.name.TextColor3 = isTarget and COL.accent or COL.white
									obj.name.Visible = true
								else
									obj.name.Visible = false
								end
																if CFG.vis.health then
									local frac = math.clamp(rec.hum.Health / math.max(1, rec.hum.MaxHealth), 0, 1)
									obj.hpBg.Visible = true
									obj.hpFill.Size = UDim2.new(1, 0, frac, 0)
									obj.hpFill.BackgroundColor3 = Color3.new(1 - frac * 0.7, 0.35 + frac * 0.5, 0.25)
								else
									obj.hpBg.Visible = false
								end
																if CFG.vis.weapon then
									local t = rec.char:FindFirstChildOfClass("Tool")
									obj.weapon.Text = t and t.Name or ""
									obj.weapon.Visible = t ~= nil
								else
									obj.weapon.Visible = false
								end
																if CFG.vis.distance then
									obj.dist.Text = math.floor(dist + 0.5) .. "m"
									obj.dist.Visible = true
								else
									obj.dist.Visible = false
								end
							else
								obj.frame.Visible = false
							end
						else
							obj.frame.Visible = false
						end
					end
				else
					obj.frame.Visible = false
				end
			end
		end
	end
end

				rec.highlight.OutlineColor = Color3.new(0, 0, 0)
				rec.highlight.OutlineTransparency = 0.2
				rec.highlight.Parent = rec.char
			end)
		elseif not want and rec.highlight then
			pcall(function() rec.highlight:Destroy() end)
			rec.highlight = nil
		end
		if rec.highlight and rec.char then
			local cam = getCam()
			local hrp = rec.char:FindFirstChild("HumanoidRootPart")
			if cam and hrp then
				local params = RaycastParams.new()
				params.FilterType = Enum.RaycastFilterType.Exclude
				params.FilterDescendantsInstances = { char or LocalPlayer.Character, rec.char }
				local dir = hrp.Position - cam.CFrame.Position
				local hit = WorkspaceService:Raycast(cam.CFrame.Position, dir, params)
				local visible = (hit == nil)
				rec.highlight.FillColor = visible
					and col3(CFG.vis.chams_visible)
					or col3(CFG.vis.chams_occluded)
				rec.highlight.FillTransparency = (CFG.vis.chams_fill or 60) / 100
			end
		end
	end
end

local function renderHUD()
	local cam = getCam()
	if not cam then return end

		fpsCount = fpsCount + 1
	local now = os.clock()
	if now - fpsClock >= 1 then
		fpsValue = fpsCount
		fpsCount = 0
		fpsClock = now
		rebuildKeybindList()
	end

		if watermarkFrame then
		watermarkFrame.Visible = CFG.vis.watermark
		if CFG.vis.watermark then
			local ping = "?"
			pcall(function()
				ping = math.floor(StatsService.Network.ServerStatsItem["Data Ping"]:GetValue())
			end)
			watermarkLabel.Text = ("NeverLose | %s | %d fps | %s ms"):format(
				LocalPlayer.DisplayName, fpsValue, tostring(ping))
		end
	end

		if fovCircle then
		local show = CFG.vis.fovcircle and CFG.rage.enabled
		fovCircle.Visible = show
		if show then
			local vp = cam.ViewportSize
			local r = math.tan(math.rad(math.clamp(CFG.rage.fov or 180, 1, 179)) / 2)
				/ math.tan(math.rad(math.clamp(cam.FieldOfView, 1, 120)) / 2)
				* (vp.Y / 2)
			r = math.clamp(r, 4, math.max(vp.X, vp.Y))
			fovCircle.Size = UDim2.new(0, r * 2, 0, r * 2)
		end
	end

		local hm = os.clock() < hitmarkerUntil
	for _, l in ipairs(hitmarkerLines) do
		l.Visible = hm
	end

		if aaArrow then
		local show = CFG.aa.enabled and CFG.aa.arrows and root and human and human.Health > 0
		aaArrow.Visible = show
		if show then
			local yaw = aaLastYaw
			if CFG.aa.fakelag and aaLagTicks > 0 and aaLastYaw then
				yaw = aaLastYaw
			else
				local look = cam.CFrame.LookVector
				yaw = atan2(-look.X, -look.Z) + math.rad(CFG.aa.yaw or 0)
			end
			local base = root.Position + Vector3.new(0, -2.4, 0)
			local tip = base + Vector3.new(-math.sin(yaw), 0, -math.cos(yaw)) * 3.2
			local bv, bOn = cam:WorldToViewportPoint(base)
			local tv, tOn = cam:WorldToViewportPoint(tip)
			if bOn and tOn and bv.Z > 0 and tv.Z > 0 then
				local dx, dy = tv.X - bv.X, tv.Y - bv.Y
				local len = math.sqrt(dx * dx + dy * dy)
				if len > 4 then
					aaArrow.Size = UDim2.new(0, math.max(2, math.min(len, 46)), 0, 2)
					aaArrow.Position = UDim2.new(0, (bv.X + tv.X) / 2, 0, (bv.Y + tv.Y) / 2)
					aaArrow.Rotation = -math.deg(atan2(dy, dx))
				else
					aaArrow.Visible = false
				end
			else
				aaArrow.Visible = false
			end
		end
	end
end

local function addSlider(parent, label, min, max, get, set, fmt)
	local row = mkRow(parent, 32)
	mkLabel(row, label)
	local valueLabel = newInst("TextLabel", {
		BackgroundTransparency = 1, Font = FONT_MED, TextSize = 12,
		TextColor3 = COL.white, Text = "", TextXAlignment = Enum.TextXAlignment.Right,
		AnchorPoint = Vector2.new(1, 0), Position = UDim2.new(1, -10, 0, 0),
		Size = UDim2.new(0, 64, 0, 14),
	}, row)
	local bar = newInst("Frame", {
		BackgroundColor3 = COL.track, BorderSizePixel = 0,
		AnchorPoint = Vector2.new(0, 0.5), Position = UDim2.new(0.45, 8, 0.62, 0),
		Size = UDim2.new(0.55, -86, 0, 4),
	}, row)
	newInst("UICorner", { CornerRadius = UDim.new(1, 0) }, bar)
	local fill = newInst("Frame", {
		BackgroundColor3 = COL.accent, BorderSizePixel = 0,
		Size = UDim2.new(0, 0, 1, 0),
	}, bar)
	newInst("UICorner", { CornerRadius = UDim.new(1, 0) }, fill)
	local knob = newInst("Frame", {
		BackgroundColor3 = COL.white, BorderSizePixel = 0,
		AnchorPoint = Vector2.new(0.5, 0.5), Position = UDim2.new(0, 0, 0.5, 0),
		Size = UDim2.new(0, 10, 0, 10), ZIndex = 2,
	}, bar)
	newInst("UICorner", { CornerRadius = UDim.new(1, 0) }, knob)

	local dragging = false
	local function alphaToValue(a)
		a = math.clamp(a, 0, 1)
		return min + (max - min) * a
	end
	local function sync()
		local v = math.clamp(get(), min, max)
		local a = (v - min) / (max - min)
		fill.Size = UDim2.new(a, 0, 1, 0)
		knob.Position = UDim2.new(a, 0, 0.5, 0)
		valueLabel.Text = fmt and fmt(v) or string.format("%g", v)
	end
	local function updateFromX(x)
		local rel = (x - bar.AbsolutePosition.X) / math.max(1, bar.AbsoluteSize.X)
		set(alphaToValue(rel))
		sync()
	end
	bar.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			updateFromX(input.Position.X)
		end
	end)
	UserInputService.InputChanged:Connect(function(input)
		if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
			or input.UserInputType == Enum.UserInputType.Touch) then
			updateFromX(input.Position.X)
		end
	end)
	UserInputService.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			dragging = false
		end
	end)
	addSync(sync)
	sync()
	return row
end

				local function serializeConfig()
		local function keyStr(k)
			return (typeof(k) == "EnumItem") and k.Name or nil
		end
		local copy = deepCopy(CFG)
		copy.menu_key = keyStr(copy.menu_key)
		copy.rage.key = keyStr(copy.rage.key)
		copy.aa.flip_key = keyStr(copy.aa.flip_key)
		return HttpService:JSONEncode(copy)
	end

	local function importConfig(json)
		local data = HttpService:JSONDecode(json)
		if type(data) ~= "table" then return false end
		local function strToKey(s)
			if type(s) ~= "string" or s == "" then return nil end
			local ok, k = pcall(function() return Enum.KeyCode[s] end)
			return ok and k or nil
		end
		if data.rage and type(data.rage.key) == "string" then data.rage.key = strToKey(data.rage.key) end
		if data.aa and type(data.aa.flip_key) == "string" then data.aa.flip_key = strToKey(data.aa.flip_key) end
		if type(data.menu_key) == "string" then data.menu_key = strToKey(data.menu_key) or Enum.KeyCode.Insert end
		local fresh = deepCopy(DEFAULTS)
		deepMerge(fresh, data)
				for k in pairs(CFG) do CFG[k] = nil end
		for k, v in pairs(fresh) do CFG[k] = v end
		refreshAllControls()
		return true
	end

	local function saveConfig()
		local sv = guiRoot:FindFirstChild("NLHvH_Config")
		if not sv then
			sv = Instance.new("StringValue")
			sv.Name = "NLHvH_Config"
			sv.Parent = guiRoot
		end
		sv.Value = serializeConfig()
		logEvent("config saved", COL.green)
	end
	saveConfigNow = saveConfig

	local function loadSavedConfig(silent)
		local sv = guiRoot:FindFirstChild("NLHvH_Config")
		if sv and sv.Value ~= "" then
			local ok = pcall(importConfig, sv.Value)
			if ok and not silent then logEvent("config loaded", COL.green) end
			return ok
		end
		return false
	end

connect(UserInputService.InputBegan, function(input, gameProcessed)
		if capturingBind then
		if input.UserInputType == Enum.UserInputType.Keyboard then
			local cb = capturingBind
			capturingBind = nil
			capturingBindBtn = nil
			if cb then cb(input.KeyCode) end
		elseif input.UserInputType == Enum.UserInputType.MouseButton1 then
			capturingBind = nil
			capturingBindBtn = nil
			refreshAllControls()
		end
		return
	end

	if typingInMenu then return end

		if CFG.menu_key and input.UserInputType == Enum.UserInputType.Keyboard
		and keyEquals(input.KeyCode, CFG.menu_key) then
		setMenuOpen(not menuOpen)
		return
	end

	if menuOpen then return end

		if CFG.rage.key and input.UserInputType == Enum.UserInputType.Keyboard
		and keyEquals(input.KeyCode, CFG.rage.key) and CFG.rage.key_mode == "toggle" then
		rageToggleState = not rageToggleState
		logEvent("ragebot " .. (rageToggleState and "enabled" or "disabled") .. " (key)", COL.accent)
	end

		if CFG.aa.flip_key and input.UserInputType == Enum.UserInputType.Keyboard
		and keyEquals(input.KeyCode, CFG.aa.flip_key) then
		flipHeld = true
		return
	end

	if gameProcessed then return end

		if equippedTool and gunStats and equippedTool:GetAttribute("ToolType") == "Gun" then
		if input.UserInputType == Enum.UserInputType.MouseButton1 then
			heldInputs[input] = true
			startFiring(input)
		elseif input.UserInputType == Enum.UserInputType.MouseButton2 then
			if UserInputService.MouseBehavior == Enum.MouseBehavior.LockCenter
				and gunStats.Behavior == "Sniper" then
				setScope(not scoped)
			end
		elseif input.UserInputType == Enum.UserInputType.Touch then
			if touchAllowedToFire(input) then
				heldInputs[input] = true
				startFiring(input)
			end
		elseif input.UserInputType == Enum.UserInputType.Keyboard then
			if input.KeyCode == Enum.KeyCode.R then
				reloadGun()
			elseif input.KeyCode == Enum.KeyCode.Z and gunStats.Behavior == "Sniper" then
				setScope(not scoped)
			end
		elseif input.UserInputType == Enum.UserInputType.Gamepad1 then
			if input.KeyCode == Enum.KeyCode.ButtonR2 then
				heldInputs[input] = true
				startFiring(input)
			elseif input.KeyCode == Enum.KeyCode.ButtonX then
				reloadGun()
			elseif input.KeyCode == Enum.KeyCode.ButtonL2 and gunStats.Behavior == "Sniper" then
				setScope(not scoped)
			end
		end
	end
end)

unload = function()
	wingSys.destroy()
	pcall(function() emoteSys.stop() end)
	if dead then return end
	dead = true
		pcall(function() saveConfigNow() end)
	for _, c in ipairs(conns) do pcall(function() c:Disconnect() end) end
	for _, n in ipairs(renderBinds) do pcall(function() RunService:UnbindFromRenderStep(n) end) end
	clearToolConns()
		pcall(function()
		local c = LocalPlayer.Character
		if c then
			local h = c:FindFirstChildOfClass("Humanoid")
			if h then h.AutoRotate = true end
		end
		LocalPlayer.CameraMode = Enum.CameraMode.Classic
		local cam = getCam()
		if cam then cam.FieldOfView = 70 end
		UserInputService.MouseBehavior = Enum.MouseBehavior.Default
		UserInputService.MouseIconEnabled = true
		UserInputService.MouseIcon = ""
		_G.cameraSensitivity = 1
	end)
		for _, rec in pairs(records) do
		destroyRecordESP(rec)
	end
	records = {}
		pcall(function() if espGui then espGui:Destroy() end end)
	pcall(function() if hudGui then hudGui:Destroy() end end)
	pcall(function() if guiRoot then guiRoot:Destroy() end end)
		if equippedTool then
		pcall(function() equippedTool:SetAttribute("Local_IsShooting", false) end)
	end
	if vanillaController then
		pcall(function() vanillaController.Disabled = false end)
	end
	warn("[neverlose] unloaded — vanilla GunController re-enabled")
end
